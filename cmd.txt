instanceFilter:
  all: true

rollout:
  disruptionBudget:
    fixed: 10
  minWaitDuration: 30s

osPolicies:
- id: enable-gceagent
  mode: ENFORCEMENT

  resourceGroups:
  - resources:
    - id: start-gceagent
      exec:
        validate:
          interpreter: POWERSHELL
          script: |
            $svc = Get-Service GCEAgent -ErrorAction SilentlyContinue
            if ($svc -and $svc.Status -eq "Running") {
              exit 100
            }
            exit 101

        enforce:
          interpreter: POWERSHELL
          script: |
            Set-Service GCEAgent -StartupType Automatic
            Start-Service GCEAgent
