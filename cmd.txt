# 1. Clone the repo
git clone <repo-url>
cd <repo-name>

# 2. Make sure you're on the right branch
git checkout <branch-name>   # e.g. main, or whatever branch has win2022-dev

# 3. Copy the directory with the new name
cp -r win2022-dev win2025-dev

# 4. Stage the new folder
git add win2025-dev

# 5. Commit
git commit -m "Copy win2022-dev to win2025-dev"

# 6. Push to SSM
git push origin <branch-name>
