def extract_nvd_cvss_score(metrics: dict, key: str):
    """
    Extract CVSS baseScore from the specified metrics dictionary.

    Parameters:
        metrics (dict): Dictionary containing CVSS metric data.
        key (str): Metric key to extract the baseScore from.

    Returns:
        base_score (float or None): The CVSS baseScore if available; otherwise, None.
    """   
    metric_list = metrics.get(key, [])
    if metric_list and isinstance(metric_list, list) and 'cvssData' in metric_list[0]:
        return metric_list[0]['cvssData'].get('baseScore')
    return None

def get_nvd_data_with_retries(url: str, start_index: int):
    """
        Fetch CVE records from the NVD CVE 2.0 API with pagination and retry logic.
        Parameters:
            url (str): Endpoint URL for the NVD CVE 2.0 API.
            start_index (int): Index to start pagination from.
        Returns:
            records (list): List of CVE records retrieved from the API.
            complete (bool): True if all records have been fetched; otherwise, False.
            next_start_index (int): Index to use for the next page of results.
        """
    max_retries = 5        
    records_per_page = 2000 
    records = []
    total_results = None  

    while True:
        params = {
            "resultsPerPage": records_per_page, 
            "startIndex": start_index,            
        }

        success = False

        for attempt in range(max_retries):
            try:
                response = requests.get(
                    url,
                    params=params,
                    timeout=60
                )
                if response.status_code in (403, 429):
                    wait_time = 60
                    logger.info(
                        f"Received {response.status_code} at index {start_index}. "
                        f"Sleeping {wait_time}s and retrying..."
                    )
                    time.sleep(wait_time)
                    continue

                response.raise_for_status()
                data = response.json()

                # Read totalResults from the first successful page
                if total_results is None:
                    total_results = data.get("totalResults", 0)
                    logger.info(f"Total results reported by API: {total_results}")
                    

                vulnerabilities = data.get("vulnerabilities", [])

                # If the API returns no vulnerabilities, assume we are done
                if not vulnerabilities:
                    logger.info("No more data returned by API. Stopping.")
                    success = True
                    break

                # Process and store CVE records
                for item in vulnerabilities:
                    cve = item.get("cve", {})

                    metrics = cve.get("metrics", {})

                    # CVSS v3: prefer V31, fallback to V30
                    cvss_v3_score = extract_nvd_cvss_score(metrics, 'cvssMetricV31')
                    if cvss_v3_score is None:
                        cvss_v3_score = extract_nvd_cvss_score(metrics, 'cvssMetricV30')

                    record = {
                        "cve_id": cve.get("id"),
                        "published": cve.get("published"),
                        "last_modified": cve.get("lastModified"),
                        "cvss_v2_score": extract_nvd_cvss_score(metrics, 'cvssMetricV2'),
                        "cvss_v3_score": cvss_v3_score,
                        "cvss_v4_score": extract_nvd_cvss_score(metrics, 'cvssMetricV40')
                    }
                    records.append(record)                 
                logger.info(
                    f"Progress: {len(records)} / {total_results} "
                    f"(startIndex={start_index})"
                )                             

                start_index +=  records_per_page 
                success = True
                time.sleep(6)
                break  # exit retry loop on success

            except (requests.exceptions.ReadTimeout,
                    requests.exceptions.ConnectionError) as e:
                wait_time = (attempt + 1) * 10
                logger.info(
                    f"Network issue at index {start_index}: {e}. "
                    f"Retry {attempt+1}/{max_retries} in {wait_time}s..."
                )
                time.sleep(wait_time)

            except Exception as e:
                logger.error(f"Unexpected error at index {start_index}: {e}")
                success = False
                break

        # If after all retries we still failed, stop and return what we have
        if not success:
            logger.info(
                f"Failed to fetch index {start_index} after {max_retries} attempts. "
                "Returning partial results."
            )
            break

        if total_results is not None and start_index >= total_results:
            logger.info("Completed. All records reported by API have been fetched.")
            break
    return records, success, start_index

def fetch_nvd_data():
    """
    Fetch CVE records from the NVD CVE 2.0 API with pagination and retry logic.

    Parameters:
        url (str): Endpoint URL for the NVD CVE 2.0 API.

    Returns:
        pd.DataFrame or None: Normalized DataFrame of CVE records if data is found; otherwise, None.
    """
 
    url = "https://services.nvd.nist.gov/rest/json/cves/2.0"

    logger.info(f"Fetching NVD CVE data from {url}")

    all_records = []
    start_index = 0
    max_retries =5
    retry_count = 0

    while True:

        records, complete, next_start_index  = get_nvd_data_with_retries(
             url=url,
             start_index=start_index
        )           
        all_records.extend(records) 
        if complete:
            break
        
        if next_start_index == start_index:
            retry_count += 1
            if retry_count >= max_retries:
                logger.error("Max retries reached. Breaking loop to avoid infinite loop.")
                raise RuntimeError("Error Fetching data.") 
        else:
            retry_count = 0        

        logger.info(f"Partial records returned.Retrying to fetch remaining.")
        start_index = next_start_index
        time.sleep(30)

    if all_records:
        logger.info(f"Totol records fetched: {len(all_records)}")
        return pd.json_normalize(all_records)
    else: 
        return None

def fetch_cisa_kev_catalog(engine: Engine):
    """
    Fetch and normalize CISA Known Exploited Vulnerabilities (KEV) catalog data.

    Returns:
        pd.DataFrame or None: Normalized DataFrame of KEV records if data is found; otherwise, None.

    """
    url ="https://www.cisa.gov/sites/default/files/feeds/known_exploited_vulnerabilities.json"
    records = []
    try:
        resp = requests.get(url, timeout=60)
        resp.raise_for_status()
        data = resp.json()
        vulnerabilities = data.get("vulnerabilities", [])
        if not vulnerabilities:
            logger.warning("No more data returned by API. Stopping.")
            return None
        
        for item in vulnerabilities:
            record = {
                "cve_id": item.get("cveID"),
                "date_added": item.get("dateAdded"),
                "due_date": item.get("dueDate"),
                "is_kev": True
            }
            records.append(record)            
    except requests.RequestException as e:
        logger.error(f"Error fetching KEV data: {e}")
        raise
    df =  pd.json_normalize(records)

    logger.info("Loading CISA KEV Catalog data to database...")
    df["etl_date"] = datetime.now().date() 
    schema="crma_hive"
    table_name = "crma_hive_cisa_kev_catalog"
    kev_df = df.drop("is_kev", axis=1)
    rw_util.bulk_upsert_with_copy(
        engine=engine,
        df=kev_df,
        schema=schema,
        target_table=table_name,
        pk_columns=["cve_id"]
    )

    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    logger.info(f"Loading table '{table_name}' to '{schema}' schema")     
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
    rw_util.bulk_upsert_with_copy(
        engine=engine2,
        df=kev_df,
        schema=schema,
        target_table=table_name,
        pk_columns=["cve_id"]
    )


    return df

def process_and_load_nvd_cisa_cve_data(df: pd.DataFrame, bucket_name: str, engine: Engine):
    """
    Clean and standardize CVE data, archive it to S3, and load it into a database.

    Parameters:
        df (pd.DataFrame): DataFrame containing CVE records to process.
        bucket_name (str): Name of the S3 bucket for archival.
        engine (Engine): SQLAlchemy Engine for database connection.

    Returns:
        None
    """
    logger.info("Standardizing date, timestamp, numeric and boolean fields.")
    cols = ["date_added", "due_date"]    
    df = dataQualityUtils.clean_date_fields(df, cols) 
    cols = ["published", "last_modified"]    
    df = dataQualityUtils.clean_timestamp_fields(df, cols)
    cols = ["cvss_v2_score", "cvss_v3_score", "cvss_v4_score"]
    df = dataQualityUtils.clean_float_fields(df, cols) 
    cols = ["is_kev"]
    df = dataQualityUtils.normalize_boolean_fields(df, cols)

    logger.info("Storing a copy in S3 bucket for archival...")
    filename = "raw_data/NVD_CISA/CRMA_HIVE_CVE_SCORE_KEV_INFO_COMBINED" 
    rw_util.load_to_s3(bucket_name, df, filename, "PARQUET")

    logger.info("Loading data to database...")
    df["etl_date"] = datetime.now().date() 
    schema="crma_hive"
    table_name = "crma_hive_cve_score_kev_info_combined"
    rw_util.bulk_upsert_with_copy(
        engine=engine,
        df=df,
        schema=schema,
        target_table=table_name,
        pk_columns=["cve_id"]
    )

    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    logger.info(f"Loading table '{table_name}' to '{schema}' schema")     
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
    rw_util.bulk_upsert_with_copy(
        engine=engine2,
        df=df,
        schema=schema,
        target_table=table_name,
        pk_columns=["cve_id"]
    )

def main():

    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 
    
    logger.info("Fetching KEV info from CISA")
    cisa_kev_df = fetch_cisa_kev_catalog(engine)

    logger.info("Fetching cve scores from NVD...")
    nvd_cve_score_df = fetch_nvd_data()

    
    logger.info("Combine datasets")
    df = pd.merge(
        nvd_cve_score_df,
        cisa_kev_df,
        how='left',
        left_on='cve_id',
        right_on='cve_id'
    )

    logger.info("Clean and load data to distination")
    process_and_load_nvd_cisa_cve_data(df, bucket_name, engine)

      
    utility.log_memory_usage() 
    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")


if __name__ == "__main__":
    main()

def load_to_processed(
        bucket_name: str,
        dataset: str,
        clean_df: pd.DataFrame,

        warning_df: Optional[pd.DataFrame] = None,
        backfill_date: Optional[str] = None

    """

        rw_util.load_to_s3(bucket_name, clean_df, target_filename, "PARQUET", True)

            prefix = "processed/CMDB/QUARANTINE/"
            s3_filename = f"CRMA_HIVE_{dataset}_QUARANTINE"

            s3_filename = f"CRMA_HIVE_{dataset}_WARNING"        
            target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, s3_filename, backfill_date, "PARQUET")

        output_file= f"processed/CMDB/CLEAN/CRMA_HIVE_{dataset}_CLEAN"

def run_backfill_mode(bucket_name: str, hist_table_name: str, backfill_date: str, datasets: list) -> None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    # check valid backfill date format
    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise
    
    for dataset in datasets:
        dataset_upper = dataset.upper().strip()

        # call processors with backfill_date
        if dataset_upper == "CMDB_CI_COMPUTER":
            logger.info("Processing CMDB_CI_COMPUTER (in backfill mode)")
            process_cmdb_ci_computer(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CMDB_CI_SERVER":
            logger.info("Processing CMDB_CI_SERVER (in backfill mode)")
            process_cmdb_ci_server(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CMDB_CI_DATABASE":
            logger.info("Processing CMDB_CI_DATABASE (in backfill mode)")
            process_cmdb_ci_database(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CMDB_CI_NETWORK":
            logger.info("Processing CMDB_CI_NETWORK (in backfill mode)")
            process_cmdb_ci_network(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CMDB_CI_VM_INSTANCE":
            logger.info("Processing CMDB_CI_VM_INSTANCE (in backfill mode)")
            process_cmdb_ci_vm_instance(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CMDB_CI_SERVICE":
            logger.info("Processing CMDB_CI_SERVICE (in backfill mode)")
            process_cmdb_ci_service(bucket_name, backfill_date)
            utility.log_memory_usage() 

def main():
    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]

    if not bucket_name:
       raise ValueError("S3 bucket not found!")

    # resolve workflow run properties
    workflow_params = utility.get_workflow_props()
    backfill_date = workflow_params.get("--backfill_date")
    hist_table_name = workflow_params.get("--hist_table_name")
    datasets = utility.parse_datasets_list( workflow_params.get("--datasets",""))

    # run in backfill mode if all backfill properties are present
    backfill_mode = bool(backfill_date and hist_table_name)

    if backfill_mode:
        run_backfill_mode(bucket_name, hist_table_name, backfill_date, datasets)
    else:
        logger.info(f"Running in Normal mode.")

        logger.info("Processing CMDB_CI_COMPUTER")
        process_cmdb_ci_computer(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing CMDB_CI_SERVER")
        process_cmdb_ci_server(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing CMDB_CI_DATABASE")
        process_cmdb_ci_database(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing CMDB_CI_NETWORK")
        process_cmdb_ci_network(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing CMDB_CI_VM_INSTANCE")
        process_cmdb_ci_vm_instance(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing CMDB_CI_SERVICE")

def load_data_to_db(
        engine: Engine,
        bucket_name: str,
        dataset: str,
        schema: str,
        backfill_date: Optional[str] = None        
) -> None:
    """
    Loads data to desttination database
    
    Parameters:
        engine (create_engine): Database connection engine
        bucket (str): Name of the S3 bucket to fetch the data from
        dataset (str): The dataset to store to database
        schema(str): The database schema in which data is loaded.
        backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525)        
    """
    logger.info(f"Preparing to load table '{dataset}' to '{schema}' schema")

    classes = ["CLEAN", "QUARANTINE", "WARNING"]

    for cls in classes:
        prefix = f"processed/CMDB/{cls}/CRMA_HIVE"
        base_name = f"{prefix}_{dataset}_{cls}"    
        
        if backfill_date:
            df = rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")
            if df is not None:
                df["etl_date"] = datetime.strptime(backfill_date, "%m%d%y").strftime("%Y-%m-%d")
        else:
            df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")
            if df is not None:
                logger.info(f"Total records retrieved: {len(df)}")
                df["etl_date"] = datetime.now().date() 

        if cls == "CLEAN":
            if df is None:
                raise ValueError("Dataset not found")
            table_name = f"crma_hive_{dataset.lower()}"
            logger.info(f"Loading table '{table_name}' to '{schema}' schema")
            rw_util.truncate_and_load_table(engine=engine, df=df, schema_name=schema, table_name=table_name, exist_type="append" )
        else:
            if df is not None:                    
                table_name = f"crma_hive_{dataset.lower()}_{cls.lower()}"
                logger.info(f"Loading table '{table_name}' to '{schema}' schema")     
                rw_util.truncate_and_load_table(engine=engine, df=df, schema_name=schema, table_name=table_name, exist_type="append" )

        if df is not None: 
            logger.info(f"Perfomming post load validation on table '{table_name}' in '{schema}' schema")
            if rw_util.post_load_validation(engine=engine, df=df, schema_name=schema, table_name=table_name):
                logger.info(f"Post load validation check passed for table '{table_name}' in '{schema}' schema")

                ##Parallel loading##
                logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
                logger.info(f"Loading table '{table_name}' to '{schema}' schema")     
                engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
                rw_util.truncate_and_load_table(engine=engine2, df=df, schema_name=schema, table_name=table_name, exist_type="append" )

            else:
                raise RuntimeError(f"Post load validation check failed for table '{table_name}' in '{schema}' schema")                      

def run_backfill_mode(engine: Engine, schema: str, bucket_name: str, hist_table_name: str, backfill_date: str, datasets: list) -> None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        engine (str): SQLalchemy engine
        schema (str): The database schema in which data is loaded.
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    # check valid backfill date format
    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise

    # enforce valid tables in datasets list when running in backfill mode
    valid_datasets = [t for t in datasets if t in VALID_TABLES]

    if len(valid_datasets) != len(datasets):
        invalid_tables = set(datasets) - set(valid_datasets)
        logger.warning(f"Invalid tables ignored: {invalid_tables}. Valid tables used: {valid_datasets}")
        datasets = valid_datasets
    else:
        logger.info(f"All {len(datasets)} tables in dataset are valid")

    if not datasets:
        logger.warning(F"No CMDB tables found!")

    for dataset in datasets:
        load_data_to_db(engine, bucket_name, dataset, schema, backfill_date)

    datasets = utility.parse_datasets_list( workflow_params.get("--datasets",""))

    # run in backfill mode if all backfill properties are present
    backfill_mode = bool(backfill_date and hist_table_name)

    if backfill_mode:
        schema = BACKFILL_SCHEMA
        run_backfill_mode(engine, schema, bucket_name, hist_table_name, backfill_date, datasets)     
    else:
        logger.info(f"Running in Normal mode.")  
        datasets = [
                    TABLE_CRMA_HIVE_CMDB_CI_COMPUTER, 
                    TABLE_CRMA_HIVE_CMDB_CI_SERVER,
                    TABLE_CRMA_HIVE_CMDB_CI_DATABASE,
                    TABLE_CRMA_HIVE_CMDB_CI_NETWORK,
                    TABLE_CRMA_HIVE_CMDB_CI_VM_INSTANCE,
                    TABLE_CRMA_HIVE_CMDB_CI_SERVICE
                ]

        for dataset in datasets:
            load_data_to_db(engine = engine, bucket_name = bucket_name, dataset=dataset, schema=schema, backfill_date=backfill_date)
        
    utility.log_memory_usage() 

    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')
