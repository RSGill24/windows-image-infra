FORMATTER = logging.Formatter('%(asctime)s - %(name)s - %(levelname)s - %(message)s')

def set_file_handler(logger, logfile_name, level):
    # Updated to use pathlib for directory and file construction
    log_dir = Path(tempfile.gettempdir()) / "logs"
    log_dir.mkdir(parents=True, exist_ok=True)
    
    log_file = log_dir / logfile_name
    file_handler = RotatingFileHandler(log_file, maxBytes=(10*(2**20)), backupCount=10)
    file_handler.setFormatter(FORMATTER)
    file_handler.setLevel(level)
    logger.addHandler(file_handler)
    return logger

def set_stream_handler(logger, level):
    stream_handler = logging.StreamHandler()
    stream_handler.setFormatter(FORMATTER)
    stream_handler.setLevel(level)
    logger.addHandler(stream_handler)
    return logger

def flatten_exports(export_data: pl.LazyFrame):
    structs = [name for name, schema in export_data.collect_schema().items() if isinstance(schema, pl.Struct)]
    export_data = export_data.with_columns(*[pl.col(name).name.prefix_fields(f"{name}_") for name in structs]).unnest(structs)
    return export_data

def null_empty_child_struct(df: Union[pl.DataFrame, pl.LazyFrame]):
    def rebuild_path(path: list, dtype: pl.DataType) -> list[pl.Expr] | None:
        field_name = path[-1]
        if not isinstance(dtype, pl.Struct):
            return []
        
        if not len(dtype.fields):
            head, *rest = path
            *children, tail = rest
            _expr = pl.col(head)
            for child in children:
                _expr = _expr.struct.field(child)
            _expr = _expr.struct.with_fields(pl.lit(None).alias(tail))
            return [_expr]
        
        child_exprs = []
        for f in dtype.fields:
            child_path = path + [f.name]
            built = rebuild_path(child_path, f.dtype)
            if not built:
                continue
            child_exprs.extend(built)

        return child_exprs
    
    out_exprs = []
    for col, dt in df.collect_schema().items():
        built = rebuild_path([col], dt)
        if not built:
            continue
        expr = built[0].alias(col)
        out_exprs.append(expr)

    return df.with_columns(out_exprs)

# --- Updated Export Logic ---

def _write_chunk_ndjson(data: list[dict], export_uuid: str, export_type: str, export_chunk_id: int, version: int, base_dir: Path):
    """
    Callback function for run_threaded(). Writes an individual chunk to its own NDJSON file.
    """
    fn = base_dir / f"{export_type}-{export_uuid}-{export_chunk_id}-{version}.ndjson"
    with fn.open('w') as f:
        for item in data:
            f.write(json.dumps(item) + "\n")

_original_get_status = ExportsIterator._get_status

def _patched_get_status(self):
    # Call the original method to do its normal checks and assignments
    status = _original_get_status(self)
    
    # Intercept a CANCELLED status and force an exception to break infinite loops
    if status.status == 'CANCELLED':
        self._log.error(f"Export {self.type} {self.uuid} was CANCELLED. Forcing exit.")
        raise TioExportsError(
                export=self.type, 
                uuid=self.uuid, 
                msg=f"Export job {self.uuid} was CANCELLED by the server."
            )
        
    return status

# Apply the patch to the class
ExportsIterator._get_status = _patched_get_status

def _run_threaded_export_with_retries(tio_method, *filters: dict, retries: int = 5, num_threads: int = 4) -> pl.LazyFrame:
    """
    Generic method to execute PyTenable exports using threads and pathlib.
    Implements retry logic for transient API and Restfly exceptions.
    """
    attempts_left = retries + 1
    if not filters:
        filters = [{}]
        
    temp_dir = Path(tempfile.gettempdir())
    base_export_dir = temp_dir / f"_export_{tio_method.__name__}_{int(time.time())}"
    
    while attempts_left > 0:
        try:
            start = datetime.now()
            
            if base_export_dir.exists():
                shutil.rmtree(base_export_dir)
            
            base_export_dir.mkdir(parents=True, exist_ok=True)
            
            for filts in filters:
                iterator = tio_method(**filts)
                
                iterator.run_threaded(
                    func=_write_chunk_ndjson,
                    kwargs={'base_dir': base_export_dir},
                    num_threads=num_threads
                )
            
            logger.debug("Took %s to complete threaded export for %s", datetime.now() - start, tio_method.__name__)
            
            glob_path = base_export_dir / "*.ndjson"
            return null_empty_child_struct(pl.scan_ndjson(glob_path, infer_schema_length=None))
            
        except (RestflyException, ServerError) as e:
            attempts_left -= 1
            logger.exception("Exception hit during %s export. Retries left: %s", tio_method.__name__, attempts_left)
            
            if base_export_dir.exists():
                shutil.rmtree(base_export_dir, ignore_errors=True)
            
            if attempts_left == 0:
                raise e
            
            time.sleep(5)

def get_assets(tio: TenableIO, filter_null_tags=True) -> pl.LazyFrame:
    assets_df = _run_threaded_export_with_retries(tio.exports.assets_v2)
    if filter_null_tags:
        assets_df = assets_df.filter(~pl.col('tags').is_null()) 
    return assets_df

def get_vulns_exports(tio: TenableIO, *filter:dict, no_nnm=True, flatten=True, retries=5):
    if no_nnm:
        if filter:
            filter = [{**NNM_FILTER, **_filts} for _filts in filter]
        else:
            filter = [NNM_FILTER]
            
    results = _run_threaded_export_with_retries(tio.exports.vulns, *filter, retries=retries)
    if flatten:
        results = flatten_exports(results)
    return results

def get_compliance_exports(tio: TenableIO, *filter:dict, flatten=True, retries=5):
    results = _run_threaded_export_with_retries(tio.exports.compliance, *filter, retries=retries)
    if flatten:
        results = flatten_exports(results)
    return results

def xref_to_sc_format(xref_key) -> pl.Expr:
    return (
        pl.col(xref_key).list.eval(
            pl.element().struct.field("type") + " #" + pl.element().struct.field("id")
        ).list.join(',').name.keep()
    )
