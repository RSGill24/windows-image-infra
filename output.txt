def concat_dfs(*args, schema=None, how='vertical', rechunk=False, parallel=True):
    to_concat = []
    for a in args:
        if a.is_empty():
            to_concat.append(pl.DataFrame(schema=schema))
        else:
            to_concat.append(a.cast(schema, strict=False))
    return pl.concat(to_concat, how=how, rechunk=rechunk, parallel=parallel)

    res = workflow_api.v1_systems_system_id_poams_poam_id_workflow_get(system_id, poam_id, _request_timeout=_request_timeout)

@csam_retry
def poam_is_closed(api, system_id:int, poam_id:int, _request_timeout=300):
    workflow_api = csam_api_client.PoamWorkflowApi(api)
    res = workflow_api.v1_systems_system_id_poams_poam_id_workflow_nextsteps_get(system_id, poam_id, _request_timeout=_request_timeout)
    if len(res) != 1:
        return False
    next_step = res[0]
    if next_step.next_steps == 'Reopen POA&M':
        return True
    return False

class RLThreadPoolExecutor(cf.ThreadPoolExecutor):
    def __init__(self, *, max_workers=None, rate_limit: RateLimit, **kwargs):
        super().__init__(max_workers=max_workers, **kwargs)
        if rate_limit is None:
            raise ValueError("You must pass a RateLimit instance")
        self._rate_limit = rate_limit

    def submit(self, fn, *args, **kwargs) -> cf.Future:
        self._rate_limit.wait()
        # logger.debug("Submitting %s: args: %s, kwargs %s", fn.__name__, str(args), str(kwargs))
        return super().submit(fn, *args, **kwargs)

def get_system_level_api_call(api_func, systems: List[int], exclude=None, **kwargs):
    if not exclude:
        exclude=set()
    else:
        exclude=set(exclude)
    futures = {}
    with RLThreadPoolExecutor(rate_limit=CSAM_RATE_LIMIT) as executor:
        for system in systems:
            if system in exclude:
                continue
            logger.debug("Submitting %s: args: %s, kwargs %s", api_func.__name__, str([system]), str(kwargs))
            futures[executor.submit(csam_retry(api_func), system, **kwargs)] = system
    return {sys: f.result() for f, sys in futures.items()}

def get_poam_level_api_call(api_func, poams: List[tuple], **kwargs):
    futures = {}
    with RLThreadPoolExecutor(rate_limit=CSAM_RATE_LIMIT) as executor:
        for poam in poams:
            logger.debug("Submitting %s: args: %s, kwargs %s", api_func.__name__, str(poam), str({}))
            futures[executor.submit(csam_retry(api_func), *poam, **kwargs)] = poam 
    return {poam: f.result() for f, poam in futures.items()}

def get_poam_milestone_level_api_call(api_func, milestones, **kwargs):
    futures = []
    with RLThreadPoolExecutor(rate_limit=CSAM_RATE_LIMIT) as executor:
        for sys_id, poam_id, mile in milestones:
            logger.debug("Submitting %s: args: %s, kwargs %s", api_func.__name__, str([sys_id, poam_id, mile.action_number]), str({}))
            futures.append((executor.submit(csam_retry(api_func), sys_id, poam_id, mile.action_number, **kwargs), poam_id, mile))
    
    to_return = defaultdict(lambda: defaultdict(dict))
    for future, poam_id, mile in futures:
        to_return[poam_id][mile.action_number] = future.result()
    return to_return

def get_control_level_api_call(api_func, sys_id, controls: List[csam_api_client.ControlInheritance], **kwargs):
    futures = {}
    with RLThreadPoolExecutor(rate_limit=CSAM_RATE_LIMIT) as executor:
        for control in controls:

def submit_to_csam(csam_func, api, drafts, _request_timeout=300):
    """Helper function used to submited POST/PUT queries to CSAM. 
    Uses a loop to ensure errors are retried. Errors that can't be handled are returned for further processing
    """
    # ensure any func has the retry
    if not getattr(csam_func, retry_flag, None):
        csam_func = csam_retry(csam_func)

    failed = []
    created = []
    while drafts:
        futures = []
        with RLThreadPoolExecutor(max_workers=10, rate_limit=CSAM_RATE_LIMIT) as executor:
            for draft in drafts:
                # logger.debug("Submitting %s: args: %s, kwargs %s", csam_func.__name__, str([api, draft]), str({}))
                futures.append((executor.submit(csam_func, api, draft, _request_timeout=_request_timeout), draft))
                if any(csam_func is func for func in [post_poam_association, post_draft_poam_milestone]):
                    time.sleep(.25) # hopefully the timer will offer enough of a delay for things like milestones that apply an
        drafts = []
        for idx, x in enumerate(futures):
            future, draft = x
            if future.exception():
                # retry issues with the connection
                if isinstance(future.exception(), ProtocolError): # took out csam_api_client.exceptions.ServiceException since a 500 may point to a larger issue
                    drafts.append(draft)
                    continue

                # if the exception is NOT related to a 500 or protocol, capture for logging and debugging
                failed.append((idx, draft, future.exception()))
                continue
            created.append((idx, future.result(), draft))
    return created, failed

def log_failed(logger, wtps, fs: list) -> None:
    if not fs:
        return
    for f in fs:
        idx, draft, exc = f
        wtp = wtps[idx]
        logger.error("POAM Issue with %s: Got '%s'\n\nwhen submitting '%s'", wtp.number, str(exc), draft)

@csam_retry
def get_csam_report(csam_api, report_name, csam_params, _request_timeout=300):
    # poam_report_name = "POAM_Details_Report"

    report_api = csam_api_client.ReportsApi(csam_api)

    def thread_report(csam_params):

        # this is the payload needed to query a report from teh API
        # this query is what the RMF team uses. Mimicking what they do as best as we can
        logger.debug("Pulling report for %s", csam_params)
        return report_api.v1_reports_description_post(report_name, csam_params, _request_timeout=_request_timeout)
    
    # the report meta data gives all available entries. entries take the format `f"{system.name} - {system.acronym}"` 
    # Figured this would be more accurate than manually creating those
    reports_meta = report_api.v1_reports_description_metadata_get(report_name, _request_timeout=_request_timeout)
    param_desc = csam_params['parameters'][0]['values'][0]
    valid_param_desc = [x for x in reports_meta[0].filters if x.parameter_description == param_desc][0].valid_values
    csam_poam_data = []
    while valid_param_desc:
        with RLThreadPoolExecutor(rate_limit=CSAM_RATE_LIMIT) as executor:
            threads = {executor.submit(thread_report, pd): pd for pd in valid_param_desc}
        valid_param_desc = []
        for thread, pd in threads.items():
            if (exc:=thread.exception()):
                logger.debug("%s, retrying '%s'", exc, pd)
                valid_param_desc.append(pd)
                continue
            csam_poam_data.extend(thread.result())

    return pl.DataFrame(csam_poam_data)

        else:
            suffix = f"_{count}_of_{split}"
        _fn = filename.split(".")
        _fn[0] = _fn[0]+suffix
        _fn = ".".join(_fn)
        fn = create_local_attachment_file(_fn, _df, sort_key)
        try:
            res = csam_retry(system_poam_api.v1_systems_system_id_poams_poam_id_artifacts_post_with_http_info)(sys_id, poam_id, fn, _request_timeout=_request_timeout)
            logger.debug("POST'ed Attachment: %s %s", poam_id, _fn)
            created[fn] = res.data
            count += 1
        except RequestTooLargeThatsWhatSheSaidException as e:
            for att_name, linkeddata in created.items():
                csam_retry(system_poam_api.v1_systems_system_id_poams_poam_id_artifacts_artifact_id_delete)(sys_id, poam_id, linkeddata.artifact_id, _request_timeout=_request_timeout)
                os.remove(att_name)
            logger.warning("%s, %s : %s too large, splitting into smaller attachments", sys_id, poam_id, fn)
            return post_attachment(system_poam_api, sys_id, poam_id, filename, df, sort_key, split=split+1, _request_timeout=_request_timeout)
        else:
            os.remove(fn)
            break

    return

@lru_cache()
def get_assignee(api: ApiClient, *, name_wildcard, email, _request_timeout=300):
    """Talking with Jessica all of the user accounts in CSAM are populated with the GAL. Seems reasonable to use the latest ID to get the latest"""
    poc_api = csam_api_client.PointsOfContactApi(api)
    pocs = poc_api.v1_pocs_get(name_wildcard=name_wildcard, email_wild_card=email, _request_timeout=_request_timeout)
    if not pocs:
        return
    pocs = filter(lambda p: p.is_active, pocs)
    return max(pocs, key=lambda x: x.person_id)
