def requests_error_handler(func):
    """decorator to handle errors and retries callables"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        counter = 0
        while True:
            try:
                res = func(*args, **kwargs)
                res.raise_for_status()
            except requests.exceptions.HTTPError as e:
                return res
            except RequestException as e:
                counter += 1
                logger.error("%s exception thrown when executing %s. Attempt %s, retrying...", e.__class__.__name__, func.__name__, counter)
                time.sleep(5)
            else:
                return res
    return wrapper

@requests_error_handler
def get_snow_table(table_name, api_url, limit=0, offset=0, **kwargs) -> dict:
    """return SNOW FISMA Inventory Table"""
    url = f"{api_url}/table/{table_name}"

    if limit:
        params['sysparm_limit'] = limit
    if offset:
        params['sysparm_offset'] = offset
    res = SNOW_REQUESTS_SESSION.get(url, params=params, timeout=600)
    return res

@execute_timer(logger=logger)
def insert_multiple(df, table_name, api_url=SNOW_API, rows_per_chunk=6000):
    cooloff = 30
    if rows_per_chunk <= 0:
        raise Exception("Rows per chunk reduced too low")

    for start_idx in range(0, len(df), rows_per_chunk):
        _df =  df.slice(start_idx, rows_per_chunk)
        to_send = {
            "records": []
        }
        for idx, record in enumerate(_df.to_dicts()):
            if idx % 1000 == 0:
                logger.debug("Processed %s to send to insertMultiple", idx)
            to_send['records'].append({k:unescape(v) for k,v in record.items() if v})
        while True:
            start = datetime.datetime.now(datetime.timezone.utc)

            try:
                res = SNOW_REQUESTS_SESSION.post(api_url+f"/import/{table_name}/insertMultiple", json=to_send, timeout=300)
            except RequestException as e:
                cooloff = 30
                logger.exception("Running %s cooloff, then retrying %s-%s", cooloff, start_idx, start_idx+rows_per_chunk)
                time.sleep(cooloff)
                continue

            if not res.ok:
                logger.debug("%s %s %s", res, res.reason, res.text)
                if str(res.status_code)[0] == '5' and "Transaction cancelled: maximum execution time exceeded" in res.text:
                    new_rows_per_chunk = rows_per_chunk-1000
                    logger.debug("Ran into issues with rows_per_chunk=%s, reducing to %s and trying again", rows_per_chunk, new_rows_per_chunk)
                    # slicing the original df. (start_idx, None) slices the full df from start_idx
                    # this way we aren't resending what's already been sent 
                    _df = df.slice(start_idx, None)
                    # recursively calling the function adjusting the rows_per_chunk
                    return insert_multiple(_df, table_name, api_url=SNOW_API, rows_per_chunk=new_rows_per_chunk)
                else:
                    logger.exception("Running %s cooloff, then retrying %s-%s", cooloff, start_idx, start_idx+rows_per_chunk)
                    time.sleep(cooloff)
                    continue

            end = datetime.datetime.now(datetime.timezone.utc)
            logger.debug("Sent %s to %s. Elapsed time %s", start_idx+rows_per_chunk, api_url+f"/import/{table_name}/insertMultiple", end-start)
            break

@execute_timer(logger=logger)
def insert_multiple_threaded(df, table_name, api_url=SNOW_API, rows_per_chunk=500):

    def send(to_send, start_idx, rows_per_chunk):
        url = api_url+f"/import/{table_name}/insertMultiple"
        logger.debug('Sending %s-%s to %s', start_idx, start_idx+rows_per_chunk, url)
        start = datetime.datetime.now(datetime.timezone.utc)
        retry_count = 10
        while True:
            if not retry_count:
                raise Exception("Max retries hit")
            
            try:
                res = SNOW_REQUESTS_SESSION.post(url, json=to_send, timeout=300)
            except RequestException as e:
                cooloff = 30
                logger.exception("Running %s cooloff, then retrying %s-%s", cooloff, start_idx, start_idx+rows_per_chunk)
                time.sleep(cooloff)
                continue

            if not res.ok:
                retry_count -= 1
                logger.debug("%s %s %s", res, res.reason, res.text)
                logger.debug('Retry Sending %s-%s to %s. Retries left: %s', start_idx, start_idx+rows_per_chunk, url, retry_count)
                time.sleep(300)
                continue
            break
        end = datetime.datetime.now(datetime.timezone.utc)
        logger.debug("Sent %s to %s. Elapsed time %s", start_idx+rows_per_chunk, api_url+f"/import/{table_name}/insertMultiple", end-start)
        return

    futures = []
    with cf.ThreadPoolExecutor(max_workers=MAX_THREADS) as executer:
        for start_idx in range(0, len(df), rows_per_chunk):
            _df =  df.slice(start_idx, rows_per_chunk)
            to_send = {
                "records": []
            }
            for idx, record in enumerate(_df.to_dicts()):
                # if idx % 1000 == 0:
                    # logger.debug("Processed %s to send to insertMultiple", idx)
                to_send['records'].append({k:unescape(v) for k,v in record.items() if v})
            futures.append(executer.submit(send, to_send, start_idx, rows_per_chunk))
    exceptions = [f.exception() for f in futures if f.exception()]
    if exceptions:
        for e in exceptions:
            try:
                raise e
            except:
                logger.exception("Thread failed:")
        raise ExceptionGroup(f"One or more threads failed insert in {api_url}", exceptions)
 
@execute_timer(logger=logger)
def send_to_snow(df, table_name, batch_limit_size=(9.8*(1024**2)), row_limit=200, max_threads=MAX_THREADS):
    """
    Function to send results from a dataframe to ServiceNow

    params:
    df: the dataframe to send
    table_name: name of the ServiceNow table we're sending to
    batch_limit_size: the ServiceNOW Batch API has limits (5MB) on the size batch queries. Default: 4MB to ensur we're under that threshold to avoid errors: https://docs.servicenow.com/bundle/vancouver-api-reference/page/integrate/inbound-rest/concept/batch-api.html
    row_limit: default=5000. Similar to batch_limit_size, more than 5000 seems to illicit unexpected behavior
    """
    s = datetime.datetime.now()
    total_to_send = len(df)

    if df.is_empty():
        return
    
    df = (
        df.drop(pl.col(SNOW_LINK_DTYPE))
        .cast(pl.String)
        .with_columns(pl.all().str.slice(0,4000)) # max char length for SNOW is 4000
    )

    def is_to_retry(_fail):
        """helper function to determine if a response from the API warants retrying"""

        # obviously any server errors
        if str(_fail['status_code'])[0]=="5":
            return True
        
        body = json.loads(base64.b64decode(_fail['body']))
        try:
            # sometimes are 200 is returned, but there is an `error` 
            detail = body['error']['detail']
            if detail and "Transaction cancelled: maximum execution time exceeded" in detail:
                return True
        except KeyError:
            # typically if it's not a server error it's a 403, which indicates an issue with the body.
            # this can include a finding that's already submitted which SNOW is filtering out
            logger.debug("WONT RETRY: %s", body)
            return False
        logger.debug("WONT RETRY: %s", body)
        return False

    def chunk_records(df, size=batch_limit_size, row_limit=row_limit):
        """helper function to chunk the larger dataframe into manageable chunks that SNOW can handle"""

        # need to estimate the memory each row in the dataframe is occupying.
        # determine what's the largest occupier, and use it 
        df = df.with_columns(est_size=pl.Series([pl.Series(x).estimated_size() for x in df.iter_rows()], dtype=pl.UInt16))
        largest_row_mem = df['est_size'].max()
        df = df.drop('est_size')

        # calculate rows and how many rows per chunk
        rows = int(size // largest_row_mem)
        rows_per_chunk = rows if rows < row_limit else row_limit
        # rows_per_chunk = row_limit
        for start in range(0, len(df), rows_per_chunk):
            yield df.slice(start, rows_per_chunk)

    @requests_error_handler
    def send_batch(batch_id, chunk_df):
        """Helper function that sends chunked dataframes to ServiceNOW"""

        # format for batch call can be found here: https://docs.servicenow.com/bundle/vancouver-api-reference/page/integrate/inbound-rest/concept/batch-api.html
        logger.debug("Batch ID %s sending %s requests to %s", batch_id, len(chunk_df), table_name)
        data = {
            "batch_request_id": str(batch_id),
            "rest_requests": []
        }
        for idx, record in enumerate(chunk_df.to_dicts()):
            data['rest_requests'].append(
                {
                    "id": str(idx+1),
                    "headers": [
                        {
                            "name": "Content-Type",
                            "value": "application/json"
                        }
                    ],
                    "url": f"/api/now/table/{table_name}" + (f"/{record['sys_id']}" if (method:=record.pop('HTTP Action').lower()) in {"put", "patch", "delete"} else ""),
                    "method": method.upper(),
                    # "body": base64.b64encode(unescape(json.dumps(record)).encode()).decode()
                    "body": base64.b64encode((json.dumps({k:unescape(v) for k,v in record.items() if v})).encode()).decode()
                }
            )
        res = SNOW_REQUESTS_SESSION.post(SNOW_API+"/v1/batch", json=data, timeout=300)

        if not res.ok:
            logger.debug("%s %s %s", res, res.reason, res.text)

        # if status is 202 need to resend the same batch to get the status of the load.
        # otherwise the response get's lost and we dont know if it ingested everything.
        while res.status_code == 202:
            logger.debug("202 Status Code, retrying batch ID %s", batch_id)
            time.sleep(60)
            req = res.request
            res = SNOW_REQUESTS_SESSION.request(method=req.method, url=req.url, headers=req.headers, data=req.body, timeout=300)

        return res
    
    # Start with ingest.
    batch_id = 0
    while True:
        # using a while loop since there will be some queries that need to be retried.
        batch_size_has_changed = False
        batch_row_limit_has_changed = False
    
        with cf.ThreadPoolExecutor(max_workers=max_threads) as executor:
            threads = {}
            for chunk in chunk_records(df, size=batch_limit_size, row_limit=row_limit):
                batch_id += 1 
                threads[executor.submit(send_batch, batch_id, chunk)] = chunk
                                        
            to_retry = pl.DataFrame()
            for future in cf.as_completed(threads):
                _df = threads[future]
                # Loop thru each future object and determine if it was good, or needs to be retried
                try:
                    res = future.result()
                except KeyboardInterrupt as e:
                    raise e
                except Exception as e:
                    # if anything exception happens in the thread, retry
                    logger.error(str(e))
                    to_retry = concat_dfs(to_retry, _df, schema=df.schema)
                    continue

                if not res.ok:
                    if res.status_code == 403:
                        # This is different from a single query returning a 403, if the BatchAPI returns a 403 it is indeed a credential issue
                        # logger.error("Invalid Credentials: %s", res.text)
                        raise BadCreds(f"Invalid Credentials: {res.text}")
                    if res.status_code == 413:
                        logger.debug("REQUEST TOO LARGE, reducing batch size by 10%")
                        if not batch_size_has_changed:
                            batch_size_has_changed = True
                            batch_limit_size *= .9
                        
                    to_retry = concat_dfs(to_retry, _df, schema=df.schema)
                    continue
                
                try:
                    results = res.json()
                except requests.exceptions.JSONDecodeError as e:
                    # a JSONDecodeError means somehting went wrong with the query
                    if "Transaction cancelled: Max output size reached for current BatchRest transaction" not in res.text:
                        logger.debug(res.text)
                        to_retry = concat_dfs(to_retry, _df, schema=df.schema)
                        continue
                    if batch_row_limit_has_changed:
                        to_retry = concat_dfs(to_retry, _df, schema=df.schema)
                        continue
                    logger.debug("Chunked dataframe with %s rows triggered a max output size error from the API. Reducing row limit by 10%%", len(_df))
                    to_retry = concat_dfs(to_retry, _df, schema=df.schema)
                    batch_row_limit_has_changed = True
                    row_limit = int(row_limit * .9)
                    continue
                    
                unserviced = set(results['unserviced_requests'])
                if unserviced:
                    logger.debug("%s UNSERVICED REQUESTS, retrying", len(unserviced))
                    idxs = [int(_id)-1 for _id in unserviced]
                    to_retry = concat_dfs(to_retry, _df.select(pl.all().gather(idxs)), schema=df.schema)

                serviced = results['serviced_requests']
                serviced_idxs = [int(x['id'])-1 for x in serviced]
                requests_mapping = list(zip(_df.select(pl.all().gather(serviced_idxs)).to_dicts(), serviced))
                failed = [_res for _res in requests_mapping if str(_res[1]['status_code'])[0] != '2']
                if failed:
                    for _fail in failed:
                        body = base64.b64decode(_fail[1]['body']).decode(encoding='utf-8', errors='ignore')
                        try:
                            err_msg = str(json.loads(body)['error'])
                        except json.JSONDecodeError:
                            err_msg = body
                        logger.debug("FAILED: STATUS '%s' ERROR MSG '%s'", _fail[1]['status_code'], err_msg)#, json.dumps(_fail[0]))
                    logger.debug("FAILED TO SEND TO %s REQUESTS TO SNOW", len(failed))
                    retry = [x[0] for x in failed if is_to_retry(x[1])]
                    to_retry = concat_dfs(to_retry, pl.from_dicts(retry, schema=df.schema), schema=df.schema)

            if not to_retry.is_empty():
                df = to_retry
                logger.debug("Retrying %s findings", len(df))
            else:
                logger.info("Avg time taken: %s findings/second. Total findings: %s", get_avg_time(s, datetime.datetime.now(), total_to_send), total_to_send)
                return

    SNOW_MILESTONE_TABLE_NAME: Schema([('actual_start_date', String), ('due_date', String), ('sys_mod_count', String), ('description', String), ('sys_updated_on', String), ('planned_finish_date', String), ('sys_tags', String), ('weakness_tracking_id', Struct({'link': String, 'value': String})), ('actual_finish_date', String), ('number', String), ('sys_id', String), ('sys_updated_by', String), ('sys_created_on', String), ('planned_start_date', String), ('milestone_assignee', Struct({'link': String, 'value': String})), ('action_number', String), ('sys_created_by', String)])
}
            
@execute_timer(logger=logger)
def get_latest_snow_table(table, api_url, offset_limit=5_000, limit=0, sysparm_fields: list = None, sysparm_query=""):
    if sysparm_fields:
        sysparm_fields = ",".join(sysparm_fields)

    timer = datetime.datetime.now()
    timer_step = 100_000

    start = 0
    data = pl.DataFrame()
    done = False
    to_retry = []
    while not done:
        if len(data):
            checkpoint = datetime.datetime.now() - timer
            avg = checkpoint / (len(data) / timer_step)
            logger.debug("Elapsed time: %s, records pulled: %s, avg time per %s: %s", checkpoint, len(data), timer_step, avg)
        if to_retry:
            logger.debug("Encountered retries, sleeping to avoid execution timeout...")
            time.sleep(60)
        with cf.ThreadPoolExecutor(max_workers=10) as executor:
            attempts = executor._max_workers
            end = start + offset_limit*attempts if not limit else limit

            """
            sys_id: UUID that SNOW populates for each finding
            scan_id: a value derived from Tenable's "vulnUniqueness" field. Typically `repositoryID,ip,port,protocol,pluginID`
            id: an incrementing ID for tracking in SNOW. e.g. FIND000001
            state: state of the finding
            days_since_discover: the diff of days between now, and First Discovered
            first_discovered: Tenable value showing when the finding was found
            last_observed: Tenable value showing when the finding was last seen
            poa_m_required: Value that I calculate
            """
            runs = []
            for step in range(start, end, offset_limit):
                params = {'offset': step, 'limit': offset_limit}
                if sysparm_fields:
                    params['sysparm_fields'] = sysparm_fields
                if sysparm_query:
                    params['sysparm_query'] = sysparm_query
                runs.append(params)

            while to_retry:
                retry = to_retry.pop(0)
                logger.debug("retrying %s", str(retry))
                runs.append(retry)

            futures = []
            for run in runs:
                logger.debug("calling %s for %s", str(run), table)
                futures.append((executor.submit(get_snow_table, table, api_url, **run),run))

        to_concat = []
        for future, run in futures:
            res = future.result()
            if not res.ok:
                sc = str(res.status_code)
                if sc[0] == '5' or sc == "429":
                    logger.debug("%s status code received, retrying", res.status_code)
                    to_retry.append(run)
                    continue

                if sc[0] == "4":
                    logger.error("%s", res.json())
                    to_retry.append(run)
                    continue
                    # raise Http400Error(res.json())

            try:
                _data = res.json()
            except (JSONDecodeError, ContentDecodingError) as e:
                logger.debug("%s exception hit, retrying...", str(e))
                to_retry.append(run)
                continue

            if "error" in _data:
                logger.debug("Error returned from API, retrying - %s", _data['error'])
                to_retry.append(run)
                continue

            if not _data:
                done = True
                break

            try:
                to_concat.extend([x for x in _data['result'] if x])
            except KeyError:
                logger.debug("%s exception hit, retrying...", str(e))
                to_retry.append(run)
                continue

        else:
            if not to_retry and not to_concat:
                done = True

        # SNOW has a "datatype" that refers to a link in another SNOW table. This block
        # seeks to set the schema for those columns to match that data type. evertying else
        #  is a string. otherwise polars will puke
        # ex: {'link': 'https://iceserviceportal.servicenowservices.com/api/now/table/u_fisma_inventory/6b79fa14dbb4b60046d470340f961943', 'value': '6b79fa14dbb4b60046d470340f961943'}
        if not to_concat:
            continue
        if not (schema:=SNOW_TABLE_SCHEMAS.get(table, None)):
            schema = {k: pl.String for k in to_concat[0].keys()}
            keys = {x for t in to_concat for x in t.keys()}
            for test in to_concat:
                for k in keys:
                    if k not in test:
                        continue
                    v = test[k]
                    if isinstance(v, pl.Struct):
                        continue
                    if isinstance(v, dict) and all(t in v for t in ['link', 'value']):
                        schema[k] = SNOW_LINK_DTYPE
        if to_concat:
            _schema = {**{k:String for k in to_concat[0]}, **schema}
            data = concat_dfs(data, pl.from_dicts(to_concat, schema=_schema, infer_schema_length=None), schema=_schema)
        start = end
    fn = os.path.join(tempfile.gettempdir(), f"get_latest_snow_table_{table}_{time.time()}.parquet")
    data.write_parquet(fn, use_pyarrow=True)
    return pl.scan_parquet(fn)
