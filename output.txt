MAIN_SCHEMA = rw_util.MAIN_SCHEMA
BACKFILL_SCHEMA = rw_util.BACKFILL_SCHEMA

METADATA_TABLE = "crma_hive_load_control_metadata"
  
def create_audit_controls_table(
        engine: Engine, 
        input: dict,
        schema: str,
        backfill_date: Optional[str] = None
    ) -> pd.DataFrame:
    """ 
    Function to load select fields from their repective source tables, and perfoms joins to merge data

    Parameters:
        engine (Engine): SQLAlchmey engine
        input (dict): configuration dictionary for source tables and fields
        schema (str): The database schema in which data is loaded.
        backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525)
    Returns:
        Pandas dataframe 
    """     
    def  _read_input_data(engine, fields, table) ->pd.DataFrame:
        """Helper function to retrieve data from from the given table"""
        query = f"SELECT {fields} FROM {schema}.{table}"
        return rw_util.fetch_postgres_table(engine, query)
  
    def _parse_isso(isso):
        """Helper function to extract isso_name and isso_email 
           from `information_system_security_officer` field
        """
        isso = isso.strip("[]")
        name = isso.split(';')[0].strip()
        if "," in name:
            last, first_middle = name.split(",", 1)
            isso_name = f"{first_middle.strip()} {last.strip()}"
        else:
            isso_name = name
        email_match = re.search(r'Email:\s*([\w\.-]+@[\w\.-]+)', isso)
        isso_email = email_match.group(1) if email_match else None
        return pd.Series({'isso_name': isso_name, 'isso_email': isso_email})
        

    #Fetch SNOW poam_audit input fields
    table = input["snow_poam_audit"]["table"]
    fields = ", ".join(input["snow_poam_audit"]["fields"])
    snow_audit_df = _read_input_data(engine=engine, fields=fields, table=table)
    if snow_audit_df is None:
        raise ValueError("Failed to fetch input fields from SNOW poam_audit table.")

    #Fetch SNOW poam_data input fields
    table = input["snow_poam_data"]["table"]
    fields = ", ".join(input["snow_poam_data"]["fields"])
    snow_poam_df = _read_input_data(engine=engine, fields=fields, table=table)
    if snow_poam_df is None:
        raise ValueError("Failed to fetch input fields from SNOW poam_data table.")

    # Merge snow poam_audit with snow poam_data to retrieve poam_id   
    snow_audit_df = (
            snow_audit_df
            .merge(snow_poam_df, left_on='csam_poam_data_id', right_on='sys_id',how='left')
            [["poam_id", "audit_year", "audit_type", "domain_control_family", "audit_controls"]]
        )
                
    #Fetch CSAM poam_details input fields
    table = input["csam_poam_details"]["table"]
    fields = ", ".join(input["csam_poam_details"]["fields"])
    csam_poam_df = _read_input_data(engine=engine, fields=fields, table=table)
    if csam_poam_df is None:
        raise ValueError("Failed to fetch input fields from CSAM poam_details table.") 
    
    #Fetch CSAM systems_general_details input fields
    table = input["csam_systems_details"]["table"]
    fields = ", ".join(input["csam_systems_details"]["fields"])
    csam_system_df = _read_input_data(engine=engine, fields=fields, table=table)
    if csam_system_df is None:
        raise ValueError("Failed to fetch input fields from CSAM systems_genral_details table.") 
 
    # Merge 
    df = (
        snow_audit_df
        .merge(csam_poam_df, on='poam_id', how='inner')
        .merge(csam_system_df, on='system_id', how='left')
    )

    #Parse information_system_security_officer columns to extract isso_name and isso_email
    extracted = df["information_system_security_officer"].apply(_parse_isso)

    df = pd.concat([df.drop(columns=["information_system_security_officer"]), extracted], axis=1)

    if backfill_date:
         df["etl_date"] = datetime.strptime(backfill_date, "%m%d%y").strftime("%Y-%m-%d")
    else:
        df["etl_date"] = datetime.now().date()

    return df

def load_to_db(
        engine: Engine,
        df: pd.DataFrame,
        schema: str,
        table: str, 
        load_control_table: str,
        backfill_date: Optional[str] = None
) -> bool:
    """
    Loads data to desttination database
    
    Parameters:
        engine (create_engine): Database connection engine
        df (pd.DataFrame):
        schema (str):
        table (str):
        load_control_table (str):
    """
    logger.info(f"Loading table '{table}' to '{schema}' schema")

    # replicate schema if in backfill mode
    if schema == BACKFILL_SCHEMA:
        created_objects = rw_util.replicate_schema(engine, MAIN_SCHEMA, BACKFILL_SCHEMA)
        if created_objects:
            logger.info(f"Succesfully created the following objects in backfill schema:'{BACKFILL_SCHEMA}'")
            for obj in created_objects:
                obj_type, obj_name = obj.split(".",1)
                logger.info(f"{obj_type.upper():6} {obj_name}")
        else:
            logger.info(f"'{BACKFILL_SCHEMA}' schema is up to date. No tables/views created.")

    # load data to hist table 
    load_control_etl_date = datetime.now().date()

    db_load_flag = rw_util.load_table_to_db_daily_idempotent(etl_date=load_control_etl_date, engine=engine, df=df, schema_name=schema, table_name=table, load_control_table=load_control_table, exist_type="append" )

    if db_load_flag: #if loaded, perfoma post load validation
        logger.info(f"Perfomming post load validation on table '{table}' in '{schema}' schema")
        if rw_util.post_load_validation_hist_tables(engine=engine, df=df, schema_name=schema, table_name=table, backfill_date=backfill_date):
            logger.info(f"Post load validation check passed for table '{table}' in '{schema}' schema")

            ##Parallel loading##
            logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
            engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
            logger.info(f"Loading table '{table}' to '{schema}' schema")
            rw_util.load_table_to_db_daily_idempotent(etl_date=load_control_etl_date, engine=engine2, df=df, schema_name=schema, table_name=table, load_control_table=load_control_table, exist_type="append" )

        else:
            raise RuntimeError(f"Post load validation check failed for {table} in '{schema}' schema")
    
    return db_load_flag

def run_backfill_mode(engine: Engine, input: dict, bucket_name: str, hist_table_name: str, backfill_date: str ):
    """
    Backfill mode will query and load data from/to backfill schema and load a copy of reults to s3, overwring existing file if one exists

    Parameters:
        input (dict): configuration dictionary for source tables and fields
        engine (Engine): SQLAlchmey engine 
        bucket_name (str): Name of S3 bucket name to write output file
        backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525) 
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={HIST_TABLE_NAME}, date={backfill_date}")
    
    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if hist_table_name.lower() == HIST_TABLE_NAME.lower():    
        # retrieve input data
        df = create_audit_controls_table(engine=engine, input=input, schema=BACKFILL_SCHEMA, backfill_date=backfill_date)
        
        # load input data to hist table
        is_success = load_to_db(engine = engine, df=df, schema=BACKFILL_SCHEMA, table=HIST_TABLE_NAME, load_control_table= METADATA_TABLE, backfill_date=backfill_date)

        # overwrite/write file to s3
        if is_success:
            prefix = "processed/MISC/CLEAN/"
            base_filename = f"CRMA_HIVE_POAM_AUDIT_CONTROLS"
            target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, base_filename, backfill_date, "PARQUET")
            rw_util.load_to_s3(bucket_name, df, target_filename, "PARQUET", True)


def main():
    start_time = time.time()

                                "fields": ["system_id", "information_system_security_officer"],
                                "table": "crma_hive_csam_systems_general_details_query"

    # resolve workflow run properties
    workflow_params = utility.get_workflow_props()

    hist_table_name = workflow_params.get("--hist_table_name")

import pandas as pd
import urllib3
from datetime import datetime

logger = logging.getLogger(__name__)

# Instantiate classes
utility = Utility()
rw_util = Read_Write_Utility()

# Instantiate DQ utilities
dataQualityUtils = DataQualityUtilities()

prefix = "processed/SNOW/CLEAN/"
poam_data_file_name_prefix = "CRMA_HIVE_SNOW_POAM_DATA_CLEAN_"
poam_audit_file_name_prefix = "CRMA_HIVE_SNOW_POAM_AUDIT_CLEAN_"
sys_user_file_name_prefix = "CRMA_HIVE_SNOW_SYS_USER_CLEAN_"
fisma_details_file_name_prefix = "CRMA_HIVE_SNOW_FISMADetails_CLEAN_"

def process_date(bucket, file_configs, date_str):
    """
    Process files to create a poam_audit_historical record for a single date: find latest file -> join -> enrich -> return dataframe

    Parameters:
        bucket(str): s3 bucket name
        file_config (list): list of dicts with prefix and base_filename
        date_str (str): date of the archived files to ingest and process

    Returns:
        pd.DataFrame or None: Enriched dataframe combining poam_audit_historical fields found in poam, poam_audit, fisma_deails, and sys_user archived files
    
    """
    logger.info(f"Processing files for date: {date_str}")

    dfs = {}

    # get files from s3
    for i, config in enumerate(file_configs,1):
        prefix = config["prefix"]
        base_filename = config["base_filename"]
        df = rw_util.fetch_file_from_s3(bucket, prefix,base_filename+date_str,"PARQUET")
        if df is not None and not df.empty:
            # rename column to fix schema in earlier versions of archived poam_audit file
            if base_filename == prefix+poam_audit_file_name_prefix and "csam_poam_data" in df.columns:
                dfs[base_filename] = df.rename(columns={"csam_poam_data": "csam_poam_data_id"})
                logger.info(f"Renamed csam_poam_data to csam_poam_data_id for {base_filename}_{date_str}")
            else:
                dfs[base_filename] = df
            logger.info(f"Loaded {base_filename}: {len(df)} rows")
        else:
            logger.warning(f"Warning No file found for {base_filename}")
            



    try:
        # join poam and poam_audit
        logger.info(f"Before Join: {dfs[prefix+poam_data_file_name_prefix][dfs[prefix+poam_data_file_name_prefix]['poam_id'] == 215158]}")
        logger.info(f"Before Join: {dfs[prefix+poam_audit_file_name_prefix][dfs[prefix+poam_audit_file_name_prefix]['csam_poam_data_id'] == '49702b27873c02d4e767a9b60cbb3594']}")
        poam_and_audit_join = pd.merge (
            dfs[prefix+poam_data_file_name_prefix], 
            dfs[prefix+poam_audit_file_name_prefix],
            left_on = 'sys_id',
            right_on = 'csam_poam_data_id',
            how='inner',
            suffixes=('', 'poam_audit_table')
        )
        logger.info(f"Joined {poam_data_file_name_prefix} + {poam_audit_file_name_prefix}: {len(poam_and_audit_join)} rows")
        logger.info(f"After Join: {poam_and_audit_join[poam_and_audit_join['poam_id'] == 215158]}")

        # join fisma_details to poam_and_audit_join
        poam_and_audit_and_fisma_join = pd.merge(
            poam_and_audit_join, 
            dfs[prefix+fisma_details_file_name_prefix],
            left_on='fisma_name_id',
            right_on='sys_id',
            how='left',
            suffixes=('', 'fisma_table')
        )
        logger.info(f"Joined {fisma_details_file_name_prefix} : {len(poam_and_audit_and_fisma_join)} rows")
        logger.info(f"After Join: {poam_and_audit_and_fisma_join[poam_and_audit_and_fisma_join['poam_id'] == 215158]}")

        # if sys_user file is available join sys_user to poam_and_audit_and_fisma_join 
        final_join = poam_and_audit_and_fisma_join
        if prefix+sys_user_file_name_prefix in dfs:
            final_join = pd.merge(
                poam_and_audit_and_fisma_join, 
                dfs[prefix+sys_user_file_name_prefix],
                left_on='system_security_officer_isso_id',
                right_on='sys_id',
                how='left',
                suffixes=('', 'sys_user_table')
            )
            logger.info(f"Joined {sys_user_file_name_prefix} : {len(final_join)} rows")
            logger.info(f"After Join: {final_join[final_join['poam_id'] == 215158]}")
        else:
            logger.info(f"Skipping {prefix+sys_user_file_name_prefix} join (file not found)")
        


        # add metadata
        dt = datetime.strptime(date_str, "%m%d%y")
        etl_date = dt.strftime("%Y-%m-%d")
        final_join["etl_date"] = etl_date

        # if sys_user file not present add missing columns
        if "email" not in final_join.columns:
            final_join["email"] = ""
        if "name" not in final_join.columns:
            final_join["name"] = ""

        # reformat name
        def _reformat_name(df):
            df = df.copy()
            # capture last name, first_middle name, and strip out "(CTR)"
            parts = df["name"].str.split(',', n=1, expand=True)
            df["last"] = parts[0].str.strip()
            df["first_middle"] = parts[1].str.strip().str.replace("(CTR)","",regex=False) if 1 in parts.columns else pd.NA
            mask =  df["last"].notna() & df["first_middle"].notna() & (df["first_middle"] != "")
            # select only rows where mask is true and add formatted name
            df.loc[mask,"name_formatted"] = (df["first_middle"] + " " + df["last"])
            return df

        final_join= _reformat_name(final_join)

        # rename/drop columns
        renamed = dataQualityUtils.rename_fields(final_join, {"csam_id": "system_id", "email": "isso_email", "name_formatted":"isso_name"} ) # snow_poam_data.csam_id -> csam_poam_details_report.system_id
       
        target_columns = [  "poam_id", "poam_title", "status", "system_name", "system_id", "acronym", "portfolio", 'audit_year', 
                            'audit_type', 'domain_control_family', 'audit_controls', 
                            "actual_start_date","actual_finish_date", "detailed_weakness_description", "create_date",
                            "user_identified_criticality", "item_identified_during", "accepted_risk", "scheduled_completion_date",
                            "assigned_to", "isso_name", "isso_email", "etl_date" ]                                    

        logger.info("Selecting target columns...")

        result = dataQualityUtils.select_fields(renamed, target_columns)

        #Apply DQ4: Convert numeric columns to Integer 
        cols = ["audit_year"]
        result = dataQualityUtils.clean_int_fields(result, cols)

        logger.info(f"Successfully joined {len(result)} rows for {date_str}")

        return result
    except Exception as e:
        logger.error(f"Error processing {date_str}: {e}")
        return None


def main():
    """
    Example run:
    python crma_hive_snow_backfill_audit_controls_hist.py
    """
    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = "aws-glue-edae-crma-hive-landing-zone" #connection.get_secret("edae-crma-hive")["lz_bucket_name"]

    if not bucket_name:
       raise ValueError("S3 bucket not found!")

    logger.info("Creating Backfill file for poam_audit_historical table")

    # define variables
    output_file_name = "crma_hive_poam_audit_historical_"+ datetime.today().strftime("%Y%m%d")
    output_path = "test/backfill/" + output_file_name

    file_configs = [
        {"prefix": prefix, "base_filename": prefix+poam_data_file_name_prefix},
        {"prefix": prefix, "base_filename": prefix+poam_audit_file_name_prefix},
        {"prefix": prefix, "base_filename": prefix+fisma_details_file_name_prefix},        
        {"prefix": prefix, "base_filename": prefix+sys_user_file_name_prefix}, # file archive starts 9/24
    ]

    #TODO: Dynamically generate backfill dates starting from 091125
    backfill_dates = [
        "091125",
        "091525",
        "092225",
        "100625",
        "101325",
        "102025",
        "102725",
        "110325",
        "111025",
        "111725",
        "112425",
        "120125",
        "120825",
        "121525",
    ]

    logger.info(f"Processing {len(backfill_dates)} specific dates: {backfill_dates}")

    # process specific dates in order
    frames = []
    succesful_dates = 0
    for date_str in backfill_dates:
        df = process_date(bucket_name, file_configs, date_str)

        if df is not None and not df.empty:
            frames.append(df)
            succesful_dates +=1

    # combine records for all dates and write to s3
    if frames:
        final_df = pd.concat(frames, ignore_index=True)
        logger.info(f"Concatenated {len(frames)} dates -> {len(final_df)} total rows")

        rw_util.load_to_s3(bucket_name, final_df, output_path, "CSV")
        final_df.to_csv("./" + output_file_name + ".csv", index=False)

        logger.info(f"Success: wrote {output_path} ({succesful_dates} dates, {len(final_df)} rows)")
    else:
        logger.error("No dates processed succesfully")

    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')
    utility.log_memory_usage() 


if __name__ == "__main__":
    main()
