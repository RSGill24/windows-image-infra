def load_ucmm_score(
            bucket_name: str, 
            engine: Engine 
        ) -> None:
    """
    Load UCMM scores to db.
    Args:
        bucket_name: Name of the S3 bucket containing input.
        engine: SQLAlchemy Engine connected to the primary Postgres database.
    """
    logger.info(f"Fetching data from landing zone...")
    
    # Creating variables and the dataframe
    prefix = "raw_data/UCMM/"
    dataset = "UCMM_Scores"
    base_name = prefix + dataset

    df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "XLSX")   
    logger.info(f"Total records fetched: {len(df)}")

    target_columns = [
             "System ID", 	"Org", "Acronym",	"Actual Maturity", 	"Notional Maturity", 	"Maturity Level Delta", "Absolute Maturity Delta", 	"Data Quality Threshold Met"
        ]
    logger.info("Selecting target columns...")    
    df = dataQualityUtils.select_fields(df, target_columns)

    map = {
            "System ID": "system_id", "Org": "org", "Acronym": "acronym",	"Actual Maturity": "actual_maturity", 	"Notional Maturity": "notional_maturity", 	
            "Maturity Level Delta": "maturity_level_delta", "Absolute Maturity Delta": "absolute_maturity_delta", 	"Data Quality Threshold Met": "data_quality_threshold_met",
        }
    df = dataQualityUtils.rename_fields(df, map)

    #clean columns with null values in the following fields 
    cols = ["system_id", "org",]
    df = df.dropna(subset=cols)

    #Convert numeric columns to Integer 
    cols = ["system_id"]
    df = dataQualityUtils.clean_int_fields(df, cols)

    logger.info("Loading data to database...")
    df["etl_date"] = datetime.now().date() 
    schema="crma_hive"
    table_name = f"crma_hive_{dataset.lower()}"
    rw_util.truncate_and_load_table(
           engine=engine, 
           df=df, 
           schema_name=schema, 
           table_name=table_name, 
           exist_type="append"            
    )

    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    logger.info(f"Loading table '{table_name}' to '{schema}' schema")     
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
    rw_util.truncate_and_load_table(
           engine=engine2, 
           df=df, 
           schema_name=schema, 
           table_name=table_name, 
           exist_type="append"            
    )

def main():
    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 

    if not bucket_name:
       raise ValueError("S3 bucket not found!")
    
    logger.info("Load UCMM datasets to database.")

    load_ucmm_score(
            bucket_name=bucket_name,
            engine=engine
        )

    logger.info("Loading UCMM ICE Impact Rankings...")
    load_ucmm_ice_impact_rankings(
            bucket_name=bucket_name,
            engine=engine
        )

    logger.info("Loading UCMM POAM FISMA System Impact Rankings...")
    load_ucmm_poam_fisma_system_impact_rankings(
            bucket_name=bucket_name,
            engine=engine
        )
   
    # log runtime 
    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')

if __name__ == "__main__":
    main()
