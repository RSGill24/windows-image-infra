// Subject builder
def poamSubject(String status) {
    return "Weakness Tracking Automated POAM Cloudbees Job ${status}"
}

// Email body builder
def poamEmailBody(String status, String branchName) {
    return """
        <h2>Build ${status} Notification</h2>
        <p>Check Cloudbees console output for details.<br><br>
        Note: If error occurred within script, additional logs are available in AWS MAA-non prod account<br>
        under S3 Bucket "sara-weakness-tracking/weakness_tracking_poam" with file name: logs_&ltdate&gt_${branchName}.zip</p>
        <p><b>Job:</b> <a href=${env.BUILD_URL}>${env.JOB_NAME}/${env.BUILD_NUMBER}</a></p>
        <p><b>Build Number:</b> ${env.BUILD_NUMBER}</p>
        <p><b>Status:</b> <span style='color:red;'>${status}</span></p>
    """
}

def sendEmail(String status, String email, String branchName) {
    def local_mime_type  = 'text/html'
    def logFiles = findFiles(glob: "logs_*.zip")
    
    // 1. Log the recipient
    echo "Preparing to send email to: ${email}"
    
    // 2. Log the list of attachments found
    if (logFiles.length > 0) {
        echo "Found the following attachments:"
        logFiles.each { file ->
            echo " - ${file.name} (${file.length} bytes)"
        }
    } else {
        echo "No log files found, sending email without attachments."
    }

    // 3. Trigger the email
    emailext(
            subject: poamSubject(status),
            body: poamEmailBody(status, branchName),
            mimeType: local_mime_type,
            to: email,
            attachmentsPattern: logFiles.length > 0 ? "logs_*.zip" : ""
        )
}
