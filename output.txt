# use ansbile to copy the files/scripts into /home/zzadmin/scripts

- name: Install python3.12
  ansible.builtin.package:
    name: python3.12
    state: present
  become: true

- name: register python3.12 in alternatvies
  community.general.alternatives:
    name: python
    path: /usr/bin/python3.12
    link: /usr/bin/python
  become: true

- name: set python3.12
  ansible.builtin.command: alternatives --set python /usr/bin/python3.12
  become: true

- name: print python version
  ansible.builtin.command: python -V
  register: output

- name: print python version
  debug:
    var: output.stdout

- name: ensure pip
  ansible.builtin.command: python -m ensurepip --upgrade

- name: upgrade pip
  ansible.builtin.command: python -m pip install -U pip

- name: Install botocore and boto3 so we can use ansible s3 module
  ansible.builtin.command: python -m pip install botocore boto3

- name: create /opt/local
  file:
    path: /opt/local
    state: directory
    owner: zzadmin
    group: zzadmin
    mode: '0755'
  become: true

- name: Copy scripts
  ansible.builtin.copy:
    src: files/scripts/
    dest: /opt/local/scripts/
    mode: '0755'

- name: Create .venv in /opt/local
  ansible.builtin.command: python -m venv /opt/local/.venv

- name: Install requirements
  ansible.builtin.command: /opt/local/.venv/bin/python -m pip install -r /opt/local/scripts/requirements.txt
