gcloud builds submit --config=cloudbuild-docker.yaml . --project=ggn-nmfs-winde-dev-1


#!/bin/bash
cat > cloudbuild-docker.yaml << 'EOF'
steps:
  - name: 'gcr.io/cloud-builders/docker'
    args:
      - 'build'
      - '-t'
      - 'us-east4-docker.pkg.dev/$PROJECT_ID/image-builders/windows-packer-builder:latest'
      - '-f'
      - 'win2022-software-jira-install-dev/packer/scripts/Dockerfile'
      - '.'
images:
  - 'us-east4-docker.pkg.dev/$PROJECT_ID/image-builders/windows-packer-builder:latest'

serviceAccount: 'projects/ggn-nmfs-winde-dev-1/serviceAccounts/images-build-sa1@ggn-nmfs-winde-dev-1.iam.gserviceaccount.com'

options:
  logging: CLOUD_LOGGING_ONLY
EOF

echo "cloudbuild-docker.yaml created successfully"
cat cloudbuild-docker.yaml



cd win2022-oracle-client-dev/packer && sed -i '/variable "vm_network".*{.*type.*=.*string.*default/d' customize.pkr.hcl && sed -i '/variable "vm_subnet".*{.*type.*=.*string.*default/d' customize.pkr.hcl && cat >> customize.pkr.hcl << 'ENDFIX'

variable "vm_network" {
  type    = string
  default = "app-network"
}

variable "vm_subnet" {
  type    = string
  default = "app-subnet1"
}
ENDFIX
echo "Fixed!" && cd ../..
