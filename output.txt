def load_to_processed(
        bucket_name: str,
        dataset: str,
        clean_df: pd.DataFrame,
        quarantine_df: Optional[pd.DataFrame] = None,
        warning_df: Optional[pd.DataFrame] = None,
        backfill_date: Optional[str] = None
) -> None:
    """
    Helper function to call s3 util functions to laod process data s3 bucket
    
    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525)
        clean_df(pd.DataFrame): The clean dataframe
        quarantine_df(pd.DataFrame): The dataframe with erronous data
        warnning_df(pd.DataFrame): Dataframe with rows containing warning flangs 
    """
    if backfill_date:
        prefix = "processed/MOBIUS/CLEAN/"
        s3_filename = f"CRMA_HIVE_MOBIUS_{dataset}_CLEAN"
        target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, s3_filename, backfill_date, "PARQUET") 
        rw_util.load_to_s3(bucket_name, clean_df, target_filename, "PARQUET", True)
       
        if quarantine_df is not None:
            prefix = "processed/MOBIUS/QUARANTINE/"
            s3_filename = f"CRMA_HIVE_MOBIUS_{dataset}_QUARANTINE"
            target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, s3_filename, backfill_date, "PARQUET")
            rw_util.load_to_s3(bucket_name, quarantine_df, target_filename, "PARQUET", True)   

        if warning_df is not None:
            prefix = "processed/MOBIUS/WARNING/"
            s3_filename = f"CRMA_HIVE_MOBIUS_{dataset}_WARNING"        
            target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, s3_filename, backfill_date, "PARQUET")
            rw_util.load_to_s3(bucket_name, warning_df, target_filename, "PARQUET", True)     
    else:    
        output_file= f"processed/MOBIUS/CLEAN/CRMA_HIVE_MOBIUS_{dataset}_CLEAN"
        rw_util.load_to_s3(bucket_name, clean_df, output_file, "PARQUET")

        if quarantine_df is not None:
            output_file= f"processed/MOBIUS/QUARANTINE/CRMA_HIVE_MOBIUS_{dataset}_QUARANTINE"
            rw_util.load_to_s3(bucket_name, quarantine_df, output_file, "PARQUET")   

        if warning_df is not None:
            output_file= f"processed/MOBIUS/WARNING/CRMA_HIVE_MOBIUS_{dataset}_WARNING"
            rw_util.load_to_s3(bucket_name, warning_df, output_file, "PARQUET")

def process_trm_workbook(bucket_name: str, backfill_date: Optional[str] = None) -> None:
    """
    Handles  data cleaning and enrichement of the MOBIUS trm_workbook 
    
    In backfill mode, fetch the file whose name contains the backfill date and 
    overwrite existing file in /processed prefix.
    
    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)

    Returns:
        None
    """
    logger.info(f"Fetching data from landing zone...")
    
    # Creating variables and the dataframe
    dataset = "TRM_WORKBOOK"
    prefix = "raw_data/MOBIUS/"
    base_name = prefix + "CRMA_HIVE_MOBIUS_" + dataset

    if backfill_date:
        df = rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")
    else:
        df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")

    if df is None:
        raise ValueError("No data fetched from landing zone")

    target_columns = [
        "Hardware/Software", "Manufacturer", "Manufacturer Website", "Product",	"Product Comment",	"Product Description",	"Version",	"Version Label", "Model Series",	
        "Edition",	"Platform",	"Status",	"Status Date",	"Component",	"Max Number of Users",	"Restriction Criteria",	"Governance Record Comments",	"Fisma Restriction",	"Audit Type Acronym",	"Audit ID",	
        "Date Released",	"Release Date Source",	"Date EndOfSupport",	"EndOfSupport Date Source",	"Date EndOfExtendedSupport", "Tax Category ID",	"Foreign Owned",	"Manufacturer C-SCRM Assessment",	"Product C-SCRM Assessment",
        "Model/Version C-SCRM Assessment",	"508 Determination",	"508 URL",	"Manufacturer_ID",	"Product_ID",	"Version_ID",	"GOVERNANCE_ID",	"Fisma_System_ID",  "Product Code",	"Has AI",
    ]
    logger.info("Selecting target columns...")
    df = dataQualityUtils.select_fields(df, target_columns)

    #Replace '', 'NULL', 'NA' values with null
    df.replace(['', 'NULL', 'NA'], pd.NA, inplace=True)

    #Clean column names:
    df = dataQualityUtils.clean_column_names(df)

    #Apply DQ2:Standardize date columns to ISO 8601 format
    cols = ["status_date", "date_released",  "date_endofextendedsupport"]    
    df = dataQualityUtils.clean_date_fields(df, cols)

    #Apply DQ4: Convert numeric columns to Integer 
    cols = ["max_number_of_users", "manufacturer_id", "product_id", "version_id", "governance_id", "fisma_system_id"   ]
    df = dataQualityUtils.clean_int_fields(df, cols)

    #Apply DQ1 & W: Not nullabe, Warnning 
    cols = ["status"]
    not_nullable_warning = dataQualityUtils.apply_not_nullable_w_constraint(df, cols)

    #Apply DQ1 & Q: Not nullabe, Quarantine
    cols = ["hardware_software", "manufacturer","manufacturer_id", "product_id"]
    df, quarantine_not_null = dataQualityUtils.apply_not_nullable_q_constraint(df, cols)

    # #Merge quarantine dataframes
    quarantine_df = quarantine_not_null.copy()

    #warning df
    warning_df = not_nullable_warning.copy()

    #Load to destination 
    logger.info("Loading processed data to destination...")
    load_to_processed(bucket_name, dataset, df, quarantine_df, warning_df, backfill_date)

    #....for testing only ... #
    # load_to_test(bucket_name, dataset, df, quarantine_df, warning_df, backfill_date)
          
def run_backfill_mode(bucket_name: str, hist_table_name: str, backfill_date: str, datasets: list) -> None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    # check valid backfill date format
    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise
    
    for dataset in datasets:
        dataset_upper = dataset.upper().strip()

        # call processors with backfill_date
        if dataset_upper == "MOBIUS_TRM_WORKBOOK":
            logger.info("Processing TRM_WORKBOOK (in backfill mode)")
            process_trm_workbook(bucket_name, backfill_date)
            utility.log_memory_usage() 

def main():
    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]

    if not bucket_name:
       raise ValueError("S3 bucket not found!")

    # resolve workflow run properties
    workflow_params = utility.get_workflow_props()
    backfill_date = workflow_params.get("--backfill_date")
    hist_table_name = workflow_params.get("--hist_table_name")
    datasets = utility.parse_datasets_list( workflow_params.get("--datasets",""))

    # run in backfill mode if all backfill properties are present
    backfill_mode = bool(backfill_date and hist_table_name)

    if backfill_mode:
        run_backfill_mode(bucket_name, hist_table_name, backfill_date, datasets)
    else:
        logger.info(f"Running in Normal mode.")
        logger.info("Processing TRM_WORKBOOK")
        process_trm_workbook(bucket_name)
        utility.log_memory_usage() 
    
    # log runtime 
    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')

if __name__ == "__main__":
    main()
