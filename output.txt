class NonRetryableHTTPError(Exception):
    pass

def get_data_with_retries(
        url: str ,
        token: str,
    ) -> pd.DataFrame:
    """
    Get data with retries and  backof
    Parameters:
        url (str): api url
        token (str): api authorization token
    Returns:
        Pandas dataframe of the data retrieved from the table
    """
    session = requests.Session()
    session.headers.update({
        "Content-Type": "application/json",
        "Accept": "application/json;application/xml;odata.metadata=minimal;odata.streaming=true",
        "Connection": "keep-alive",
        "Authorization": f"Bearer {token}" 
    })

    timeout = (10, 720)
    max_retries = 10
    backoff_factor = 2.5
    attempt = 0
    retryable_status_codes = (429, 500, 502, 503, 504)  

    while True:
        try:
            # Make the GET request
            resp = session.get(url, timeout=timeout, verify=False)  
            
            # Handle successful response
            if resp.status_code == 200:
                data = resp.json()
                if not isinstance(data, list): 
                    logger.error(f"Unexpected response format: {data}")
                    raise ValueError("Invalid response format. Expected a list of records.")
                
                df = pd.DataFrame(data)
                df.drop_duplicates(keep='first', inplace=True, ignore_index=True)
                return df

            if resp.status_code not in retryable_status_codes:
                logger.error(f"Non-retryable status code {resp.status_code} received: {resp.text}")
                raise NonRetryableHTTPError(f"Non-retryable status code: {resp.status_code}")
            
            resp.raise_for_status() 

        except ValueError as ve: 
            logger.error(f"Rerutned invalid response format: {ve}")
            raise

        except NonRetryableHTTPError as nre: 
            logger.error(f"Non-retryable HTTP error occurred: {nre}")
            raise
        
        except Exception as e:
            attempt +=1
            if attempt > max_retries:
                logger.error(f"Max retries exceeded: {e}")
                raise Exception("Max retries exceeded") 
            sleep = max(30, min(backoff_factor * (2 ** attempt), 300)) # Sleep between 30 and 300 seconds
            logger.warning(f"Retry {attempt} for URL {url} after error: {e}, sleeping for {sleep} seconds")
            time.sleep(sleep)

def load_to_processed(
        bucket_name: str,
        dataset: str,
        clean_df: pd.DataFrame,
        quarantine_df: Optional[pd.DataFrame] = None,
        warning_df: Optional[pd.DataFrame] = None,
        backfill_date: Optional[str] = None
) -> None:
    """
    Helper function to call s3 util functions to laod process data s3 bucket
    
    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525)
        clean_df(pd.DataFrame): The clean dataframe
        quarantine_df(pd.DataFrame): The dataframe with erronous data
        warnning_df(pd.DataFrame): Dataframe with rows containing warning flangs 
    """
    if backfill_date:
        prefix = "processed/CSAM/CLEAN/"
        s3_filename = f"CRMA_HIVE_CSAM_{dataset}_CLEAN"
        target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, s3_filename, backfill_date, "PARQUET") 
        rw_util.load_to_s3(bucket_name, clean_df, target_filename, "PARQUET", True)
       
        if quarantine_df is not None:
            prefix = "processed/CSAM/QUARANTINE/"
            s3_filename = f"CRMA_HIVE_CSAM_{dataset}_QUARANTINE"
            target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, s3_filename, backfill_date, "PARQUET")
            rw_util.load_to_s3(bucket_name, quarantine_df, target_filename, "PARQUET", True)   

        if warning_df is not None:
            prefix = "processed/CSAM/WARNING/"
            s3_filename = f"CRMA_HIVE_CSAM_{dataset}_WARNING"        
            target_filename = rw_util.find_or_generate_filename(bucket_name, prefix, s3_filename, backfill_date, "PARQUET")
            rw_util.load_to_s3(bucket_name, warning_df, target_filename, "PARQUET", True)     
    else:    
        output_file= f"processed/CSAM/CLEAN/CRMA_HIVE_CSAM_{dataset}_CLEAN"
        rw_util.load_to_s3(bucket_name, clean_df, output_file, "PARQUET")

        if quarantine_df is not None:
            output_file= f"processed/CSAM/QUARANTINE/CRMA_HIVE_CSAM_{dataset}_QUARANTINE"
            rw_util.load_to_s3(bucket_name, quarantine_df, output_file, "PARQUET")   

        if warning_df is not None:
            output_file= f"processed/CSAM/WARNING/CRMA_HIVE_CSAM_{dataset}_WARNING"
            rw_util.load_to_s3(bucket_name, warning_df, output_file, "PARQUET")

def run_backfill_mode(bucket_name: str, hist_table_name: str, backfill_date: str, datasets: list) -> None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    # check valid backfill date format
    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise
    
    for dataset in datasets:
        dataset_upper = dataset.upper().strip()

        # call processors with backfill_date
        if dataset_upper == "CSAM_SYSTEMS_GENERAL_DETAILS_QUERY":
            logger.info("Processing SYSTEM_GENERAL_DETAILS (in backfill mode)")
            process_systems_general_details_query(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CSAM_POAM_DETAILS_REPORT":
            logger.info("Processing POAM_DETAILS (in backfill mode)")
            process_poam_details_report(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CSAM_SYSTEM_PRIVACY_DETAILS_REPORT":
            logger.info("Processing CSAM_SYSTEM_PRIVACY_DETAILS_REPORT (in backfill mode)")
            process_system_privacy_details_report(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "CSAM_SYSTEM_POINTS_OF_CONTACT_QUERY":
            logger.info("Processing CSAM_SYSTEM_POINTS_OF_CONTACT_QUERY (in backfill mode)")
            process_system_points_of_contact_query(bucket_name, backfill_date)            
            utility.log_memory_usage() 

        if dataset_upper == "CSAM_SECURITY_RTM_REPORT":
            logger.info("Processing CSAM_SECURITY_RTM_REPORT (in backfill mode)")
            process_security_rtm_report(bucket_name, backfill_date) 
            utility.log_memory_usage()

def load_data_to_db(
        engine: Engine,
        bucket_name: str,
        dataset: str,
        schema: str,
        backfill_date: Optional[str] = None
) -> None:
    """
    Loads data to desttination database
    
    Parameters:
        engine (create_engine): Database connection engine
        bucket (str): Name of the S3 bucket to fetch the data from
        dataset (str): The dataset to store to database
        schema(str): The database schema in which data is loaded.
        backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525)
    """
    
    logger.info(f"Preparing to load table '{dataset}' to '{schema}' schema")

    classes = ["CLEAN", "QUARANTINE", "WARNING"]

    for cls in classes:
        prefix = f"processed/CSAM/{cls}/CRMA_HIVE"
        base_name = f"{prefix}_{dataset}_{cls}"    
        
        if backfill_date:
            df = rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")
            if df is not None:
                df["etl_date"] = datetime.strptime(backfill_date, "%m%d%y").strftime("%Y-%m-%d")
        else:
            df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")
            if df is not None:
                df["etl_date"] = datetime.now().date() 
        
        if cls == "CLEAN":
            if df is None:
                raise ValueError("Dataset not found")
            table_name = f"crma_hive_{dataset.lower()}"
            logger.info(f"Loading table '{table_name}' to '{schema}' schema")
            rw_util.truncate_and_load_table(engine=engine, df=df, schema_name=schema, table_name=table_name, exist_type="append" )
        else:
            if df is not None:                    
                table_name = f"crma_hive_{dataset.lower()}_{cls.lower()}"
                logger.info(f"Loading table '{table_name}' to '{schema}' schema")               
                rw_util.truncate_and_load_table(engine=engine, df=df, schema_name=schema, table_name=table_name, exist_type="append" )

        logger.info(f"Perfomming post load validation on table '{table_name}' in '{schema}' schema")
        if rw_util.post_load_validation(engine=engine, df=df, schema_name=schema, table_name=table_name):
            logger.info(f"Post load validation check passed for table '{table_name}' in '{schema}' schema")

            ##Parallel loading##
            logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
            logger.info(f"Loading table '{table_name}' to '{schema}' schema")     
            engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
            rw_util.truncate_and_load_table(engine=engine2, df=df, schema_name=schema, table_name=table_name, exist_type="append" ) 

        else:
            raise RuntimeError(f"Post load validation check failed for table '{table_name}' in '{schema}' schema")                      

def run_backfill_mode(engine: Engine, schema: str, bucket_name: str, hist_table_name: str, backfill_date: str, datasets: list) -> None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        engine (str): SQLalchemy engine
        schema (str): The database schema in which data is loaded.
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    # check valid backfill date format
    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise

    # enforce valid tables in datasets list when running in backfill mode
    valid_datasets = [t for t in datasets if t in VALID_TABLES]

    if len(valid_datasets) != len(datasets):
        invalid_tables = set(datasets) - set(valid_datasets)
        logger.warning(f"Invalid tables ignored: {invalid_tables}. Valid tables used: {valid_datasets}")
        datasets = valid_datasets
    else:
        logger.info(f"All {len(datasets)} tables in dataset are valid")

    if not datasets:
        logger.warning(F"No CSAM tables found!")

    for dataset in datasets:
        load_data_to_db(engine, bucket_name, dataset, schema, backfill_date)
