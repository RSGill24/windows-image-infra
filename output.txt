def build_fisma_hwam_score_sql() -> str:
    """
    Build the HWAM Score per device data preparation SQL Command using CTEs.
    """

    sql = f"""
        WITH
        -- Step 1: Select eligible devices 
        eligible_devices AS(
            SELECT 
                device_id,
                fisma_id, 
                devicetype,
                preferred_hostname,
                preferred_ipv4,
                os,
                device_last_seen
            FROM crma_hive.crma_hive_fismascorecard_input_devices
            WHERE fisma_id iLIKE 'ICE-%%'
        ),
        -- Step 2: Apply identification rule
        asset_identification_status AS (
        SELECT
                *,
                CASE
                    WHEN devicetype IS NOT NULL
                    AND preferred_ipv4 IS NOT NULL
                    AND preferred_hostname IS NOT NULL
                    AND (
                        -- If device_type is 'server', 'workstation', or 'laptop', device_os must NOT be null
                        (
                            devicetype IN ('ENDPOINT - SERVER', 'ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                            AND os IS NOT NULL
                        )
                        OR
                        (
                            devicetype NOT IN ('ENDPOINT - SERVER', 'ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                        )
                    )
                    AND (
                        -- If device_type is 'workstation' or 'laptop', hostname cannot be an IP address
                        (
                            devicetype IN ('ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                            AND preferred_hostname !~ '^\d{1,3}(\.\d{1,3}){3}$'
                        )
                        OR
                        (
                            devicetype NOT IN ('ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                        )
                    )
                    THEN TRUE
                    ELSE FALSE
                END AS is_identified
            FROM eligible_devices
        ),
        -- Step 3:  Count assets by catagory
        asset_counts AS (
            SELECT
                fisma_id,
                COUNT(*) FILTER (WHERE is_identified = TRUE) AS identified_and_managed_assets,
                COUNT(*) AS managed_assets,
                COUNT(*) AS total_assets
            FROM asset_identification_status
            GROUP BY fisma_id
        ),
        -- Step 4: Score
        scores AS (
            SELECT
                fisma_id,
                identified_and_managed_assets,
                managed_assets,
                total_assets,
                CASE
                    WHEN managed_assets = 0 OR total_assets = 0 THEN NULL
                    ELSE ROUND(
                        (identified_and_managed_assets::float / managed_assets) * 100
                    )::int
                END AS hwam_score
            FROM asset_counts
        ),
        --  Step 6: Enrich with fisma details
        fisma_hwam_scores_enriched AS (
            SELECT 
                f.fisma_id,
                f.fisma_name,
                f.security_support_portfolio AS portfolio,
                f.dhs_hva,
                f.public_facing,
                f.cfo_designated,
                f.fips_199_security_categorization_overall,
                f.active,
                f.iscm_waiver, 
                f.fisma_reportable,
                f.fisma_inventory_selc_status,
                f.fisma_category,
                s.identified_and_managed_assets,
                s.managed_assets,
                s.total_assets,
                s.hwam_score    
            FROM crma_hive.crma_hive_snow_fismadetails f 
            LEFT JOIN scores s
                ON s.fisma_id = f.fisma_id
        )
        SELECT 
            fisma_id,
            fisma_name,
            portfolio,
            dhs_hva,
            public_facing,
            cfo_designated,
            fips_199_security_categorization_overall,
            active,
            iscm_waiver, 
            fisma_reportable,
            fisma_inventory_selc_status,
            fisma_category,
            identified_and_managed_assets,
            managed_assets,
            total_assets,
            hwam_score 
        FROM fisma_hwam_scores_enriched

    """

    return sql

def build_component_hwam_score_sql() -> str:
    """
    Build the HWAM Score for component(ICE) System data preparation SQL Command using CTEs. 
    """
    sql = f"""
        WITH
        -- Step 1: Select eligible devices 
        eligible_devices AS(
            SELECT 
                d.device_id,
                d.fisma_id, 
                d.devicetype,	
				d.asset_auth_status,
                d.preferred_hostname,
                d.preferred_ipv4,
                d.os,
                d.device_last_seen,
                f.fisma_reportable,
                f.iscm_waiver,
                f.fisma_inventory_selc_status,
                f.fisma_category
            FROM  crma_hive.crma_hive_fismascorecard_input_devices  d
            LEFT JOIN crma_hive.crma_hive_snow_fismadetails f
                ON d.fisma_id=f.fisma_id
        ),
        eligible_fisma AS(
            SELECT 
                *
            FROM  eligible_devices		
            WHERE ((fisma_reportable iLike 'yes'
                AND fisma_category in ('MAJ', 'GSS')
                AND fisma_inventory_selc_status iLike 'Operational'
                AND COALESCE(iscm_waiver, '') not ILIKE 'yes'
                ) 
                OR fisma_id is null
            )
        ),
        -- Step 2: Get asset managment status
        asset_management_status AS (
            SELECT
                *,
                CASE
                    WHEN asset_auth_status IS NOT NULL THEN 'managed'
                    ELSE 'unauthorized'
                END AS asset_status
            FROM eligible_fisma
        ),
        -- Step 3: Apply identification rule
        asset_identification_status AS (
        SELECT
                *,
                CASE
                    WHEN devicetype IS NOT NULL
                    AND preferred_ipv4 IS NOT NULL
                    AND preferred_hostname IS NOT NULL
                    AND (
                        -- If devicetype is 'server', 'workstation', or 'laptop', device os must NOT be null
                        (
                            devicetype IN ('ENDPOINT - SERVER', 'ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                            AND os IS NOT NULL
                        )
                        OR
                        (
                            devicetype NOT IN ('ENDPOINT - SERVER', 'ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                        )
                    )
                    AND (
                        -- If devicetype is 'workstation' or 'laptop', hostname cannot be an IP address
                        (
                            devicetype IN ('ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                            AND preferred_hostname !~ '^\d{1,3}(\.\d{1,3}){3}$'
                        )
                        OR
                        (
                            devicetype NOT IN ('ENDPOINT - WORKSTATION', 'ENDPOINT - LAPTOP')
                        )
                    )
                    THEN TRUE
                    ELSE FALSE
                END AS is_identified
            FROM asset_management_status
        ),
        -- Step 4:  Count assets by catagory
        component_asset_counts AS (
            SELECT
                'ICE' AS component,
                COUNT(*) FILTER (WHERE is_identified = TRUE AND asset_status = 'managed') AS identified_and_managed_assets,
                COUNT(*) FILTER (WHERE asset_status = 'managed') AS managed_assets,
                COUNT(*) FILTER (WHERE asset_status = 'unauthorized') AS unauthorized_assets,
                (SELECT COUNT(*) FROM eligible_devices) AS total_assets	
            FROM asset_identification_status
        ),
        componenet_hwan_score AS (
            SELECT
                component,
                identified_and_managed_assets,
                managed_assets,
                unauthorized_assets,
                total_assets,
                CASE
                    WHEN managed_assets = 0 OR total_assets = 0 THEN NULL
                    ELSE ROUND(
                        ((identified_and_managed_assets::float / managed_assets) - (unauthorized_assets::float / total_assets)) * 100
                    )::int
                END AS hwam_score
            FROM component_asset_counts
        )
        SELECT 
            component,
            identified_and_managed_assets,
            managed_assets,
            unauthorized_assets,
            total_assets,
            hwam_score
        FROM componenet_hwan_score

            """
    return sql

def run_hwam_score_workflow(
        engine: Engine,
        bucket_name: str
    )->None:
    """
    Execute the Software Assets Management(HWAM) score workflow, including building query, 
    param preparation, execute query, store output.

    Parameters:
        engine (Engine): Database engine for data access.
        bucket_name (str): Name of the storage bucket for output.

    Returns:
        None
    """
    etl_date= date.today()
    logger.info(f"Computing Scorecard HWAM Score for : {etl_date}") 


    logger.info("Build FISMA Systems HWAM score query.")
    sql = build_fisma_hwam_score_sql()


    logger.info("Excecute FISMA Systems HWAM score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
    )
    
    if df is not None:
        logger.info(f"Per FISMA System HWAM score received. Total records: {len(df)}")         
        logger.info(f"Load fisma system score dataset to target table.")
        df["etl_date"] = etl_date
        dataset = "crma_hive_fismascorecard_fisma_system_hwam_score"
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )

    else:
        logger.warning("Query returned no data")   
        return 

    logger.info("Build Component HWAM score query.")
    sql = build_component_hwam_score_sql()

    logger.info("Excecute Componenet HWHM score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
    )

    if df is not None:       
        logger.info(f"Load Component HWAM score dataset to target table and save a copy to S3 for archival.")
        df["etl_date"] = etl_date 
        dataset='crma_hive_fismascorecard_component_hwam_score'
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )
       
    else:
        logger.warning("Query returned no data")      
         

def main():

    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 


    run_hwam_score_workflow(engine=engine, bucket_name=bucket_name)       
      
    utility.log_memory_usage() 
    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")


if __name__ == "__main__":
    main()
