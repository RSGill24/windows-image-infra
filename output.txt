def build_system_security_authz_score_sql() -> str:
    """
    Build the Security Authz Score per system data preparation SQL Command using CTEs.
    """

    sql = f"""
        WITH 
        -- Step 2.0: Select eligible systems
        eligible_systems AS (
            SELECT  csam.*, fisma.fisma_category, fisma.iscm_waiver, (SELECT MAX(dt) FROM unnest(csam.published_in_federal_register) as dt) as sorn_date
            FROM crma_hive.crma_hive_csam_systems_general_details_query csam
            left join crma_hive.crma_hive_snow_fismadetails fisma 
            ON csam.system_id = fisma.csam_id AND trim(csam.fisma_id) = trim(fisma.fisma_id)
            WHERE csam.fisma_reportable = 'Yes' AND csam.system_operational_status = 'Operational' 
            AND trim(fisma.fisma_category) in ('GSS','MAJ')
        ),
        -- Step 2.1: Compute partial security authz checks
        partial_securith_auth_checks AS(
            SELECT 
            *,
            CASE
                WHEN s.ato_date IS NOT NULL 
                AND s.ato_date <= current_date 
                AND (s.ato_expiration IS NULL OR s.ato_expiration >= current_date )
                THEN TRUE
                ELSE FALSE
            END AS auth_status_check,
     		CASE
     			WHEN s.document_review_approval_date_completed IS NOT NULL 
     			AND s.document_review_approval_date_completed <= current_date  
     			THEN TRUE
     			ELSE FALSE
     		END AS document_review_check,		
            CASE
                WHEN s.contingency_plan_review_date_completed IS NOT NULL 
                AND s.contingency_plan_review_date_completed <= current_date 
                THEN TRUE
                ELSE FALSE
            END AS contingency_plan_check,
            CASE
                WHEN s.contingency_plan_test_review_date_completed IS NOT NULL 
                AND s.contingency_plan_test_review_date_completed <= current_date 
                AND s.contingency_plan_test_review_date_completed >= NOW() - INTERVAL '1 year' 
                THEN TRUE
                ELSE FALSE
            END AS contingency_plan_test_check,		
            CASE
                WHEN s.pta_ipa_date_completed IS NOT NULL
                AND TO_DATE(s.pta_ipa_date_completed, 'MM-DD-YYYY') <= current_date
                AND TO_DATE(s.pta_ipa_date_completed, 'MM-DD-YYYY') >= NOW() - INTERVAL '3 year' 
                THEN TRUE
                ELSE FALSE
            END AS pta_valid,
            CASE
                WHEN trim(s.privacy_impact_assessment_status) = 'Required under E-Government Act'
                THEN TRUE
                ELSE FALSE
            END as pia_required,
            CASE
                WHEN trim(s.privacy_impact_assessment_status) = 'Required under E-Government Act'
                AND s.privacy_impact_assessment_date_completed IS NOT NULL
                AND s.privacy_impact_assessment_date_completed <= current_date
                THEN TRUE
                WHEN trim(s.privacy_impact_assessment_status) = 'Not Required under E-Government Act'
                THEN TRUE
                ELSE FALSE
            END AS pia_valid,
            CASE
                WHEN trim(s.system_of_records_notice) = 'Required under Privacy Act'
                THEN TRUE
                ELSE FALSE
            END as sorn_required,		
            CASE
                WHEN trim(s.system_of_records_notice) = 'Required under Privacy Act'
				AND s.sorn_date IS NOT NULL 
				AND s.sorn_date <= current_date 
                THEN TRUE
                WHEN trim(s.system_of_records_notice) = 'Not Required under Privacy Act'
                THEN TRUE
                ELSE FALSE
            END AS sorn_valid
            FROM eligible_systems s
        ),
        -- Step 2.2: Compute full security authz checks        
        full_securith_auth_checks AS(
            SELECT 
            *,
            CASE 
                WHEN s.pta_valid = TRUE
                AND (s.pia_required = FALSE OR s.pia_valid = TRUE)
                AND (s.sorn_required = FALSE OR s.sorn_valid = TRUE)
                THEN TRUE
                ELSE FALSE
            END as privacy_check
            FROM partial_securith_auth_checks s
        ),
        -- Step 3.0: Compute full security authz checks           
        securith_authz_score AS(
            SELECT 
     		*,
            CASE
                WHEN s.auth_status_check = TRUE
     			AND s.document_review_check = TRUE
                AND s.contingency_plan_check = TRUE
                AND s.contingency_plan_test_check = TRUE
                AND s.privacy_check = TRUE
                THEN 1
                ELSE 0
            END as score
            FROM full_securith_auth_checks s
        )	
        SELECT 
            fisma_id,
            system_name,
            org,
            sub_org as portfolio,
            acronym,
            high_value_asset as dhs_hva,
            fisma_reportable,
	        fisma_category,
            system_category as fips_199_security_categorization_overall,
            system_operational_status as fisma_inventory_selc_status,
            cfo_designation as cfo_designated,
            contractor_system,
            iscm_waiver,
        --  iscm_waiver_exp_date,
            ato_type,
            ato_date as authorization_date,
            ato_expiration,
            document_review_approval_date_completed as document_review_date,
            contingency_plan_date_completed as contingency_plan_date,
            contingency_plan_test_date_completed as contingency_plan_test_date,
            pta_ipa_date_completed as privacy_threshold_analysis_date,
            privacy_impact_assessment_date_completed as privacy_impact_assesment_date,
            pia_required,
            sorn_date,
            sorn_required,
            auth_status_check,
            document_review_check,
            contingency_plan_check,
            contingency_plan_test_check,
            privacy_check,
            pta_valid,
            pia_valid,
            sorn_valid,
            score as security_authz_score
        FROM securith_authz_score
    """

    return sql    


def run_security_authz_score_workflow(
        engine: Engine,
        bucket_name: str
    )->None:
    """
    Execute the Security Authorization score workflow, including building query, 
    param preparation, execute query, store output.

    Parameters:
        engine (Engine): Database engine for data access.
        bucket_name (str): Name of the storage bucket for output.

    Returns:
        None
    """
    etl_date= date.today()
    logger.info(f"Computing Scorecard Sec Authz Score for : {etl_date}") 

    logger.info("Build Security Authz score query for FISMA Systems.")
    sql = build_system_security_authz_score_sql()


    logger.info("Excecute Security Authz score query for FISMA Systems.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql
    )
    
    if df is not None:
        logger.info(f"Total number of FISMA Systems scored: {len(df)}")         
        logger.info(f"Load FISMA score dataset to target table.")
        df["etl_date"] = etl_date
        dataset = "crma_hive_fismascorecard_fisma_system_security_authz_score"
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )

    else:
        logger.warning("Query returned no data")
             

def main():

    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 

    run_security_authz_score_workflow(engine=engine, bucket_name=bucket_name)       
      
    utility.log_memory_usage() 
    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")


if __name__ == "__main__":
    main()
