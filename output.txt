import pandas as pd
import logging
import time
import gc
import urllib3
from crma_hive_connections import Connection
from crma_hive_utilities import Utility, Read_Write_Utility
from crma_hive_tvm_utilities import TvmUtility
from typing import Optional

# Turning off warnings
urllib3.disable_warnings()

# Setting up logger
logger = logging.getLogger(__name__)
logger.setLevel(logging.INFO)

# Instantiate classes
connection = Connection()
utility = Utility()
rw_util = Read_Write_Utility()
tvmUtil = TvmUtility()


def main():
    start_time = time.time()

    logger.info("Retrieving account credentials...")
    ACCESS_KEY = connection.get_secret("edae-crma-hive")["tvm_access_key"]
    SECRETE_SECRET = connection.get_secret("edae-crma-hive")["tvm_secrete_key"]
    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    base_url ="https://fedcloud.tenable.com"

    # log memory usage
    utility.log_memory_usage()
    #setup
    headers = {
        'accept': 'application/json',
        'content-type': 'application/json',
        'X-ApiKeys': f'accessKey={ACCESS_KEY}; secretKey={SECRETE_SECRET}'
    }
 
    targets = [
    {
            "dataset": "vulns",
            "output_filename": "CRMA_HIVE_TVM_VULNS",
            "payload": {
                "filters": {
                    "severity": ["low", "medium", "high", "critical"],
                    "state": ["OPEN", "REOPENED", "FIXED" ],
                },            
            "num_assets": 1500
            }        
    },  
    ]

    for target in targets:    

        dataset =   target["dataset"] 

        logger.info(f"Fetching TenableVM {dataset} dataset using Tenable export API") 

        if dataset=="vulns":
            tvmUtil.setup_and_start_tvm_vulns_export(
                base_url=base_url,
                headers = headers,
                target = target,
                bucket_name=bucket_name,
            )
        utility.log_memory_usage() 

    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")


if __name__ == "__main__":
    main()
