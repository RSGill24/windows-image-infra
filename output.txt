import logging

import pandas as pd
import time
from datetime import datetime, date
import os

import urllib3

from crma_hive_connections import Connection
from crma_hive_utilities import Utility, Read_Write_Utility
from crma_hive_scorecard_utilities import FismaScorecardUtility


from sqlalchemy.engine import Engine

# Turning off warnings
urllib3.disable_warnings()

# Setting up logger
logger = logging.getLogger(__name__)
logger.setLevel(logging.INFO)

utility = Utility()
connection = Connection()
rw_util = Read_Write_Utility()
fismaScorecardUtility = FismaScorecardUtility()

def build_device_hbd_score_sql() -> str:
    """
    Build the HBD Score per device data preparation SQL Command using CTEs.
    """

    sql = f"""
        WITH 
        -- Step 1: Select eligible devices
        eligible_devices AS (
            SELECT 
                device_id,
                preferred_hostname AS hostname, 
                preferred_ipv4  AS ip,
                fisma_id,
                adapters,
                device_last_seen,
                devicerole,
                crowd_strike_last_seen,
                edr_enabled
            FROM  crma_hive.crma_hive_fismascorecard_input_devices
            WHERE devicerole = 'ENDPOINT'
                AND fisma_id iLIKE 'ICE-%%'
        ),
        -- Step 2: Add a flag to indicate if a device is associated with CrowdStrike adapter or not
        device_crowd_strike_adapter_association AS(
            SELECT
                device_id,
                hostname, 
                ip,
                fisma_id,
                adapters,
                device_last_seen,
                crowd_strike_last_seen,
                edr_enabled,
                ARRAY_POSITION(adapters, 'crowd_strike_adapter') IS NOT NULL AS crowd_strike_adapter
            FROM eligible_devices
        ),
        -- Step 3: Add EDR and AV compliance field
        device_edr_and_av_compliance AS (
            SELECT
                device_id,
                hostname, 
                ip,
                fisma_id,
                adapters,
                device_last_seen,
                crowd_strike_last_seen,
                edr_enabled,
                crowd_strike_adapter,
                CASE 
                    WHEN crowd_strike_last_seen >= current_date - INTERVAL '15' DAY THEN 1
                    ELSE 0
                END AS av_compliance,
                CASE 
                    WHEN edr_enabled IS True THEN 1
                    ELSE 0
                END AS edr_compliance
            FROM device_crowd_strike_adapter_association
        ),
        -- Step 4: Score
        device_hbd_score AS(
            SELECT
                device_id,
                hostname, 
                ip,
                fisma_id,
                device_last_seen,
                adapters,
                crowd_strike_last_seen,
                edr_enabled,
                crowd_strike_adapter,
                av_compliance,
                edr_compliance,
                CASE
                    WHEN crowd_strike_adapter IS NOT TRUE THEN 0
                    ELSE ROUND(((0.5 * av_compliance) + (0.5 * edr_compliance)) * 100)::int
                END AS hbd_score
            FROM device_edr_and_av_compliance
        ),
        -- Step 5: Enrich with FISMA system information
        device_hbd_scores_enriched AS (
            SELECT 
                dhs.device_id,
                dhs.hostname, 
                dhs.ip,
                dhs.fisma_id,
                fd.fisma_name,
                fd.security_support_portfolio AS portfolio,
                fd.dhs_hva,
                fd.public_facing,
                fd.cfo_designated,
                fd.fips_199_security_categorization_overall,
                fd.active,
                fd.iscm_waiver, 
                fd.fisma_reportable,
                fd.fisma_inventory_selc_status,
                fd.fisma_category,
                dhs.adapters,
                dhs.device_last_seen,
                dhs.crowd_strike_last_seen,
                dhs.edr_enabled,
                dhs.crowd_strike_adapter,
                dhs.av_compliance,
                dhs.edr_compliance,
                dhs.hbd_score		 
            FROM device_hbd_score dhs
            LEFT JOIN crma_hive.crma_hive_snow_fismadetails fd
                ON dhs.fisma_id = fd.fisma_id
        )
        SELECT 
            device_id,
            hostname, 
            ip,
            fisma_id,
            fisma_name,
            portfolio,
            dhs_hva,
            public_facing,
            cfo_designated,
            fips_199_security_categorization_overall,
            active,
            iscm_waiver, 
            fisma_reportable,
            fisma_inventory_selc_status,
            fisma_category,
            adapters,
            device_last_seen,
            crowd_strike_last_seen,
            edr_enabled,
            crowd_strike_adapter,
            av_compliance,
            edr_compliance,
            hbd_score		 
        FROM device_hbd_scores_enriched
      
    """

    return sql

def build_fisma_hbd_score_sql() -> str:
    """
    Build the HBD Score per FISMA System data preparation SQL Command using CTEs.
    """
    sql = f"""
        WITH 
        fisma_hbd_score AS (
            SELECT
                fisma_id,
                ROUND(AVG(hbd_score))::integer AS  hbd_score
            FROM crma_hive.crma_hive_fismascorecard_device_hbd_score
            WHERE etl_date = %(etl_date)s
            GROUP BY fisma_id
        )
        SELECT 
            a.fisma_id,
            a.fisma_name,
            a.security_support_portfolio AS portfolio,
            a.dhs_hva,
            a.public_facing,
            a.cfo_designated,
            a.fips_199_security_categorization_overall,
            a.active,
            a.iscm_waiver,
            a.fisma_reportable,
            a.fisma_inventory_selc_status,
            a.fisma_category,
            b.hbd_score
        FROM crma_hive.crma_hive_snow_fismadetails a
        LEFT JOIN fisma_hbd_score b
            ON a.fisma_id = b.fisma_id
    """

    return sql

def run_hbd_score_workflow(
        engine: Engine,
        bucket_name: str
    )->None:
    """
    Execute the Configration Management score workflow, including building query, 
    param preparation, execute query, store output.

    Parameters:
        engine (Engine): Database engine for data access.
        bucket_name (str): Name of the storage bucket for output.

    Returns:
        None
    """
    etl_date= date.today()
    logger.info(f"Computing Scorecard HBD Score for : {etl_date}") 

    logger.info("Build Device HBD score query.")
    sql = build_device_hbd_score_sql()


    logger.info("Excecute Device HBD score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
    )
    
    if df is not None:
        logger.info(f"Per device hbd score received. Total number of devices: {len(df)}")         
        logger.info(f"Load device score dataset to target table.")
        df["etl_date"] = etl_date
        dataset = "crma_hive_fismascorecard_device_hbd_score"
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )

    else:
        logger.warning("Query returned no data")   
        return 

    logger.info("Build FISMA System HBD score query.")
    sql = build_fisma_hbd_score_sql()

    logger.info("Excecute FISMA System HBD score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
        params={"etl_date": etl_date}
    )

    if df is not None:       
        logger.info(f"Load FISMA System HBD score dataset to target table and save a copy to S3 for archival.")
        df["etl_date"] = etl_date 
        dataset='crma_hive_fismascorecard_fisma_system_hbd_score'
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )
       
    else:
        logger.warning("Query returned no data")      
         

def main():

    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 

    run_hbd_score_workflow(engine=engine, bucket_name=bucket_name)       
      
    utility.log_memory_usage() 
    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")


if __name__ == "__main__":
    main()
