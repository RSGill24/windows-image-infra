def fetch_single_report(
    engine: Engine,
    sql_statment: str,
    query_params: Optional[dict] = None,
) -> pd.DataFrame:
    """
    Executes one SQL query against the CRMA HIVE database and returns the result as a pandas DataFrame.

    Parameters:
        engine: Active SQLAlchemy database engine.
        sql_statment: SQL statement text to execute.
        query_params: Optional dictionary of query parameters to pass into the database query utility.

    Returns:
        pd.DataFrame: Query results as a DataFrame. Returns an empty DataFrame if the query utility returns None.
    """
    logger.info("Executing database query to fetch a single data report.")

    dataframe = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql_statment,
        params=query_params,
    )

    if dataframe is None:
        dataframe = pd.DataFrame()

    logger.info(f"Database query completed. Returned {len(dataframe)} row(s).")

    return dataframe

def fetch_all_reports(
    engine: Engine,
    report_queries: list[ReportQuery],
    query_params: Optional[dict] = None,
) -> dict[str, pd.DataFrame]:
    """
    Executes multiple configured report queries and returns a dictionary of worksheet names mapped to DataFrames.

    Parameters:
        engine: Active SQLAlchemy database engine.
        report_queries: List of ReportQuery objects containing worksheet names and SQL statements.
        query_params: Optional query parameters to pass into each database query.

    Returns:
        dict[str, pd.DataFrame]: Dictionary keyed by worksheet name, with each value containing the corresponding query result DataFrame.
    """
    reports = {}

    for report_query in report_queries:
        logger.info(f"Fetching report worksheet: {report_query.worksheet_name}")

        reports[report_query.worksheet_name] = fetch_single_report(
            engine=engine,
            sql_statment=report_query.sql,
            query_params=query_params,
        )

    return reports

def evaluate_sync_status(
    count_dataframe: pd.DataFrame,
    health_check: list[str],
    exclude_columns: Optional[list[str]] = None
) -> bool:
    """
    Evaluates a health-check DataFrame using a caller-provided list of count columns.
    The report is considered healthy only when all configured health-check count columns are zero.

    Parameters:
        count_dataframe: DataFrame containing one or more count columns used for health evaluation.
        health_check: List of column names to inspect. Each column should contain a numeric count where zero
                      means no discrepancy and any value greater than zero means maintenance is required.
        exclude_columns: Optional list of column names to exclude from health evaluation.             

    Returns:
        bool: True if all configured health-check values are zero.
              False if any configured health-check value is greater than zero.
    """
    logger.info("Evaluating health check metrics.")

    if count_dataframe is None or count_dataframe.empty:
        raise ValueError("Health check dataframe is empty.")

    if not health_check:
        raise ValueError("health_check must contain at least one column name.")

    exclude_columns = exclude_columns or []
    exclude_columns_set = set(exclude_columns)

    filtered_health_check = [
        column for column in health_check
        if column not in exclude_columns_set
    ]

    if not filtered_health_check:
        raise ValueError(
            "No health-check columns remain after applying exclude_columns."
        )

    missing_columns = [
        column for column in filtered_health_check
        if column not in count_dataframe.columns
    ]

    if missing_columns:
        raise ValueError(
            f"Health check dataframe is missing required column(s): {missing_columns}"
        )

    counts = count_dataframe.loc[:, filtered_health_check].iloc[0]
    counts = pd.to_numeric(counts, errors="coerce").fillna(0)

    is_in_sync = not (counts > 0).any()

    logger.info(f"Evaluated health check columns: {filtered_health_check}")
    logger.info(f"Excluded health check columns: {exclude_columns}")    
    logger.info(f"Health check counts: {counts.to_dict()}")
    logger.info(f"In-sync status: {is_in_sync}")

    return bool(is_in_sync)

def normalize_sync_summary(
    count_dataframe: pd.DataFrame,
    health_check: list[str],
) -> pd.DataFrame:
    """
    Converts a single-row health-check DataFrame into a normalized two-column summary format.
    Output columns: aggregate_name, count

    Parameters:
        count_dataframe: DataFrame containing the health-check count columns.
        health_check: List of health-check column names to include in the normalized output.

    Returns:
        pd.DataFrame: Normalized DataFrame with columns aggregate_name and count.
    """
    logger.info("Normalizing health check summary.")

    if count_dataframe is None or count_dataframe.empty:
        return pd.DataFrame(columns=["aggregate_name", "count"])

    if not health_check:
        raise ValueError("health_check must contain at least one column name.")

    missing_columns = [
        column for column in health_check
        if column not in count_dataframe.columns
    ]

    if missing_columns:
        raise ValueError(
            f"Cannot normalize sync summary. Missing column(s): {missing_columns}"
        )

    row = count_dataframe.loc[:, health_check].iloc[0]
    row = pd.to_numeric(row, errors="coerce").fillna(0)

    return pd.DataFrame(
        {
            "aggregate_name": health_check,
            "count": [int(row[column]) for column in health_check],
        }
    )

def create_in_memory_excel(dataframes_dict: dict[str, pd.DataFrame]) -> io.BytesIO:
    """
    Creates a multi-tab Excel workbook in memory. Each dictionary item is written to one worksheet.

    Parameters:
        dataframes_dict: Dictionary where each key is a worksheet name and each value is the DataFrame to write to that worksheet.

    Returns:
        io.BytesIO: Binary stream containing the generated Excel workbook.
    """
    logger.info("Starting generation of multi-tab Excel workbook.")

    excel_stream = io.BytesIO()

    with pd.ExcelWriter(excel_stream, engine="xlsxwriter") as writer:
        for worksheet_name, dataframe in dataframes_dict.items():
            safe_sheet_name = str(worksheet_name)[:31]

            if dataframe is None:
                dataframe = pd.DataFrame()

            dataframe.to_excel(
                writer,
                sheet_name=safe_sheet_name,
                index=False,
            )

    excel_stream.seek(0)

    logger.info("Multi-tab Excel workbook generated successfully.")

    return excel_stream

def _extract_history_records(
    dataframe: pd.DataFrame,
    source_column: str,
    system_type: str,
    change_type: str,
    counter_column: str,
    entity_column: str,
) -> pd.DataFrame:
    """
    Extracts and normalizes one discrepancy DataFrame into the unified software
    history format.

    Parameters:
        dataframe: Source discrepancy DataFrame.
        source_column: Source software-name column to extract.
        system_type: Target system type value.
        change_type: Target change type value.
        counter_column: Source column containing the calculated consecutive_weeks value.
        entity_column: Target entity-name column in the history dataframe, such as software_name or os_name.

    Returns:
        pd.DataFrame: DataFrame with software_name, system_type, change_type, and consecutive_weeks.
    """
    if not entity_column or str(entity_column).strip() == "":
        raise ValueError("entity_column is required.")

    entity_column = str(entity_column).strip()

    output_columns = [
        entity_column,
        "system_type",
        "change_type",
        "consecutive_weeks",
    ]

    if dataframe is None or dataframe.empty:
        return pd.DataFrame(columns=output_columns)

    required_columns = [
        source_column,
        counter_column,
    ]

    missing_columns = [
        column for column in required_columns
        if column not in dataframe.columns
    ]

    if missing_columns:
        raise ValueError(
            f"Report dataframe is missing required column(s): {missing_columns}"
        )

    normalized_dataframe = dataframe.loc[:, [source_column, counter_column]].copy()

    normalized_dataframe = normalized_dataframe.rename(
        columns={
            source_column: entity_column,
            counter_column: "consecutive_weeks",
        }
    )

    normalized_dataframe[entity_column] = (
        normalized_dataframe[entity_column]
        .astype("string")
        .str.strip()
    )

    normalized_dataframe = normalized_dataframe[
        normalized_dataframe[entity_column].notna()
        & (normalized_dataframe[entity_column] != "")
    ]

    normalized_dataframe["system_type"] = system_type
    normalized_dataframe["change_type"] = change_type

    normalized_dataframe["consecutive_weeks"] = pd.to_numeric(
        normalized_dataframe["consecutive_weeks"],
        errors="coerce",
    ).fillna(1).astype(int)

    normalized_dataframe = normalized_dataframe.drop_duplicates(
        subset=[entity_column, "system_type", "change_type"],
        keep="first",
    )

    return normalized_dataframe.loc[:, output_columns]

def _validate_source_mappings(source_mappings: list[dict[str, str]]) -> None:
    """
    Validates configurable source mappings used for software audit history.

    Each mapping must include:
        worksheet_name: Key in the reports dictionary.
        source_column: Source column containing the software name.
        system_type: System type value to write to history.
        change_type: Change type value to write to history.
        counter_column: Source column containing the calculated consecutive_weeks value.

    Parameters:
        source_mappings: List of mapping dictionaries.

    Returns:
        None

    Raises:
        ValueError: If source_mappings is empty or malformed.
    """
    if not source_mappings:
        raise ValueError("source_mappings must contain at least one mapping.")

    required_mapping_keys = [
        "worksheet_name",
        "source_column",
        "system_type",
        "change_type",
        "counter_column",
    ]

    for index, mapping in enumerate(source_mappings):
        if not isinstance(mapping, dict):
            raise ValueError(
                f"source_mappings[{index}] must be a dictionary."
            )

        missing_keys = [
            key for key in required_mapping_keys
            if key not in mapping or mapping[key] is None or str(mapping[key]).strip() == ""
        ]

        if missing_keys:
            raise ValueError(
                f"source_mappings[{index}] is missing required key(s): {missing_keys}"
            )

def build_audit_report_dataframe(
    reports: dict[str, pd.DataFrame],
    source_mappings: list[dict[str, str]],
    entity_column: str,
) -> pd.DataFrame:
    """
    Builds today's persistent audit report DataFrame from report outputs.
    - Extracts the configured source entity-name column.
    - Renames it to the configured entity_column.
    - Adds configured system_type and change_type values.
    - Extracts the configured counter_column as consecutive_weeks.
    - Deduplicates by entity_column, system_type, change_type.
    - Adds today's etl_date.


    Parameters:
        reports: Dictionary of report DataFrames from fetch_all_reports.
        source_mappings: Configurable list describing how report worksheets map into software history records.
        target_table: Name of target table dataframe will be loaded into
        entity_column: Target entity-name column for the output dataframe. Examples: software_name, os_name.

    Returns:
        pd.DataFrame: Today's history records ready for insertion.
    """
    logger.info("Building today's software audit history dataframe from report query counters.")

    if reports is None:
        raise ValueError("reports dictionary is required.")

    if not entity_column or str(entity_column).strip() == "":
        raise ValueError("entity_column is required.")

    entity_column = str(entity_column).strip()    

    _validate_source_mappings(source_mappings)

    today = date.today()
    history_frames = []

    for mapping in source_mappings:
        worksheet_name = mapping["worksheet_name"]
        source_column = mapping["source_column"]
        system_type = mapping["system_type"]
        change_type = mapping["change_type"]
        counter_column = mapping["counter_column"]

        logger.info(
            "Extracting software history records from worksheet "
            f"{worksheet_name} using source column {source_column} "
            f"and counter column {counter_column}, and entity column {entity_column}."
        )

        history_frames.append(
            _extract_history_records(
                dataframe=reports.get(worksheet_name),
                source_column=source_column,
                system_type=system_type,
                change_type=change_type,
                counter_column=counter_column,
                entity_column=entity_column
            )
        )

    expected_columns = [
        entity_column,
        "system_type",
        "change_type",
        "consecutive_weeks",
        "etl_date",
    ]

    today_history_dataframe = pd.concat(
        history_frames,
        ignore_index=True,
        sort=False,
    )

    if today_history_dataframe.empty:
        logger.info("No unresolved software discrepancies found for today's run.")
        return pd.DataFrame(columns=expected_columns)

    today_history_dataframe = today_history_dataframe.drop_duplicates(
        subset=[entity_column, "system_type", "change_type"],
        keep="first",
    )

    today_history_dataframe["etl_date"] = today

    final_dataframe = today_history_dataframe.loc[:, expected_columns]

    logger.info(f"Built {len(final_dataframe)} software audit history row(s).")

    return final_dataframe

def load_audit_report_with_overwrite(
    engine: Engine,
    audit_records: pd.DataFrame,
    table_name: str,
    schema_name: str,
    entity_column: str
) -> pd.DataFrame:
    """
    Loads today's audit report records into HVIE with idempotent overwrite behavior(truncate and load).

    Parameters:
        engine: Active SQLAlchemy database engine.
        audit_records: The audit records to load into table.
        table_name: Name of table used to store hist audit records.
        schema_name: Name of schema used to store hist audit records.

    Returns:
        pd.Dataframe: the audit report records loaded into table
    """
    logger.info("Loading audit history with idempotent overwrite logic.")

    if not entity_column or str(entity_column).strip() == "":
        raise ValueError("entity_column is required.")

    quoted_history_table = f"{schema_name}.{table_name}"

    expected_columns = [
        entity_column,
        "system_type",
        "change_type",
        "consecutive_weeks",
        "etl_date",
    ]

    if audit_records is None:
        audit_records = pd.DataFrame(columns=expected_columns)

    missing_columns = [
        column for column in expected_columns
        if column not in audit_records.columns
    ]

    if missing_columns:
        raise ValueError(
            f"Input dataframe is missing required column(s): {missing_columns}"
        )

    audit_records = audit_records.loc[:, expected_columns].copy()

    with engine.begin() as conn:
        existing_count = conn.execute(
            text(
                f"""
                SELECT COUNT(*) 
                FROM {quoted_history_table}
                WHERE etl_date = CURRENT_DATE
                """
            )
        ).scalar()

        existing_count = int(existing_count or 0)

        if existing_count > 0:
            logger.warning(f"Data already loaded for '{quoted_history_table}' where etl_date = '{audit_records['etl_date']}'. Executing overwrite.")

            conn.execute(
                text(
                    f"""
                    DELETE FROM {quoted_history_table}
                    WHERE etl_date = CURRENT_DATE
                    """
                )
            )
        else:
            logger.info(f"No existing data found for '{quoted_history_table}' where etl_date = '{audit_records['etl_date']}'. Proceeding with clean load.")

        if audit_records.empty:
            logger.info("No audit history rows to insert.")
            return audit_records

        audit_records.to_sql(
            name=table_name,
            con=conn,
            schema=schema_name,
            if_exists="append",
            index=False,
            method="multi",
            chunksize=1000,
        )

    logger.info(
        f"Completed loading audit records into '{quoted_history_table}'. Inserted {len(audit_records)} row(s)."
    )
    return audit_records

def main() -> None:
    logger.info("Starting CRMA Hive software map audit workflow.")

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive","crmahivedb_fqt")
    date_str = datetime.now().strftime("%Y%m%d") 
    report_file_name = f"{FILE_NAME}_{date_str}.xlsx"

    # build report on discrepencies between os refrence map and source systems.
    reports = fetch_all_reports(
        engine=engine,
        report_queries=REPORT_QUERIES,
        query_params=None,
    )
    utility.log_memory_usage()

    is_in_sync = evaluate_sync_status(
        count_dataframe=reports["sync_summary"], 
        health_check=HEALTH_CHECK_COLUMNS,
        exclude_columns=EXCLUDED_HEALTH_CHECK_COLUMNS
    )

    reports["sync_summary"] = normalize_sync_summary(
        count_dataframe=reports["sync_summary"], 
        health_check=HEALTH_CHECK_COLUMNS
    )

    # transforms all reports into single dataframe
    os_history_dataframe = build_audit_report_dataframe(
        reports=reports,
        source_mappings=OS_HISTORY_SOURCE_MAPPINGS,
        entity_column="map_id"
    )
    utility.log_memory_usage()

    # loads dataframe into hist table.
    audit_records = load_audit_report_with_overwrite(
        engine=engine,
        audit_records=os_history_dataframe,
        table_name=TABLE_AUDIT_OS_MAP,
        schema_name=MAIN_SCHEMA,
        entity_column="map_id"
    )
    utility.log_memory_usage()

    # upload loaded audit records to s3
    upload_audit_records_to_s3(
        records=audit_records,
        bucket_name=bucket_name,
        file_name=TABLE_AUDIT_OS_MAP
    )

    # create excel workbook
    excel_stream = create_in_memory_excel(reports)
    raw_excel_bytes = excel_stream.getvalue()
    utility.log_memory_usage()
    
    # upload excel workbook to s3    
    upload_report_to_s3(
        excel_stream=io.BytesIO(raw_excel_bytes),
        bucket_name=bucket_name,
        file_name=FILE_NAME,
    )

    # build email message and attach excel workbook
    email_message = build_email_message(
        excel_stream=io.BytesIO(raw_excel_bytes),
        is_in_sync=is_in_sync,
        attachment_filename=report_file_name,
        email_from=EMAIL_FROM,
        email_to=EMAIL_TO,
        reference_map=REFERENCE_MAP,
        sync_summary_dataframe=reports["sync_summary"]        
    )
    utility.log_memory_usage()

    # send email to HIVE dev team
    send_email_via_relay(
        email_message=email_message,
        smtp_hostname=SMTP_HOST,
        smtp_port=SMTP_PORT
    )

    logger.info(f"Completed CRMA HIVE {REFERENCE_MAP} audit workflow.")

if __name__ == "__main__":
    main()
