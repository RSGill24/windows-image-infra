class NonRetryableHTTPError(Exception):
    pass

def get_with_retries(
        session: requests.Session,
        url: str,
        params: dict,
        timeout: tuple,
        max_retries: int,
        backoff_factor: float,
) -> List[dict]:
    """
    Get request with retries and exponential backoff
    """
    attempt = 0
    retryable_status_codes = (429, 500, 502, 503, 504)  

    while True:
        try:
            resp = session.get(url, params=params, timeout=timeout, verify=False)
            
            if resp.status_code == 200:
                result = resp.json().get("result", [])
                if not isinstance(result, list):
                    raise ValueError("Unexpected response format: 'result' is not a list")
                return result 
            
            if resp.status_code not in retryable_status_codes:
                logger.error(f"Non-retryable status code {resp.status_code} received: {resp.text}")
                raise NonRetryableHTTPError(f"Non-retryable status code: {resp.status_code}")
                
            resp.raise_for_status()  
            
        except ValueError as ve: 
            logger.error(f"Rerutned invalid response format: {ve}")
            raise

        except NonRetryableHTTPError as nre: 
            logger.error(f"Non-retryable HTTP error occurred: {nre}")
            raise
        
        except Exception as e:
            attempt +=1
            if attempt > max_retries:
                logger.error(f"Max retries exceeded: {e}")
                raise

            sleep = min(backoff_factor * (2 ** attempt), 60)  # Cap backoff at 60 seconds
            logger.warning(f"Retry {attempt} for URL {url} with params {params} after error: {e}, sleeping for {sleep} seconds")
            time.sleep(sleep)

def fetch_cmdb_data(
        user: str, 
        password: str, 
        url: str ,
        table_name: str,
        fields: Optional[List[str]]= None,
        query: Optional[str] = None,
    ) -> pd.DataFrame:
    """
    Fetches data from a SNOW table.

    Parameters:
        user (str): API username
        password (str): API user password
        url (str): Table url
        table_name (str): Table name
        fields(str): Fields to fetch
        query(str): Query to filter records records to fetch
    Returns:
        Pandas dataframe of the data retrieved from the table
    """
    # Request params
    page_size = 1500
    timeout = (10, 240)
    max_retries = 5
    backoff_factor = 1.5

    session = requests.Session()
    session.auth = HTTPBasicAuth(user, password)
    session.headers.update({
        "Content-Type": "application/json", 
        "Accept": "application/json",
        "Connection": "keep-alive",
    })

    params = {
        "sysparm_limit": page_size,
        "sysparm_offset": 0,
        "sysparm_display_value": "true",
        "sysparm_exclude_reference_link": "true",
    }

    all_records = []
    total_fetched = 0
    logger.info(f"Fetching data from SNOW Tables API: {url}")

    while True:

        records = get_with_retries(session, url,params, timeout, max_retries, backoff_factor)

        if not records:
            logger.info(f"Completed fetching records for {table_name}. Total records fetched: {total_fetched}")
            break
        
        all_records.extend(records)
        now_fetched = len(records)
        total_fetched += now_fetched
        params["sysparm_offset"] += now_fetched
        logger.info(f"Total records fetched so far: {total_fetched}")
    
    df = pd.DataFrame(all_records)
    df = df.drop_duplicates(subset="sys_id", keep="first")

    return df
