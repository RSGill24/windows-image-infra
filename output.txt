class NonRetryableHTTPError(Exception):
    pass

def get_with_retries(
        session: requests.Session,
        url: str,
        params: dict,
        timeout: tuple,
        max_retries: int,
        backoff_factor: float,
) -> List[dict]:
    """
    Get request with retries and exponential backoff
    """
    attempt = 0
    retryable_status_codes = (429, 500, 502, 503, 504)  

    while True:
        try:
            resp = session.get(url, params=params, timeout=timeout, verify=False)
            
            if resp.status_code == 200:
                result = resp.json().get("result", [])
                if not isinstance(result, list):
                    raise ValueError("Unexpected response format: 'result' is not a list")
                return result 
            
            if resp.status_code not in retryable_status_codes:
                logger.error(f"Non-retryable status code {resp.status_code} received: {resp.text}")
                raise NonRetryableHTTPError(f"Non-retryable status code: {resp.status_code}")
                
            resp.raise_for_status()  
            
        except ValueError as ve: 
            logger.error(f"Rerutned invalid response format: {ve}")
            raise

        except NonRetryableHTTPError as nre: 
            logger.error(f"Non-retryable HTTP error occurred: {nre}")
            raise
        
        except Exception as e:
            attempt +=1
            if attempt > max_retries:
                logger.error(f"Max retries exceeded: {e}")
                raise

            sleep = min(backoff_factor * (2 ** attempt), 60)  # Cap backoff at 60 seconds
            logger.warning(f"Retry {attempt} for URL {url} with params {params} after error: {e}, sleeping for {sleep} seconds")
            time.sleep(sleep)

def fetch_snow_data(
        user: str, 
        password: str, 
        url: str ,
        table_name: str,
        fields: Optional[List[str]]= None,
        query: Optional[str] = None,
    ) -> pd.DataFrame:
    """
    Fetches data from a SNOW table.

    Parameters:
        user (str): API username
        password (str): API user password
        url (str): Table url
        table_name (str): Table name
        fields(str): Fields to fetch
        query(str): Query to filter records records to fetch
    Returns:
        Pandas dataframe of the data retrieved from the table
    """
    # Request params
    page_size = 1500
    timeout = (10, 60)
    max_retries = 5
    backoff_factor = 1.5

    session = requests.Session()
    session.auth = HTTPBasicAuth(user, password)
    session.headers.update({
        "Content-Type": "application/json", 
        "Accept": "application/json",
        "Connection": "keep-alive",
    })

    params = {
        "sysparm_limit": page_size,
        "sysparm_offset": 0,
        "sysparm_display_value": "false",
        "sysparm_exclude_reference_link": "true",
        "sysparm_orderby": "sys_updated_on, sys_id",
    }
    if fields:
        params["sysparm_fields"] = ",".join(fields)
    if query:
        params["sysparm_query"] = query

    all_records = []
    total_fetched = 0
    logger.info(f"Fetching data from SNOW Tables API: {url}")

    while True:

        records = get_with_retries(session, url,params, timeout, max_retries, backoff_factor)

        if not records:
            logger.info(f"Completed fetching records for {table_name}. Total records fetched: {total_fetched}")
            break

        all_records.extend(records)
        now_fetched = len(records)
        total_fetched += now_fetched
        params["sysparm_offset"] += now_fetched
        logger.debug(f"Total records fetched so far: {total_fetched}")
    
    df = pd.DataFrame(all_records)
    df = df.drop_duplicates(subset="sys_id", keep="first")

    return df
        
def fetch_associations(session, base_url, batch_keys, query_field, sysparm_fields, table_name) -> list:
    """
    Fetch enrichment data from SNOW Table API for a batch of keys with parameterized field names.

    Parameters:
        session (requests.Session): Authenticated requests session for persistent connections.
        base_url (str): Base URL of the SNOW instance and Table API endpoint
        batch_keys (list): List of key values to query in SNOW.
        query_field(str): Name of the field to be used in sysparm_query
        sysparm_fields (list): List of fields to retreive in the response
        table_name (str): SNOW table to query

    Returns:
        records: List of dicts representing the enriched records from API response.
    """

    logging.debug(f"Fetching {len(batch_keys)} records on field '{query_field}'")

    # construct OR query string for multiple key_field values(eg., 'sys_id=123^ORsys_id=345')
    query = "^OR".join([f"{query_field}={key}" for key in batch_keys])

    # build url
    url = f"{base_url}/{table_name}"

    # request params
    page_size= 20000 
    timeout = (10, 60)
    max_retries = 5
    backoff_factor = 1.5

    # query param
    params = {
        "sysparm_limit": page_size,
        "sysparm_query": query,
        "sysparm_display_value": "false",
        "sysparm_exclude_reference_link": "true",
    }

    if sysparm_fields:
        params["sysparm_fields"] = ",".join(sysparm_fields)

    records = get_with_retries(session, url, params, timeout, max_retries, backoff_factor)
    logger.debug(f"Succesfully fetched {len(records)} records for this batch")
    return records

def fetch_associations_in_bulk(df, user, password, base_url, key_field, query_field, batch_size, sysparam_fields, table_name) -> pd.DataFrame:
    """
    Enrich the oringial Dataframe by fetching enrichment data in batches with flexiblel field names.

    Parameters:
        df (pandas.Dataframe): Original Dataframe containing column matching key_field
        user (str): username for basic authentication
        password (str): password for basic authentication
        base_url (str): URL for SNOW Table API
        key_field (str): Name of the primary key field in the dataframe to be used as batch ids
        query_field(str): Name of the field to be used in sysparm_query
        batch_size (int): Number of keys to send per batch query.
        sysparm_fields(list): List of fields to request from the API for enrichment
        table_name (str): SNOW table to query

    Returns:
        aggregated_df: pandas dataframe containing fields returned from SNOW API.
    """

    associations= []
    total_batches = (len(df) + batch_size - 1) // batch_size
    logger.info(f"Fetching associations in {total_batches} batches with batch size {batch_size} for {table_name}")

    # create persistent session for all batch requests
    session = requests.Session()
    session.auth = HTTPBasicAuth(user, password)
    session.headers.update({
        "Content-Type": "application/json", 
        "Accept": "application/json",
        "Connection": "keep-alive"
    })

    # process keys in batches
    for i in range(0, len(df), batch_size):
        batch_num = (i // batch_size) + 1
        batch_keys = df[key_field].iloc[i:i+batch_size].tolist()
        logger.debug(f"Fetch batch {batch_num}/{total_batches}: Keys {batch_keys}")
        batch_result = fetch_associations(session, base_url, batch_keys, query_field, sysparam_fields, table_name)
        associations.extend(batch_result)

    # create df from associations
    associations_df = pd.DataFrame(associations)
    
    if not associations_df.empty:
        # aggregate by concatening all column values as comma sperated strings 
        logger.info(f"Aggregating associations on {query_field}")
        aggregated_df = associations_df.groupby(query_field, as_index=False).agg(lambda x: ", ".join(map(str,x)))
    else:
        aggregated_df = associations_df

    logger.info(f"Completed fetching all associations in {table_name}: Orginal rows = {len(df)}, associations = {len(aggregated_df)}")

    return aggregated_df


def fetch_snow_data_router( user: str,  password: str,  bucket_name: str, base_url: str , table_name: str, fields: Optional[List[str]]= None, query: Optional[str] = None) -> pd.DataFrame:
    """
    Routes requests to fetch_associations_in_bulk or fetch_snow_data

    Parameters:
        user (str): username for basic authentication
        password (str): password for basic authentication
        bucket_name: Name of the s3 bucket 
        base_url (str): URL for SNOW Table API
        key_field (str): Name of the primary key field in the dataframe to be used as batch ids
        query_field(str): Name of the field to be used in sysparm_query
        batch_size (int): Number of keys to send per batch query.
        sysparm_fields(list): List of fields to request from the API for enrichment
        table_name (str): SNOW table to query

    Returns:
        Pandas dataframe containing fields returned from SNOW API.
    """
    if table_name == "cmdb_ci":
        # fetch CHANGE_REQUEST_TO_CONFIGURATION_ITEM_ASSC file form s3
        prefix = "raw_data/SNOW/"
        s3_filename = prefix + "CRMA_HIVE_SNOW_" + "M2M_CHANGE_REQUEST_TO_CONFIGURATION_ITEM_ASSC"
        logger.info(f"Fetching data from S3: {prefix}{s3_filename}")
        file = rw_util.fetch_file_from_s3(bucket_name, prefix, s3_filename, "PARQUET")

        # flatten ci_item (aggregate column) and drop duplicates
        logger.info(f"{s3_filename} contains {len(file)} records before normalization process.")
        file["ci_item"] = file["ci_item"].str.split(',')
        file = file.explode("ci_item").reset_index(drop=True)
        file = file.drop_duplicates(subset=["ci_item"])
        logger.info(f"{s3_filename} contains {len(file)} records after normalization process.")

        # return CI's associated with a change request
        return fetch_associations_in_bulk(file, user, password, base_url, "ci_item", "sys_id", 100, fields, table_name)

    else:
        table_url = f"{base_url}/{table_name}"
        return fetch_snow_data(user, password, table_url, table_name, fields, query)

def run_backfill_mode(bucket_name: str, hist_table_name: str, backfill_date: str, datasets: list) -> None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    # check valid backfill date format
    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise
    
    for dataset in datasets:
        dataset_upper = dataset.upper().strip()

        # call processors with backfill_date
        if dataset_upper == "SNOW_POAM_DATA":
            logger.info("Processing POAM_DATA (in backfill mode)")
            process_poam_data(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "SNOW_POAM_AUDIT":
            logger.info("Processing POAM_AUDIT (in backfill mode)")
            process_poam_audit(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "SNOW_FISMADETAILS":
            logger.info("Processing FISMADetails (in backfill mode)")
            process_fisma_details(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "SNOW_PROGRAM":
            logger.info("Processing PROGRAM (in backfill mode)")
            process_program(bucket_name, backfill_date)            
            utility.log_memory_usage() 

        if dataset_upper == "SNOW_SYS_USER":
            logger.info("Processing SYS_USER (in backfill mode)")
            process_sys_user(bucket_name, backfill_date) 
            utility.log_memory_usage()            

        if dataset_upper == "SNOW_REQUEST":
            logger.info("Processing REQUEST (in backfill mode)")
            process_request(bucket_name, backfill_date)
            utility.log_memory_usage()              

        if dataset_upper == "SNOW_WEAKNESS_TRACKING":
            logger.info("Processing WEAKNESS_TRACKING (in backfill mode)")
            process_weakness_tracking(bucket_name, backfill_date)            
            utility.log_memory_usage()  

        if dataset_upper == "SNOW_ISVM":
            logger.info("Processing ISVM (in backfill mode)")
            process_isvm(bucket_name, backfill_date)            
            utility.log_memory_usage() 

        if dataset_upper == "SNOW_CHANGE_REQUEST":
            logger.info("Processing CHANGE_REQUEST (in backfill mode)")
            process_change_request(bucket_name, backfill_date)            
            utility.log_memory_usage() 

        if dataset_upper == "SNOW_CONFIGURATION_ITEM":
            logger.info("Processing CONFIGURATION_ITEM (in backfill mode)")
            process_configuration_item(bucket_name, backfill_date)            
            utility.log_memory_usage() 

def main():
    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]

    if not bucket_name:
       raise ValueError("S3 bucket not found!")

    # resolve workflow run properties
    workflow_params = utility.get_workflow_props()
    backfill_date = workflow_params.get("--backfill_date")
    hist_table_name = workflow_params.get("--hist_table_name")
    datasets = utility.parse_datasets_list( workflow_params.get("--datasets",""))

    # run in backfill mode if all backfill properties are present
    backfill_mode = bool(backfill_date and hist_table_name)


    if backfill_mode:
        run_backfill_mode(bucket_name, hist_table_name, backfill_date, datasets)
    else:
        logger.info(f"Running in Normal mode.")

        logger.info("Processing POAM_DATA")
        process_poam_data(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing POAM_AUDIT")
        process_poam_audit(bucket_name)
        utility.log_memory_usage()

        logger.info("Processing FISMADetails")
        process_fisma_details(bucket_name)
        utility.log_memory_usage()    

        logger.info("Processing PROGRAM")
        process_program(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing SYS_USER")
        process_sys_user(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing REQUEST")
        process_request(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing WEAKNESS_TRACKING")
        process_weakness_tracking(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing ISVM")
        process_isvm(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing CHANGE_REQUEST")
        process_change_request(bucket_name)
        utility.log_memory_usage() 

        logger.info("Processing CONFIGURATION_ITEM")
        process_configuration_item(bucket_name)
        utility.log_memory_usage() 

    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')

if __name__ == "__main__":

def build_load_config(schema: str, datasets: list[str], backfill_mode: bool =False) -> dict[str, dict]:
    """
    Dynamically generate load configuration payload based on input dataset
    
    The load configuration determines: tables to proccess, views to refresh, and steps or actions to execute for each group of tables
    If step includes enforcing FK constraint then add constraint mappings and compliant step to filter out orphan records before loading records.

    Parameters:
        datasets: list of table names to process
        backfill_mode: Boolean flag. When false, forces all sections to be returned. When True, uses condiditonal logic to generate payload for backfill workflow.

    Return:
        load_config: dictionary containing dynamically generated load configuration
    """
    logger.info(f"Buliding LOAD_CONFIG for dataset: {datasets}, backfill_mode: {backfill_mode}")

    # add CRMA_HIVE prefix to all tables in datasets
    if backfill_mode:
        datasets = [f"CRMA_HIVE_{table}" for table in datasets]

    # enforce valid tables in datasets list
    valid_datasets = [t for t in datasets if t in VALID_TABLES]

    if len(valid_datasets) != len(datasets):
        invalid_tables = set(datasets) - set(valid_datasets)
        logger.warning(f"Invalid tables ignored: {invalid_tables}. Valid tables used: {valid_datasets}")
        datasets = valid_datasets
    else:
        logger.info(f"All {len(datasets)} tables in dataset are valid")

    if not datasets:
        logger.warning("No SNOW tables found!")
        return {}

    dataset_set = set(datasets)

    # check if fisma group should be completed
    has_fisma_tables = bool(dataset_set & FISMA_GROUP)
    if has_fisma_tables:
        logger.info("FISMA tables detected. Adding all FISMA group tables to QUARANTINE/WARNING")
        quarantine_warning_tables = list(dataset_set | FISMA_GROUP)
    
    # dynamically build load_config 
    load_config = {}

    # CLEAN_1 group: add group if dataset contains any of the three tables in FISMA group. Always add group if not running in backfill mode.
    if not backfill_mode or has_fisma_tables:
        if backfill_mode:
            logger.info(f"Include CLEAN_1 section (FISMA group detected)")
        load_config["CLEAN_1"] = {
            "prefix": "processed/SNOW/CLEAN/",
            "schema": schema,
            "tables": list(FISMA_GROUP),
            "steps": [STEP_COMPLIANT_TABLES, STEP_TRUNCATE_LOAD_TABLES],
            "refresh_mvs": [],
            "constraints": {
                TABLE_CRMA_HIVE_SNOW_POAM_DATA: {
                    "child_table": TABLE_CRMA_HIVE_SNOW_POAM_DATA,
                    "parent_table": TABLE_CRMA_HIVE_SNOW_FISMADETAILS,
                    "child_fk_column": "fisma_name_id",
                    "parent_pk_column": "sys_id"

                },
                TABLE_CRMA_HIVE_SNOW_POAM_AUDIT: {
                    "child_table": TABLE_CRMA_HIVE_SNOW_POAM_AUDIT,
                    "parent_table": TABLE_CRMA_HIVE_SNOW_POAM_DATA,
                    "child_fk_column": "csam_poam_data_id",
                    "parent_pk_column": "sys_id"
                }
            },
        }
    else:
        logger.info("CLEAN_1 section excluded (no FISMA tables detected)")

    # CLEAN_2 group: add group if dataset contains table not in FISMA group. Always add group if not running in backfill mode.
    has_other_tables = bool(dataset_set - FISMA_GROUP)
    if not backfill_mode or has_other_tables:
        other_tables = list(dataset_set - FISMA_GROUP)
        if backfill_mode:
            logger.info(f"Include CLEAN_2 section with {len(other_tables)} tables: {list(other_tables)}")
        load_config["CLEAN_2"] = {
                "prefix": "processed/SNOW/CLEAN/",
                "schema": schema,
                "tables": other_tables,
                "steps": [STEP_TRUNCATE_LOAD_TABLES],
                "refresh_mvs": []
        }
    else:
        logger.info("Exclude CLEAN_2 section (no other tables detected)")    

    # QUARANTINE group: add all tables in datasets with quarantine suffix
    logger.info(f"Include QUARANTINE section with {len(datasets)} tables")
    quarantine_tables = [f"{t}{QUARANTINE_SUFFIX}" for t in quarantine_warning_tables]
    load_config["QUARANTINE"] = {
        "prefix": "processed/SNOW/QUARANTINE/",
        "schema": schema,
        "tables": quarantine_tables,
        "steps": [STEP_TRUNCATE_LOAD_TABLES]
    }

    # WARNING group: add all tables in datasets with warning suffix
    logger.info(f"Include WARNING section with {len(datasets)} tables")
    warning_tables = [f"{t}{WARNING_SUFFIX}" for t in quarantine_warning_tables]
    load_config["WARNING"] = {
        "prefix": "processed/SNOW/WARNING/",
        "schema": schema,
        "tables": warning_tables,
        "steps": [STEP_TRUNCATE_LOAD_TABLES]
    }    

    logger.info(f"LOAD_CONFIG generated with {len(load_config)} sections")

    return load_config

def main():

    schema = MAIN_SCHEMA
    # set bucket and db client
    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    if not bucket_name:
        ValueError("S3 bucket not found!")
    engine = connection.postgres_engine("edae-crma-hive")
    if not engine:
        ValueError("SQLAlchemy database connection manager not found!")    
    db_client = CrmaHiveSnowETLManager(engine)

    ##Parallel loading##
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
    if not engine2:
        ValueError("SQLAlchemy database connection manager not found!")    
    db_client2 = CrmaHiveSnowETLManager(engine2)

    # resolve workflow run properties
    workflow_params = utility.get_workflow_props()
    backfill_date = workflow_params.get("--backfill_date")
    hist_table_name = workflow_params.get("--hist_table_name")
    datasets = utility.parse_datasets_list( workflow_params.get("--datasets",""))

    # run in backfill mode if all backfill properties are present
    backfill_mode = bool(backfill_date and hist_table_name)
    
    if backfill_mode:
        schema = BACKFILL_SCHEMA
        logger.info(f"Running in BACKFILL mode.")
        logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")
        # check valid backfill date format
        if not utility.is_valid_backfill_date(backfill_date):
            logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
            raise
    else:
        logger.info(f"Running in Normal mode.")
        datasets = [ TABLE_CRMA_HIVE_SNOW_FISMADETAILS,
        TABLE_CRMA_HIVE_SNOW_POAM_DATA,
        TABLE_CRMA_HIVE_SNOW_POAM_AUDIT,
        TABLE_CRMA_HIVE_SNOW_PROGRAM,
        TABLE_CRMA_HIVE_SNOW_SYS_USER,
        TABLE_CRMA_HIVE_SNOW_REQUEST,
        TABLE_CRMA_HIVE_SNOW_WEAKNESS_TRACKING,
        TABLE_CRMA_HIVE_SNOW_ISVM,
        TABLE_CRMA_HIVE_SNOW_CHANGE_REQUEST,
        TABLE_CRMA_HIVE_SNOW_CONFIGURATION_ITEM]

    # dynamically generate db load configuration
    LOAD_CONFIG = build_load_config(schema, datasets, backfill_mode)

    # Iterate through each group in LOAD_CONFIG and run etl steps: CLEAN -> QUARANTINE -> WARNING
    try:
        for group_name, group_info in LOAD_CONFIG.items():
            prefix = group_info["prefix"]
            schema = group_info["schema"]
            tables = group_info["tables"]
            steps = group_info["steps"]
            constraints = group_info.get("constraints",{})
            refresh_mvs = group_info.get("refresh_mvs",[])
            
            logger.info(f"Started processsing {group_name} files.")

            # fetch files from S3 for each table in group and capture record count metric
            dataframes = {}
            for table in tables:
                base_name = f"{prefix}{table}"
                if backfill_mode:
                    df = rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")
                    df["etl_date"] = datetime.strptime(backfill_date, "%m%d%y").strftime("%Y-%m-%d")
                else:
                    df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")
                    df["etl_date"] = datetime.now().strftime("%Y-%m-%d")
                record_count = len(df)
                logger.info(f"Fetched {record_count} rows for {group_name} {table} from S3.")
                table = table.upper()
                dataframes[table] = df
                db_client.add_file_metrics(table, record_count)

                ##Parallel loading##
                db_client2.add_file_metrics(table, record_count) 

            # Run ETL steps to process and load the tables
            if tables:
                db_client.run_steps(dataframes,schema,tables,steps,constraints,refresh_mvs, group_name)
                logger.info(db_client.get_metrics())

                ##Parallel loading##
                db_client2.run_steps(dataframes,schema,tables,steps,constraints,refresh_mvs, group_name)


            else:
                logger.warning(f"No tables found in group: {group_name}")

    finally:
        db_client.close() 
        #Parallel loading##
        db_client2.close()

    metrics = db_client.get_metrics()

    # run load validaiton checks - if we have orhphan records then we will want to adjust the count for the table being validated 
    for table, file_count in metrics["files"].items():
        base_table_count = metrics["main"].get(table) if metrics["main"].get(table) else 0
        if "QUARANTINE" in table:
            # if validation run on quarantine table then subtract orphan records from quarantine table
            constraint_violation_count =  metrics["constraint_violations"].get(f"{table}") if metrics["constraint_violations"].get(f"{table}") else 0
            adjusted_base_table_count = base_table_count - constraint_violation_count
        else:
            # if validation run on non-quarantine table then add orphan records to table
            constraint_violation_count =  metrics["constraint_violations"].get(f"{table}{QUARANTINE_SUFFIX}") if metrics["constraint_violations"].get(f"{table}{QUARANTINE_SUFFIX}") else 0
            adjusted_base_table_count = base_table_count + constraint_violation_count
        if adjusted_base_table_count == file_count:
            logger.info(f"Post load validation check passed for {table}.")
        else:
            logger.error(f"Post load validation check failed for {table}! File contains {file_count} records but {adjusted_base_table_count} records loaded into base tables!")
            metrics["hasETLError"] = True
    
    if metrics["hasETLError"]:
        raise RuntimeError("Load script failed one or more processing steps! Check logs for details!")

    logger.info(f"Succesfully loaded all files!")
if __name__ == "__main__":
    main()
