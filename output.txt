def load_to_db_and_perform_post_load_validation(
        engine: Engine,
        df: pd.DataFrame,
        table: str, 
) -> None:
    """
    Loads data to desttination database
    
    Parameters:
        engine (create_engine): Database connection engine
        df (pd.DataFrame):
        schema (str):
        table (str):
        load_control_table (str):
    """
    logger.info(f"Loading table '{table}' to '{MAIN_SCHEMA}' schema")

    rw_util.bulk_truncate_and_load_with_copy(
        engine=engine,
        df=df,
        schema=MAIN_SCHEMA,
        target_table=table
    )
    logger.info(f"Perfomming post load validation on table '{table}' in '{MAIN_SCHEMA}' schema")
    if rw_util.post_load_validation(
            engine=engine,
            df=df,
            schema_name=MAIN_SCHEMA,
            table_name=table
        ): 
        logger.info(f"Post load validation check passed for table '{table}' in '{MAIN_SCHEMA}' schema")

        ##Parallel loading##
        logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
        logger.info(f"Loading table '{table}' to '{MAIN_SCHEMA}' schema")     
        engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
        rw_util.bulk_truncate_and_load_with_copy(
            engine=engine2,
            df=df,
            schema=MAIN_SCHEMA,
            target_table=table
         )
         
    else:
        raise RuntimeError(f"Post load validation check failed for {table} in '{MAIN_SCHEMA}' schema")

def build_cpe_parse_query() -> str:
    """
    Build query to parse softeare manafucturer, product and version from tvm assets installed_software field
    """
    sql = f"""
        WITH 
        installed_software AS (
            SELECT DISTINCT
                id AS asset_id,
                split_part(cpe, ':', 3) AS manufacturer,
                split_part(cpe, ':', 4) AS product,
                split_part(cpe, ':', 5) AS version,
                timestamps_last_seen AS last_seen
            FROM (
                SELECT id, timestamps_last_seen, unnest(installed_software) AS cpe
                FROM crma_hive.crma_hive_tvm_assets   
            ) AS software
        )

        SELECT 
            asset_id, 
            manufacturer, 
            product, 
            version,
            last_seen 
        FROM installed_software;             
    """
    return sql

def execute_workflow(
        engine: Engine,
        bucket_name: str,
    ) -> None:
    """
    Run the TVM  software normalization workflow.

    - Query CPE/software data from the source database.
    - Normalize software name and version fields.
    - De-duplicate, add ETL date, and load into target table.
    - Archive the normalized dataset to S3.
    - Derive unique software names and upsert into target tvm unique softwares
    table and archive to S3.

    Args:
        engine: SQLAlchemy Engine for Postgres database.
        bucket_name: S3 bucket used for archival copies.
    """

    logger.info("Build query.")
    sql = build_cpe_parse_query()

    logger.info("Excecute query.")
    df= rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql
    )

    df = df.replace(['', 'NA', 'na', '#N/A', '#NA', 'N/A', 'null', 'Null', 'nan', 'None', 'none' ], np.nan)

    logger.info("Normalize softeware_name")
    df['software_name'] = df.apply(
        lambda row: dataQualityUtils.normalize_software_name(row['manufacturer'], row['product']), axis=1
    )

    logger.info("Normalize version")
    df['software_version'] = df['version'].apply(dataQualityUtils.normalize_version)
    df = df.drop(columns=['version'])
    df = df.drop_duplicates()
    df["etl_date"] = datetime.now().date() 

    logger.info("Load to database")
    table="crma_hive_tvm_normalized_softwares"
    rw_util.bulk_truncate_and_load_with_copy(
            engine=engine,
            df=df,
            schema=MAIN_SCHEMA,
            target_table=table,
        )
    
    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    logger.info(f"Loading table '{table}' to '{MAIN_SCHEMA}' schema")     
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
    rw_util.bulk_truncate_and_load_with_copy(
        engine=engine2,
        df=df,
        schema=MAIN_SCHEMA,
        target_table=table
        )

    logger.info(f"Total records loaded to DB : {len(df)}")

    logger.info("Save a copy to S3 for archival")
    archive_to_s3(
        df=df,
        bucket_name=bucket_name,
        dataset=table
    )

    logger.info("Identify unique TVM Softwares")
    unique_softwares_df = df[["software_name", "etl_date"]].drop_duplicates().reset_index(drop=True)

    logger.info("Update TVM unique softwares table")
    table="crma_hive_tvm_unique_softwares"
    rw_util.bulk_upsert_with_copy(
        engine=engine,
        df=unique_softwares_df,
        schema=MAIN_SCHEMA,
        target_table=table,
        pk_columns=["software_name"]
    )

    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    logger.info(f"Loading table '{table}' to '{MAIN_SCHEMA}' schema")     
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")  
    logger.info(f"Total records loaded to DB : {len(df)}")
    rw_util.bulk_upsert_with_copy(
        engine=engine2,
        df=unique_softwares_df,
        schema=MAIN_SCHEMA,
        target_table=table,
        pk_columns=["software_name"]
    )

    logger.info(f"Total records loaded to DB : {len(unique_softwares_df)}")

    logger.info("Save a copy to S3 for archival")
    archive_to_s3(
        df=unique_softwares_df,
        bucket_name=bucket_name,
        dataset=table
    )

        

def main():

    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 

    execute_workflow(engine=engine, bucket_name=bucket_name)

      
    utility.log_memory_usage() 
    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")

    main()

def process_and_load_tvm_to_mobius_software_map(bucket_name: str, engine: Engine ) -> None:
    """
    Load the TVM-to-Mobius software mapping from S3 into Postgres.

    Steps:
      - Read the XLSX mapping file from the landing zone in S3.
      - Select relevant mapping columns and standardize null-like values.
      - Archive a processed copy to S3 in Parquet format.
      - Add ETL date and load the data into the target table in the primary DB.
      - Load the same data into the corresponding table in the secondary DB.

    Args:
        bucket_name: Name of the S3 bucket containing input and archive data.
        engine: SQLAlchemy Engine connected to the primary Postgres database.
    """
    logger.info(f"Fetching data from landing zone...")
    
    # Creating variables and the dataframe
    dataset = "CRMA_HIVE_TVM_TO_MOBIUS_SOFTWARE_MAP"
    prefix = "raw_data/SWAM/"
    base_name = prefix + dataset

    df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "CSV")  

    logger.info(f"Total records fetched: {len(df)}")

    target_columns = [
        "tvm_software_name", "mobius_software_name", "mobius_product_id"]
    logger.info("Selecting target columns...")
    
    df = dataQualityUtils.select_fields(df, target_columns)

    df = df.replace(['', 'NA', 'na', '#N/A', '#NA', 'N/A', 'null', 'Null', 'nan' ], np.nan)


    logger.info("Storing a copy in S3 bucket for archival...")
    filename =f"processed/SWAM/{dataset}" 
    rw_util.load_to_s3(bucket_name, df, filename, "PARQUET")

    logger.info("Loading data to database...")
    df["etl_date"] = datetime.now().date() 
    schema="crma_hive"
    table_name = f"{dataset.lower()}"
    rw_util.truncate_and_load_table(
           engine=engine, 
           df=df, schema_name=schema, 
           table_name=table_name, 
           exist_type="append"            
    )

    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    logger.info(f"Loading table '{table_name}' to '{schema}' schema")     
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
    rw_util.truncate_and_load_table(
           engine=engine2, 
           df=df, schema_name=schema, 
           table_name=table_name, 
           exist_type="append"            
    )
 

def main():
    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]

def build_select_query() -> str:
    """
    Build query to parse softeare manafucturer, product and version from tvm assets installed_software field
    """
    sql = f"""
        SELECT 
            hardware_software, 
            manufacturer, 
            product, 
            version, 
            version_label,
            status, 
            component,
            status_date
            manufacturer_id,
            product_id,
            version_id,
            governance_id,
            fisma_system_id,
            CONCAT_WS('_',
                COALESCE(manufacturer_id::text, '0'),
                COALESCE(product_id::text, '0'),
                COALESCE(version_id::text, '0'),
                COALESCE(governance_id::text, '0'),
                COALESCE(fisma_system_id::text, '0')				  
            ) AS mobius_id,
            CONCAT_WS('_',
                COALESCE(manufacturer_id::text, '0'),
                COALESCE(product_id::text, '0')		  
            ) AS mobius_product_id                       
        FROM crma_hive.crma_hive_mobius_trm_workbook
        WHERE hardware_software = 'Software'
            AND component IN ('ICE', 'MGMT')            
    """
    return sql
