def build_fisma_scan_compliance_score_sql() -> str:
    """
    Build the Scan Compliance Score per device data preparation SQL Command using CTEs.
    """

    sql = f"""
        WITH 
        -- Step 1: Select Eligible devices
        deduped_devices AS (
            -- Step 1.1: dedup
            SELECT *
            FROM (
                SELECT *,
                    ROW_NUMBER() OVER (
                        PARTITION BY asset_id
                        ORDER BY device_last_seen DESC
                    ) AS asset_rn
                FROM crma_hive.crma_hive_fismascorecard_input_devices 
                WHERE fisma_id iLIKE 'ICE-%%'
                    AND asset_id IS NOT NULL
            ) ranked
            WHERE asset_rn = 1 
            UNION ALL
            SELECT *,
                NULL AS asset_rn
            FROM crma_hive.crma_hive_fismascorecard_input_devices 
            WHERE asset_id IS NULL
        ),
        eligible_devices AS (
            -- Step 1.2: apply eligibility filters
            SELECT 
                device_id,
                fisma_id,
                devicerole,
                device_last_seen,
                asset_id,
                is_credentialed,
                last_authenticated_scan_date
            FROM  deduped_devices
            WHERE devicerole IN ('ENDPOINT','NETWORKING DEVICE')
        ),
        -- Step 2: Erich device data with FISMA details
        fisma_enriched_eligible_devices AS(
            SELECT 
                d.device_id,
                d.fisma_id,
                f.acronym,
                f.fisma_name,
                f.security_support_portfolio AS portfolio,
                f.dhs_hva,
                f.public_facing,
                f.cfo_designated,
                f.fips_199_security_categorization_overall,
                f.active,
                f.iscm_waiver, 
                f.fisma_reportable,
                f.fisma_inventory_selc_status,
                f.fisma_category,
                d.devicerole,
                d.device_last_seen,
                d.asset_id,
                d.is_credentialed,
                d.last_authenticated_scan_date
            FROM  eligible_devices d
            LEFT JOIN crma_hive.crma_hive_snow_fismadetails f
            ON d.fisma_id = f.fisma_id 
        ),
        -- Step 3: compute FISMA system scores
        fisma_scan_compliance_score AS (
            SELECT
                fisma_id,
                MAX(acronym) AS acronym,
                MAX(fisma_name) AS fisma_name,
                MAX(portfolio) AS portfolio,
                MAX(dhs_hva) AS dhs_hva,
                MAX(public_facing) AS public_facing,
                MAX(cfo_designated) AS cfo_designated ,
                MAX(fips_199_security_categorization_overall) AS fips_199_security_categorization_overall,
                BOOL_OR(active) AS active,
                MAX(iscm_waiver) AS iscm_waiver, 
                MAX(fisma_reportable) AS fisma_reportable,
                MAX(fisma_inventory_selc_status) AS fisma_inventory_selc_status,
                MAX(fisma_category) AS fisma_category,
                COUNT(*) AS total_assets,
                COUNT(*) FILTER (WHERE is_credentialed) AS total_credentialed_assets,
                CAST(
                    (COUNT(*) FILTER (WHERE is_credentialed)::float / COUNT(*)::float) * 100
                    AS INT
                ) AS scan_compliance_score
            FROM fisma_enriched_eligible_devices
            GROUP BY fisma_id
        )
        SELECT 
            fisma_id,
            acronym,
            fisma_name,
            portfolio,
            dhs_hva,
            public_facing,
            cfo_designated ,
            fips_199_security_categorization_overall,
            active,
            iscm_waiver, 
            fisma_reportable,
            fisma_inventory_selc_status,
            fisma_category,
            total_credentialed_assets,
            total_assets,	
            scan_compliance_score
        FROM fisma_scan_compliance_score

    """

    return sql

def build_component_scan_compliance_score_sql() -> str:
    """
    Build the Scan Compliance Score for component(ICE) System data preparation SQL Command using CTEs. 
    """
    sql = f"""
        WITH 
        -- Step 1: Select Eligible devices
        deduped_devices AS (
            -- Step 1.1: dedup
            SELECT *
            FROM (
                SELECT *,
                    ROW_NUMBER() OVER (
                        PARTITION BY asset_id
                        ORDER BY device_last_seen DESC
                    ) AS asset_rn
                FROM crma_hive.crma_hive_fismascorecard_input_devices 
                WHERE asset_id IS NOT NULL
            ) ranked
            WHERE asset_rn = 1 
            UNION ALL
            SELECT *,
                NULL AS asset_rn
            FROM crma_hive.crma_hive_fismascorecard_input_devices 
            WHERE asset_id IS NULL
        ),
        eligible_devices AS (
            -- Step 1.2: apply eligibility filters
            SELECT 
                device_id,
                fisma_id,
                devicerole,
                device_last_seen,
                asset_id,
                is_credentialed,
                last_authenticated_scan_date
            FROM  deduped_devices
            WHERE devicerole IN ('ENDPOINT','NETWORKING DEVICE')
        ),
        -- Step 2: Erich device data with FISMA details and filter for elibible fisma systems
        fisma_enriched_eligible_devices AS(
            SELECT 
                d.device_id,
                d.fisma_id,
                f.acronym,
                f.fisma_name,
                f.security_support_portfolio AS portfolio,
                f.dhs_hva,
                f.public_facing,
                f.cfo_designated,
                f.fips_199_security_categorization_overall,
                f.active,
                f.iscm_waiver, 
                f.fisma_reportable,
                f.fisma_inventory_selc_status,
                f.fisma_category,
                d.devicerole,
                d.device_last_seen,
                d.asset_id,
                d.is_credentialed,
                d.last_authenticated_scan_date
            FROM  eligible_devices d
            LEFT JOIN crma_hive.crma_hive_snow_fismadetails f
            ON d.fisma_id = f.fisma_id 
            WHERE fisma_reportable iLike 'yes'
                AND fisma_category in ('MAJ', 'GSS')
                AND fisma_inventory_selc_status iLike 'Operational'
                AND COALESCE(iscm_waiver, '') not ILIKE 'yes' 
        ),
        component_scan_compliance_score AS (
            SELECT
                'ICE' AS component,
                COUNT(*) AS total_assets,
                COUNT(*) FILTER (WHERE is_credentialed) AS total_credentialed_assets,
                CASE
                    WHEN COUNT(*) = 0 THEN NULL
                    ELSE CAST(
                        (COUNT(*) FILTER (WHERE is_credentialed)::float / COUNT(*)::float ) * 100
                        AS INT )
                END AS scan_compliance_score
            FROM fisma_enriched_eligible_devices
        )
        SELECT 
            component,
            total_credentialed_assets,
            total_assets,	
            scan_compliance_score
        FROM component_scan_compliance_score
        """
    return sql

def run_scan_compliance_score_workflow(
        engine: Engine,
        bucket_name: str
    )->None:
    """
    Execute the Scan Compliance score workflow, including building query, 
    param preparation, execute query, store output.

    Parameters:
        engine (Engine): Database engine for data access.
        bucket_name (str): Name of the storage bucket for output.

    Returns:
        None
    """
    etl_date= date.today()
    logger.info(f"Computing Scorecard SC Score for : {etl_date}")

    logger.info("Build FISMA Systems Scan Compliance score query.")
    sql = build_fisma_scan_compliance_score_sql()


    logger.info("Excecute FISMA Systems Scan Compliace score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
    )
    
    if df is not None:
        logger.info(f"Per FISMA System Scan compliance score received. Total records: {len(df)}")         
        logger.info(f"Load fisma system score dataset to target table.")
        df["etl_date"] = etl_date
        dataset = "crma_hive_fismascorecard_fisma_system_scan_compliance_score"
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )

    else:
        logger.warning("Query returned no data")   
        return 

    logger.info("Build Component Scan Compliance score query.")
    sql = build_component_scan_compliance_score_sql()

    logger.info("Excecute Componenet Scan Compliance score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
    )

    if df is not None:       
        logger.info(f"Load Component Scan Compliance score dataset to target table and save a copy to S3 for archival.")
        df["etl_date"] = etl_date 
        dataset="crma_hive_fismascorecard_component_scan_compliance_score"
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )
       
    else:
        logger.warning("Query returned no data")      
         

def main():

    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 


    run_scan_compliance_score_workflow(engine=engine, bucket_name=bucket_name)       
      
    utility.log_memory_usage() 
    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")


if __name__ == "__main__":
    main()
