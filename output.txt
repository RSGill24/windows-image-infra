from typing import Optional

import pandas as pd
import urllib3
from datetime import datetime
import logging
import time
from crma_hive_utilities import Utility, Read_Write_Utility
from crma_hive_tvm_utilities_spark import SparkTvmUtility
from crma_hive_connections import Connection
from sqlalchemy.engine import Engine
import re

from pyspark.sql import SparkSession


# Turning off warnings
urllib3.disable_warnings()

# Setting up logger
logger = logging.getLogger()
logger.setLevel(logging.INFO)

# Instantiate classes
utility = Utility()
rw_util = Read_Write_Utility()
connection = Connection()
sparkTvmUtil = SparkTvmUtility()


# Define global vars
MAIN_SCHEMA = rw_util.MAIN_SCHEMA
BACKFILL_SCHEMA = rw_util.BACKFILL_SCHEMA
TABLE_CRMA_HIVE_TVM_VULNS= "TVM_VULNS" 

VULNS_HIST_TABLE_NAME = "crma_hive_tvm_vulns_hist"
METADATA_TABLE = "crma_hive_load_control_metadata"
  
def run_normal_mode(
    engine: Engine, 
    spark: SparkSession,
    jdbc_url: str,
    user: str,
    password: str, 
    bucket_name: str, 
    schema: str
) -> None:
    """
    Prepares input parameters and calls a utility function to load TVM historical tables.

    Parameters:
        engine (str): SQLalchemy engine
        schema (str): The database schema in which data is loaded.
        bucket_name (str): Name of S3 bucket containing raw and processed integration files 
    Returns:
        None
    """
    logger.info("Running in NORMAL mode.")

    logger.info(f"Preparing to load TVM {TABLE_CRMA_HIVE_TVM_VULNS} to {VULNS_HIST_TABLE_NAME}")
    sparkTvmUtil.load_tvm_data_to_db_hist_tables(
        engine=engine, 
        spark=spark,
        jdbc_url=jdbc_url,
        user=user,
        password=password, 
        bucket_name=bucket_name, 
        dataset=TABLE_CRMA_HIVE_TVM_VULNS,
        target_table=VULNS_HIST_TABLE_NAME,
        load_control_table=METADATA_TABLE,
        schema=schema,
        )

    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    logger.info(f"Loading table '{VULNS_HIST_TABLE_NAME}' to '{schema}' schema")     
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")  
    jdbc_url2, user2, password2 = connection.parse_pg_uri_to_jdbc("edae-crma-hive", "crmahivedb_fqt")
    sparkTvmUtil.load_tvm_data_to_db_hist_tables(
        engine=engine2, 
        spark=spark,
        jdbc_url=jdbc_url2,
        user=user2,
        password=password2,
        bucket_name=bucket_name, 
        dataset=TABLE_CRMA_HIVE_TVM_VULNS,
        target_table=VULNS_HIST_TABLE_NAME,
        load_control_table=METADATA_TABLE,
        schema=schema,
        )

    utility.log_memory_usage() 

def run_backfill_mode(
    engine: Engine,
    spark: SparkSession,
    jdbc_url: str,
    user: str,
    password: str, 
    schema: str, 
    bucket_name: str, 
    hist_table_name: str, 
    backfill_date: str
)->None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        engine (str): SQLalchemy engine
        schema (str): The database schema in which data is loaded.
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
   
    Returns:
        None
    """
    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}")

    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if hist_table_name.lower() == VULNS_HIST_TABLE_NAME.lower():
        logger.info(f"Preparing to load TVM {TABLE_CRMA_HIVE_TVM_VULNS} to {hist_table_name}")
        sparkTvmUtil.load_tvm_data_to_db_hist_tables(
            engine=engine, 
            jdbc_url=jdbc_url,
            user=user,
            password=password, 
            bucket_name=bucket_name, 
            dataset=TABLE_CRMA_HIVE_TVM_VULNS,
            target_table=VULNS_HIST_TABLE_NAME,
            load_control_table=METADATA_TABLE,
            schema=schema,
            backfill_date=backfill_date,
            )
        utility.log_memory_usage() 

def main():
    start_time = time.time()

    engine = connection.postgres_engine("edae-crma-hive")
    jdbc_url, user, password = connection.parse_pg_uri_to_jdbc("edae-crma-hive")

    try: 
        bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]  
    except Exception as e:
       logger.error(f"Error retrieving bucket name:{e}")   
       raise     
    try: 
        spark= SparkSession.builder.getOrCreate()
    except Exception as e:
       logger.error(f"Error creating spark session:{e}")   
       raise  


    logger.info("Extracting workflow run protperies.")
    workflow_params = utility.get_workflow_props()
    backfill_date = workflow_params.get("--backfill_date")
    hist_table_name = workflow_params.get("--hist_table_name") 

    backfill_mode = bool(backfill_date and hist_table_name)

    if backfill_mode:
        run_backfill_mode(
            engine=engine, 
            spark=spark,
            jdbc_url=jdbc_url,
            user=user,
            password=password, 
            schema=BACKFILL_SCHEMA, 
            bucket_name=bucket_name, 
            hist_table_name=hist_table_name, 
            backfill_date=backfill_date
        )   
    
    else:
        run_normal_mode(
            engine=engine, 
            spark=spark,
            jdbc_url=jdbc_url,
            user=user,
            password=password, 
            bucket_name=bucket_name, 
            schema=MAIN_SCHEMA
        )


    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')


if __name__ == "__main__":
    main()
