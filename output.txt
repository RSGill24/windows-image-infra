def build_device_swam_score_sql() -> str:
    """
    Build the SWAM Score per device data preparation SQL Command using CTEs.
    """

    sql = f"""
        WITH
        -- Step 1: Join tvm normalized softwares to mobius reference map on softwar name
                -- LEFT JOIN normalized_tvm_software to tvm_to_mobius_software map ON software_name
        partial_match  AS(
            SELECT DISTINCT 
                ts.asset_id,
                ts.last_seen AS assets_last_seen,
                map.tvm_software_name,
                ts.software_version AS tvm_software_version,
                map.mobius_software_name,
                map.mobius_product_id
            FROM crma_hive.crma_hive_tvm_normalized_softwares ts
            LEFT JOIN crma_hive.crma_hive_tvm_to_mobius_software_map map
                ON ts.software_name=map.tvm_software_name
            WHERE ts.last_seen >= %(lookback_date)s 
        ),
        -- Step 2: Check if tvm-mobius  software name  match also match on version 
                -- LEFT JOIN partial_match to normalized_mobius map ON software_name and version(or version_label)
                -- x in version is wildcard character and should match any number. 
        full_match AS(
            SELECT DISTINCT 
                pm.asset_id,
                pm.tvm_software_name,
                pm.tvm_software_version,
                pm.assets_last_seen,
                ms.software_name AS mobius_software_name,
                ms.normalized_version AS mobius_software_version,
                ms.normalized_version_label AS mobius_version_label,
                ms.status
            FROM partial_match pm
            LEFT JOIN crma_hive.crma_hive_mobius_normalized_softwares ms
                ON pm.mobius_product_id=ms.mobius_product_id
                AND(
                    pm.tvm_software_version = ms.normalized_version
                        OR pm.tvm_software_version = ms.normalized_version_label
                        OR ms.normalized_version ~ ('^' || regexp_replace(pm.tvm_software_version, 'x', '[0-9]+', 'g') || '$')
                        OR ms.normalized_version_label ~ ('^' || regexp_replace(pm.tvm_software_version, 'x', '[0-9]+', 'g') || '$')
                        -- escape parentheses in normalized_version field before running regex_replace			
                        OR pm.tvm_software_version ~ ('^' || regexp_replace(
                                                                regexp_replace( 
                                                                    regexp_replace(ms.normalized_version, '\(', '\\(', 'g'), 
                                                                    '\)', '\\)', 'g'
                                                                ),
                                                                'x', '[0-9]+', 'g'
                                                            ) || '$')
                        OR pm.tvm_software_version ~ ('^' || regexp_replace(
                                                                regexp_replace( 
                                                                    regexp_replace(ms.normalized_version_label, '\(', '\\(', 'g'), 
                                                                    '\)', '\\)', 'g'
                                                                ),
                                                                'x', '[0-9]+', 'g'
                                                            ) || '$')                        
                    )
        ),
        -- Step 3: Join full_match with axonius devices data to identify eligible devices
        deduped_devices AS(	
                -- Step 3.1: Dedup asset_id 
                    -- group by asset_id -> rank by device_last_seen, keep top ranked
            SELECT *
            FROM (
                SELECT *,
                    ROW_NUMBER() OVER (
                        PARTITION BY asset_id
                        ORDER BY device_last_seen DESC
                    ) AS asset_rn
                FROM crma_hive.crma_hive_fismascorecard_input_devices
                WHERE fisma_id iLIKE 'ICE-%%'
                    AND asset_id IS NOT NULL
            ) ranked
            WHERE asset_rn = 1 
            UNION ALL
            SELECT *,
                NULL AS asset_rn
            FROM crma_hive.crma_hive_fismascorecard_input_devices 
            WHERE asset_id IS NULL
        ),
        eligible_device_softwares AS (
                -- 3.2: Join
            SELECT  DISTINCT 
                dev.device_id,
                preferred_hostname AS hostname, 
                preferred_ipv4  AS ip,
                dev.fisma_id,
                dev.devicerole,
                dev.device_last_seen,
                dev.asset_id,
                dev.is_credentialed,
                dev.last_authenticated_scan_date,
                sw.tvm_software_name AS software,
                sw.tvm_software_version AS version,
                sw.status
            FROM deduped_devices dev
            LEFT JOIN full_match sw  
                ON dev.asset_id = sw.asset_id
            WHERE dev.devicerole = 'ENDPOINT'
            AND dev.is_credentialed IS TRUE
        ),
        -- Step 4: Standardaize software status field to prep for scoring
        remove_exclude_list_sofwares AS (
            SELECT *
            FROM eligible_device_softwares
            WHERE LOWER(software) NOT IN (
                'microsoft internet explorer',
                'microsoft visioviewer',
                'microsoft office compatibility pack',
                'microsoft visio viewer'
        )
        ),
        -- Step 5: Standardaize software status field to prep for scoring
        standardaized_software_status AS(
            SELECT DISTINCT 
                device_id,
                hostname,
                ip,
                fisma_id,
                device_last_seen,
                asset_id,
                is_credentialed,
                last_authenticated_scan_date,
                software,
                version,		
                CASE
                    WHEN status ILIKE 'Prohibited' OR status ILIKE 'Not Authorized' THEN 'Prohibited'
                    ELSE NULL
                END AS sw_status
            FROM remove_exclude_list_sofwares 
        ),
        -- Step 6: Per device aggregates 
        per_device_aggs AS (
            SELECT
                device_id,
                MAX(hostname) AS hostname,
                MAX(ip)  As ip,
                MAX(fisma_id) AS fisma_id,
                asset_id,
                MAX(device_last_seen) AS device_last_seen,
                BOOL_OR(is_credentialed) AS is_credentialed,
                MAX(last_authenticated_scan_date) AS last_authenticated_scan_date,       
                COUNT(*) AS total_sw_count,
                SUM(CASE WHEN sw_status = 'Prohibited' THEN 1 ELSE 0 END) AS prohibited_sw_count,
                STRING_AGG(
                    DISTINCT CONCAT(software, ' ', COALESCE(version, '')),
                    ', '
                ) FILTER (WHERE sw_status = 'Prohibited') AS prohibited_softwares_list
            FROM standardaized_software_status
            GROUP BY device_id, asset_id
        ),
        -- Step 7: Compute device SWAM scores  
        swam_scores AS(
            SELECT 
                device_id,
                hostname,
                ip,
                fisma_id,
                asset_id,
                device_last_seen,
                is_credentialed,
                last_authenticated_scan_date,       
                total_sw_count,
                prohibited_sw_count,
                prohibited_softwares_list, 
                CASE 
                    WHEN prohibited_sw_count = 0 THEN 1   -- pass
                    ELSE 0   --fail
                END AS swam_score

            FROM per_device_aggs

        ),
        -- Step 8: Join enrich device scores with FISMA details 
        device_swam_scores_enriched AS (
            SELECT 
                sw.device_id,
                sw.hostname,
                sw.ip,
                sw.device_last_seen,
                sw.fisma_id,
                fd.fisma_name,
                fd.security_support_portfolio AS portfolio,
                fd.dhs_hva,
                fd.public_facing,
                fd.cfo_designated,
                fd.fips_199_security_categorization_overall,
                fd.active,
                fd.iscm_waiver, 
                fd.fisma_reportable,
                fd.fisma_inventory_selc_status,
                fd.fisma_category,
                sw.asset_id,
                sw.last_authenticated_scan_date,
                sw.is_credentialed,
                sw.prohibited_sw_count,
                sw.prohibited_softwares_list,
                sw.total_sw_count,
                sw.swam_score    
            FROM swam_scores sw
            LEFT JOIN crma_hive.crma_hive_snow_fismadetails fd
                ON sw.fisma_id = fd.fisma_id
        )
        SELECT 
            device_id,
            hostname,
            ip,
            fisma_id,
            fisma_name,
            portfolio,
            dhs_hva,
            public_facing,
            cfo_designated,
            fips_199_security_categorization_overall,
            active,
            iscm_waiver, 
            fisma_reportable,
            fisma_inventory_selc_status,
            fisma_category,
            device_last_seen,
            asset_id,
            is_credentialed,
            last_authenticated_scan_date,
            prohibited_sw_count,
            prohibited_softwares_list,
            total_sw_count,
            swam_score
        FROM device_swam_scores_enriched

    """

    return sql

def build_fisma_swam_score_sql() -> str:
    """
    Build the SWAM Score per FISMA System data preparation SQL Command using CTEs.
    """
    sql = f"""
        WITH 
        fisma_swam_score AS (
            SELECT
                fisma_id,
                CAST(SUM(CASE WHEN swam_score = 1 THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS INT) AS swam_score
            FROM crma_hive.crma_hive_fismascorecard_device_swam_score
            WHERE etl_date = %(etl_date)s
            GROUP BY fisma_id
        )
        SELECT 
            a.fisma_id,
            a.fisma_name,
            a.security_support_portfolio AS portfolio,
            a.dhs_hva,
            a.public_facing,
            a.cfo_designated,
            a.fips_199_security_categorization_overall,
            a.active,
            a.iscm_waiver,
            a.fisma_reportable,
            a.fisma_inventory_selc_status,
            a.fisma_category,
            b.swam_score
        FROM crma_hive.crma_hive_snow_fismadetails a
        LEFT JOIN fisma_swam_score b
            ON a.fisma_id = b.fisma_id
    """
    return sql

def run_swam_score_workflow(
        engine: Engine,
        bucket_name: str
    )->None:
    """
    Execute the Software Assets Management(SWAM) score workflow, including building query, 
    param preparation, execute query, store output.

    Parameters:
        engine (Engine): Database engine for data access.
        bucket_name (str): Name of the storage bucket for output.

    Returns:
        None
    """
    etl_date= date.today()
    logger.info(f"Computing Scorecard SWAM Score for : {etl_date}") 
    params = fismaScorecardUtility.compute_date_params(etl_date)
    params.pop("cve_kev_cutoff_date", None)
    logger.info("Lookback date: %s", params["lookback_date"])

    logger.info("Build Device SWAM score query.")
    sql = build_device_swam_score_sql()

    logger.info("Excecute Device SWAM score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
        params=params
    )
    
    if df is not None:
        logger.info(f"Per device SWAM score received. Total number of devices: {len(df)}")         
        logger.info(f"Load device score dataset to target table.")
        df["etl_date"] = etl_date
        dataset = "crma_hive_fismascorecard_device_swam_score"
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )

    else:
        logger.warning("Query returned no data")   
        return 

    logger.info("Build FISMA System SWAM score query.")
    sql = build_fisma_swam_score_sql()

    logger.info("Excecute FISMA System SWaM score query.")
    df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=sql,
        params={"etl_date": etl_date}
    )

    if df is not None:       
        logger.info(f"Load FISMA System SWAM score dataset to target table and save a copy to S3 for archival.")
        df["etl_date"] = etl_date 
        dataset='crma_hive_fismascorecard_fisma_system_swam_score'
        fismaScorecardUtility.load_to_db_and_perform_post_load_validation(
            engine=engine,
            df=df,
            table=dataset,
            etl_date=etl_date
        )

        fismaScorecardUtility.archive_to_s3(
            df=df,
            bucket_name=bucket_name,
            dataset=dataset
        )
       
    else:
        logger.warning("Query returned no data")      
         

def main():

    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]
    engine = connection.postgres_engine("edae-crma-hive") 


    run_swam_score_workflow(engine=engine, bucket_name=bucket_name)       
      
    utility.log_memory_usage() 
    elapsed_minutes = (time.time() - start_time) / 60 
    logger.info(f"Complete. Total elapsed time : {elapsed_minutes}")


if __name__ == "__main__":
    main()
