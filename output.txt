import pandas as pd
import numpy as np
import re
import math
import logging
from datetime import datetime, timedelta, timezone
from typing import Optional, Tuple, Any, List, Union
from pandas import Series

logger = logging.getLogger(__name__)

logger.setLevel(logging.INFO)

class DataAggregator:

    def __init__(self):
        self.logger = logger

    def _is_array_column(self, series: pd.Series) -> bool:
        """
        Detect if column contains arrays/lists
        """
        if series.empty or series.isna().all():
            return False
        sample = series.dropna().iloc[0] 
        return isinstance(sample, (list, np.ndarray)) or (sample is not None and hasattr(sample, '__iter__') and not isinstance(sample, str))

    def _parse_wildcard_pattern(self, pattern: str) -> tuple[bool,bool,str]:
        """
        Convert wildcard pattern to flags and clean pattern
        """
        is_prefix = pattern.endswith('*')
        is_suffix = pattern.startswith('*')
        clean_pattern = pattern.strip('*')
        return is_prefix, is_suffix, clean_pattern
    
    def _scalar_wildcard_filter(self, series: Series, pattern: Union[str, List[Any]]) -> Series:
        """
        Filter scalar column with wildcard support (eg,. state = '*open*') as well as multi-value support (eg., state = 'open' | state = 'reopened')
        """
        if isinstance(pattern,list):
            masks = []
            for p in pattern:
                p_str = str(p) if p is not None else ''

                # handle booleans
                if series.dtype == 'bool':
                    if p_str.lower() in ('true', '1', 'yes'):
                        masks.append(series == True)
                    elif p_str.lower() in ('false', '0', 'no'):
                        masks.append(series == True)
                    else:
                        masks.append(pd.Series([False] * len(series), index=series.index))
                # handle numerics
                elif pd.api.types.is_numeric_dtype(series):
                    try:
                        p_num = pd.to_numeric(p)
                        masks.append(series == p_num)
                    except:
                       masks.append(pd.Series([False] * len(series), index=series.index))
                # string wildcard 
                else:
                    prefix_wild, suffix_wild, clean_pattern = self._parse_wildcard_pattern(p_str)
                    series_str = series.astype(str).str.lower()
                    clean_pattern_lower = clean_pattern.lower()

                    if prefix_wild and suffix_wild:
                        mask = series_str.str.contains(clean_pattern_lower, na=False, regex=False)
                    elif prefix_wild:
                        mask = series_str.str.fullmatch(f'^{re.escape(clean_pattern_lower)}.*', na=False)
                    elif suffix_wild:
                        mask = series_str.str.fullmatch( f'.*{re.escape(clean_pattern_lower)}$', na=False)
                    else:
                        mask = series_str == p_str.lower()     
                    masks.append(mask.fillna(False).astype(bool))

            # OR all masks togheter
            combined_mask = pd.Series(np.logical_or.reduce(masks), index=series.index, dtype=bool)
            return combined_mask
        else:
                        
            # convert pattern to string
            pattern_str = str(pattern) if pattern is not None else ''

            # handle booleans explicitly
            if series.dtype == 'bool':
                if pattern_str.lower() in ('true', '1', 'yes'):
                    return pd.Series(series == True, index=series.index, dtype=bool) #series == True
                elif pattern_str.lower() in ('false', '0', 'no'):
                    return pd.Series(series == False, index=series.index, dtype=bool) #return series == False
                else:
                    return pd.Series([False] * len(series), index=series.index, dtype=bool)
            
            # handle numeric columns
            if pd.api.types.is_numeric_dtype(series):
                try:
                    pattern_num = pd.to_numeric(pattern)
                    return pd.Series(series == pattern_num, index=series.index, dtype=bool)
                except:
                    return pd.Series([False] * len(series), index=series.index)
                
            # string columns with wildcard support
            prefix_wild, suffix_wild, clean_pattern = self._parse_wildcard_pattern(pattern_str)
            series_str = series.astype(str).str.lower()
            clean_pattern_lower = clean_pattern.lower()

            if prefix_wild and suffix_wild:
                mask = series_str.str.contains(clean_pattern_lower, na=False, regex=False)
            
            elif prefix_wild:
                mask = series_str.str.fullmatch(f'^{re.escape(clean_pattern_lower)}.*', na=False)
            elif suffix_wild:
                mask = series_str.str.fullmatch( f'.*{re.escape(clean_pattern_lower)}$', na=False)
            else:
                mask = series_str == pattern_str.lower()
            
            return pd.Series(mask.fillna(False).astype(bool), index=series.index)

    def _array_wildcard_filter(self, series: Series, pattern: Union[str, List[Any]]) -> Series:
        """
        Filter array column with wildcard support (eg,. state && ['*open*']) as well as multi-value support (eg., state && ['open*', 'reopned'])
        """
        pattern_list = [pattern] if not isinstance(pattern, list) else pattern

        

        # def _check_array_match(arr: Any, pattern_str: str) -> bool:
        def _check_array_match_any_values(arr: Any) -> bool:
            # check if valid array
            if arr is None or not isinstance(arr, (list, np.ndarray)) or len(arr) == 0:
                return False
            # convert ndarray -> list
            arr_list = list(arr) if isinstance(arr, np.ndarray) else arr

            # check if ANY pattern value matches this array
            for pattern_val in pattern_list:
                pattern_str = str(pattern_val) if pattern_val is not None else ''


                prefix_wild, suffix_wild, clean_pattern = self._parse_wildcard_pattern(pattern_str)
                clean_pattern_lower = clean_pattern.lower()

                for item in arr_list:
                    # handle boolean in arrays
                    if isinstance(item, bool):
                        item_str = 'true' if item else 'false'
                    else:
                        item_str = str(item).lower()

                    if prefix_wild and suffix_wild:
                        if clean_pattern_lower in item_str:
                            return True
                    elif prefix_wild:
                        if re.match(f'^{re.escape(clean_pattern_lower)}.*', item_str, re.IGNORECASE):   
                            return True
                    elif suffix_wild:
                        if re.match(f'.*{re.escape(clean_pattern_lower)}$', item_str, re.IGNORECASE):
                            return True
                    else:
                        if item_str == pattern_str.lower():
                            return True
            return False
                
        result = series.apply(_check_array_match_any_values)
        return pd.Series(result.fillna(False).astype(bool), index=series.index)
    
    def filter_dataframe(self, df: pd.DataFrame, filter_conditions: List[Tuple[str,Union[str, List[Any]]]], dedup_on: Optional[str] = None) -> pd.DataFrame:
        """
        Apply multiple filter conditions, supports filters on array, and supports using wildcard expresions

        Parameters:
            df: Input dataframe to filter and count
            filter_conditions: List of {column: value} filters
        
        Returns:
            pd.DataFrame: Returns all records matching filter conditions
        """
        logger.info(f"Applying {len(filter_conditions)} filter conditions to {len(df)} records")
        filtered = df.copy()
        for col, val in filter_conditions:
            if col not in filtered.columns:
                logger.error(f"Column '{col}' not found")
                raise KeyError("Missing column")
            
            val_display = str(val[:10] + ['...'] if isinstance(val, list) and len(val) > 3 else val)
            logger.info(f"Filtering {len(filtered)} records on {col}='{val_display}'")
            if self._is_array_column(filtered[col]):
                logger.info(f"Detected array column '{col}'")
                mask = self._array_wildcard_filter(filtered[col], val)
            else:
                mask = self._scalar_wildcard_filter(filtered[col], val)
            prev_count = len(filtered)
            filtered = filtered[mask]
            retention = len(filtered) / prev_count * 100 if prev_count > 0 else 0
            logger.info(f"Result: {len(filtered)} records returned ({retention:.1f}%)")
        
        # Deduplicate dataframe on dedup id
        if dedup_on is not None:
            if dedup_on not in filtered.columns:
                logger.error(f"Dedup column '{dedup_on}' not found")
                raise KeyError("Missing dedup column")
            else:
                logger.info(f"Deduplicating on '{dedup_on}' at the end")
                filtered = filtered.drop_duplicates(subset=[dedup_on])
        
        logger.info(f"Final filtered dataframe: {len(filtered)} records")
        return filtered
    
    def avg_time_days(self, df: pd.DataFrame, time_col: str = "time_taken_to_fix", conversion_factor: int = 86400) -> int:
        """
        Compute average time with optional filtering

        Parameteres:
            df: Input dataframe to filter and average
            filter_conditions: Optional List of {column: value} filters
            time_col: Time/duration column name 
            conversion_factor: Conversion factor (86400=sec->days, 24=hours->days, 1440=min->days)

        Returns:
            int: Average time in target units (0 if no data)
        """
        logger.info(f"Computing average for '{time_col}' column for {len(df)} records (units: {conversion_factor})")
        valid_times = df[time_col].dropna()
        logger.info(f"Found {len(valid_times)} valid {time_col} values")

        if len(valid_times) > 0:
            avg_smallest_units = valid_times.mean()
            avg_units = math.ceil(avg_smallest_units / conversion_factor)

            logger.info(f"Average time: {avg_units} days ({avg_smallest_units:.0f})")
            return avg_units
        else:
            logger.warning(f"No valid time values found, returning 0")
            return None
