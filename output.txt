def fetch_non_exportable_datasets(base_url, headers, dataset, bucket_name, output_filename):

    url = f"{base_url}/{dataset}"
    logger.info(f"Fetching TVM {dataset} from {url}")
    
    try:
        response = requests.get(url, headers=headers, timeout=720)
        status = response.status_code
        logger.info(f"Status code: {status} ")
        if status == 200:
            json_response = response.json()
            data = json_response.get(dataset, [])                             
            df = pd.json_normalize(data)
            store_data(bucket_name=bucket_name, df=df, output_filename=output_filename)      
        response.raise_for_status()
    except Exception as e:
        logger.error(f"Error fetching data: {e}")
        raise

def process_cisco_vulns(
        bucket_name: str,
        backfill_date: Optional[str] = None
        ) -> None:
    """
    Handles  data cleaning and enrichement of the TVM vulns dataset 
    
    In backfill mode, fetch the file whose name contains the backfill date and 
    overwrite existing file in /processed prefix.

    Parameters:
        bucket_name (str): Name of the s3 bucket
        dataset (str): name of the dataset to be processed
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)

    """
    logger.info(f"Fetching data from landing zone...")
    
    dataset = "CISCO_VULNS"
    prefix = "raw_data/TVM/"
    base_name = prefix + "CRMA_HIVE_TVM_" + dataset

    if backfill_date:
        df = rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")
    else:
        df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")

    if df is None:
        raise ValueError("No data fetched from landing zone")

    logger.info(f"Retrieve rase data, total records: {len(df)}")
    target_columns = [   
        "finding_id" , "severity", "severity_id", "state", "first_found", "last_found", "last_fixed", "indexed", "resurfaced_date", "time_taken_to_fix", "asset_uuid", "asset_operating_system", "asset_hostname", "asset_ipv4", "plugin_id", "plugin_name", "plugin_cvss3_base_score",
        "plugin_cvss3_temporal_score", "plugin_cvss3_temporal_vector", "plugin_epss_score", "plugin_exploit_available", "plugin_exploitability_ease", "plugin_family", "plugin_cve", "plugin_xrefs", "port_port",
        "port_protocol", "scan_uuid", "source",    
    ]
    logger.info("Selecting target columns...")
    df = dataQualityUtils.select_fields(df, target_columns)

    df['is_kev_tenable'] = df['plugin_xrefs'].astype(str).str.contains('CISA-KNOWN-EXPLOITED', case=False, na=False)


    logger.info("Exttract is_kev flag using the CISA KEV Catalog")

    logger.info("1. Get CVE lookup")
    engine = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")
    cve_table =f"{MAIN_SCHEMA}.crma_hive_cisa_kev_catalog" 
    cve_lookup_query = f"""
                        SELECT cve_id
                        FROM {cve_table}
                        """
    cve_lookup_df = rw_util.query_postgres_to_dataframe(
        engine=engine,
        sql_query=cve_lookup_query,
    )

    logger.info("2. Flantten plugin_cve field and clean")
    exploded_cves_df = (
        df[["finding_id", "plugin_cve"]]
        .explode("plugin_cve")
        .rename(columns={"plugin_cve": "cve_id"})
    )
    exploded_cves_df["cve_id"] = (
        exploded_cves_df["cve_id"]
        .astype("string")
        .str.strip()
        .str.upper()
    )
    exploded_cves_df = (
        exploded_cves_df
        .dropna(subset=["cve_id"])
        .reset_index(drop=True)
    )

    logger.info("3. Find rows where there is at least one match in cve lookup")
    matched_df = (
        exploded_cves_df
        .merge(
            cve_lookup_df[["cve_id"]].drop_duplicates(),
            on="cve_id",
            how="inner"
        )[["finding_id"]]
        .drop_duplicates()
        .assign(is_kev=True)
        .reset_index(drop=True)
    )

    logger.info("4. Add is_kev Columns to the Dataframe")
    df = (
        df
        .merge(matched_df, on="finding_id", how="left")
        .assign(is_kev=lambda x: x["is_kev"].fillna(False))
    )

    #Convert DataFrame columns containing lists or numpy arrays to PostgreSQL arrays
    df = dataQualityUtils.convert_arrays_for_postgres(df) 

    #Apply DQ6:Standardize timestamp to ISO 8601 timestamp format 
    cols = ["first_found", "last_found", "last_fixed", "indexed", "resurfaced_date", ]    
    df = dataQualityUtils.clean_timestamp_fields_iso8601_format(df, cols)

    #Apply DQ4: Convert numeric columns to Integer 
    cols = ["severity_id", "plugin_id",  "port_port" ]
    df = dataQualityUtils.clean_int_fields(df, cols)

    #Apply DQ9: Convert columns to float/decimal 
    cols = ["time_taken_to_fix", "plugin_cvss3_base_score", "plugin_cvss3_temporal_score", "plugin_epss_score"]
    df = dataQualityUtils.clean_float_fields(df, cols)

    #Apply DQ3: Normalize boolean fields 
    cols = ["plugin_exploit_available"]
    df = dataQualityUtils.normalize_boolean_fields(df, cols)

    #Apply DQ1 & Q: Not nullabe, Quarantine
    cols = ["finding_id", "state", "first_found","last_found", "indexed", "asset_uuid", "plugin_id", "port_port", "port_protocol", "scan_uuid" ]
    df, quarantine_not_null = dataQualityUtils.apply_not_nullable_q_constraint(df, cols)

    #Apply DQ0 & Q: No duplicates & Quarantine
    cols = [ "finding_id",]
    df, quarantine_dups = dataQualityUtils.apply_no_duplicates_constraint(df, cols)

    #Merge quarantine dataframes
    quarantine_df = pd.concat([quarantine_not_null, quarantine_dups], ignore_index=True)  
 
    #Load to destination 
    logger.info("Loading processed data to destination...")
    tvmUtil.load_processed_tvm_datasets_to_s3_in_parquet(
        bucket_name = bucket_name, 
        dataset = dataset, 
        clean_df=df, 
        quarantine_df=quarantine_df,
        backfill_date=backfill_date,
        )

    # #....**Uncomment this to save in csv for testing** ....

    #     bucket_name = bucket_name, 
    #     dataset = dataset,

def process_assets(
        bucket_name: str,
        backfill_date: Optional[str] = None
        ) -> None:
    """
    Handles  data cleaning and enrichement of the TVM assets dataset 

    In backfill mode, fetch the file whose name contains the backfill date and 
    overwrite existing file in /processed prefix. 

    Parameters:
        bucket_name (str): Name of the s3 bucket
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
    """
    logger.info(f"Fetching data from landing zone...")
    
    # Creating variables and the dataframe
    dataset = "ASSETS"
    prefix = "raw_data/TVM/"
    base_name = prefix + "CRMA_HIVE_TVM_" + dataset

    if backfill_date:
        df = rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")
    else:
        df = rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")

    if df is None:
        raise ValueError("No data fetched from landing zone")

    logger.info(f"Total records: {len(df)}")
    target_columns = [ "id",  "network_hostnames", "network_ipv4s", "network_mac_addresses", "network_fqdns", "network_netbios_names", "network_ssh_fingerprints", "scan_first_scan_time", "scan_last_authenticated_scan_date", "scan_last_authentication_attempt_date",
                      "scan_last_authentication_scan_status", "scan_last_authentication_success_date",	"scan_last_scan_id", "scan_last_scan_time", "timestamps_created_at", "timestamps_first_seen", "timestamps_last_seen",
                      "timestamps_updated_at", "tags", "operating_systems",	"system_types", "installed_software", "agent_uuid", "agent_names", "has_agent", "has_plugin_results",	"is_licensed", "is_public",
    ]
    logger.info("Selecting target columns...")
    df = dataQualityUtils.select_fields(df, target_columns)

    #Extract FISMA ID from `tag` field
    logger.info("Extracting FISMA IDs from asset tags")
    def _extract_fisma_id(s: str):
        """Private helper function to extract FISMA_ID from asset tags"""
        if s is None:
                return None

        s = str(s)
        if not s.strip() or s.strip().lower() == "null":
            return None

        matches = re.findall(
            r"'key'\s*:\s*'AXONIUS FISMA ID'.*?'value'\s*:\s*'([^']+)'",
            s,
            flags=re.DOTALL,
        )

        return matches or None

    df["tags_fisma_id"] = df["tags"].apply(_extract_fisma_id)
    

    #Convert DataFrame columns containing lists or numpy arrays to PostgreSQL arrays
    df = dataQualityUtils.convert_arrays_for_postgres(df)

    #Apply DQ6:Standardize timestamp to ISO 8601 timestamp format 
    cols = ["scan_first_scan_time", "scan_last_authenticated_scan_date",  "scan_last_authentication_attempt_date", "scan_last_authentication_success_date", "scan_last_scan_time", "timestamps_created_at", "timestamps_first_seen", "timestamps_last_seen", "timestamps_updated_at",   ]    
    df = dataQualityUtils.clean_timestamp_fields_iso8601_format(df, cols)

    #Apply DQ3: Normalize boolean fields 
    cols = ["has_agent", "has_plugin_results", "is_licensed", "is_public",  ]
    df = dataQualityUtils.normalize_boolean_fields(df, cols)

    #Apply DQ1 & Q: Not nullabe, Quarantine
    cols = ["id",]
    df, quarantine_not_null = dataQualityUtils.apply_not_nullable_q_constraint(df, cols)

    #Apply DQ0 & Q: No duplicates & Quarantine
    cols = [ "id",]
    df, quarantine_dups = dataQualityUtils.apply_no_duplicates_constraint(df, cols)

    #Merge quarantine dataframes
    quarantine_df = pd.concat([quarantine_not_null, quarantine_dups], ignore_index=True)  
 
    #Load to destination 
    logger.info("Loading processed data to destination...")
    tvmUtil.load_processed_tvm_datasets_to_s3_in_parquet(
        bucket_name = bucket_name, 
        dataset = dataset, 
        clean_df=df, 
        quarantine_df=quarantine_df,
        backfill_date=backfill_date,
        )

    # #....**Uncomment this to save in csv for testing** ....
    # tvmUtil.load_processed_tvm_datasets_to_s3_in_csv(
    #     bucket_name = bucket_name, 
    #     dataset = dataset,

    target_columns = [ "no_target",	"template_uuid", "description", "name", "owner", "visibility", "shared", "user_permissions", "last_modification_date", "creation_date", "owner_id", "id" ]
    logger.info("Selecting target columns...")
    df = dataQualityUtils.select_fields(df, target_columns)

    #Replace '', 'NULL', 'NA' values with null
    df.replace(['', 'NULL', 'NA'], pd.NA, inplace=True)

    #Apply DQ3: Normalize boolean fields 
    cols = ["no_target", ]
    df = dataQualityUtils.normalize_boolean_fields(df, cols)

    # Converting Unix timestamp to ISO8601 format
    cols = ["creation_date", "last_modification_date", ]
    df = dataQualityUtils.convert_unix_timestamp_to_iso8601_format(df, cols)

    #Apply DQ4: Convert numeric columns to Integer 
    cols = ["id", "shared", "user_permissions", "owner_id" ]
    df = dataQualityUtils.clean_int_fields(df, cols)

    #Apply DQ1 & Q: Not nullabe, Quarantine
    cols = ["id",]
    df, quarantine_not_null = dataQualityUtils.apply_not_nullable_q_constraint(df, cols)

    #Apply DQ0 & Q: No duplicates & Quarantine
    cols = [ "id",]
    df, quarantine_dups = dataQualityUtils.apply_no_duplicates_constraint(df, cols)

    #Merge quarantine dataframes
    quarantine_df = pd.concat([quarantine_not_null, quarantine_dups], ignore_index=True)  
 
    #Load to destination 
    logger.info("Loading processed data to destination...")
    tvmUtil.load_processed_tvm_datasets_to_s3_in_parquet(
        bucket_name = bucket_name, 
        dataset = dataset, 
        clean_df=df, 
        quarantine_df=quarantine_df,
        backfill_date=backfill_date,
        )

def run_backfill_mode(
        bucket_name: str,
        hist_table_name: str,
        backfill_date: str,
        datasets: list,
) -> None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """
    logger.info(f"TVM data processing running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise

    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise

    for dataset in datasets:
        dataset_upper = dataset.upper().strip()

        # call processors with backfill_date
        if dataset_upper == "TVM_CISCO_VULNS":
            logger.info("Processing TVM vulnerablities (in backfill mode)")
            process_cisco_vulns(bucket_name, backfill_date)
            utility.log_memory_usage() 

        if dataset_upper == "TVM_PLUGIN_19506_VULNS":
            logger.info("Processing TVM vulnerablities (in backfill mode)")
            process_plugin_19506_vulns(bucket_name, backfill_date)
            utility.log_memory_usage() 
        if dataset_upper == "TVM_ASSETS":
            logger.info("Processing TVM Assets (in backfill mode)")
            process_assets(bucket_name, backfill_date)
            utility.log_memory_usage()

        # if dataset_upper == "TVM_SCANS":
        #     logger.info("Processing TVM scans (in backfill mode)")
        #     process_scans(bucket_name, backfill_date)
        #     utility.log_memory_usage()

        if dataset_upper == "TVM_POLICIES":
            logger.info("Processing TVM policies (in backfill mode)")
            process_policies(bucket_name, backfill_date)
            utility.log_memory_usage()

def main():
    start_time = time.time()
    utility.log_memory_usage() 

    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]

    if not bucket_name:
       raise ValueError("S3 bucket not found!")

            backfill_date=backfill_date, 
            datasets=datasets

    else:
        run_normal_mode(
            bucket_name=bucket_name
        )

    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')

def run_normal_mode(
    engine: Engine,
    bucket_name: str, 
    schema: str,        
)->None:
    """
    Prepares input parametrs and calls utility function to load TVM data to target talbes in Normal mode.

    Parameters:
        engine (str): SQLalchemy engine
        schema (str): The database schema in which data is loaded.
        bucket_name (str): Name of S3 bucket containing raw and processed integration files   
    Returns:
        None
    """    
    logger.info("TVM data to target loading running in NORMAL mode.")
    load_configs= [
            {"dataset": TABLE_CRMA_HIVE_TVM_CISCO_VULNS, "load_strategy": "upsert", "pk_columns": ["finding_id"]},
            {"dataset": TABLE_CRMA_HIVE_TVM_ASSETS, "load_strategy": "upsert", "pk_columns": ["id"] },
            # {"dataset": TABLE_CRMA_HIVE_TVM_SCANS, "load_strategy": "upsert", "pk_columns": ["id"] },
            {"dataset": TABLE_CRMA_HIVE_TVM_POLICIES, "load_strategy": "upsert", "pk_columns": ["id"] },
            {"dataset": TABLE_CRMA_HIVE_TVM_PLUGIN_19506_VULNS, "load_strategy": "truncate_and_load"},

        ]

    for load_config in load_configs:
        tvmUtil.load_tvm_data_to_db(
            engine= engine, 
            bucket_name = bucket_name, 
            load_config=load_config,
            schema=schema
            )
        utility.log_memory_usage() 

    ##Parallel loading##
    logger.info("Load copy to 'crmahivedb_fqt' on 'eda-crmahive-np' ")
    engine2 = connection.postgres_engine("edae-crma-hive", "crmahivedb_fqt")      
    for load_config in load_configs:
         tvmUtil.load_tvm_data_to_db(
            engine= engine2, 
            bucket_name = bucket_name, 
            load_config=load_config,
            schema=schema
            )

def run_backfill_mode(
    engine: Engine,
    schema: str, 
    bucket_name: str, 
    hist_table_name: str, 
    backfill_date: str, 
    datasets: list
)->None:
    """
    Executes backfill processing for explicilty specified datasts relevant to a specific historical table.

    Parameters:
        engine (str): SQLalchemy engine
        schema (str): The database schema in which data is loaded.
        bucket_name (str): Name of S3 bucket containing raw and processed integration files
        hist_table_name (str): Name of Historical table in database (e.g., crma_hive_poam_audit_controls_hist)
        backfill_date (str): Date string used to find which integraiton source file to re-process (e.g., 121525)
        datasets (list): List of datsets upstream to a historical table
    
    Returns:
        None
    """

    logger.info(f"Running in BACKFILL mode.")
    logger.info(f"Workflow Properties: hist_table={hist_table_name}, date={backfill_date}, datasets={datasets}")

    if not utility.is_valid_backfill_date(backfill_date):
        logger.error(f"Invalid backfill date! Expected date format is MMDDYY. Date provided is '{backfill_date}'.")
        raise
    
    if not datasets:
        logger.error(F"No dataset specified for backfill for {hist_table_name}!")
        raise

    valid_datasets = [t for t in datasets if t in VALID_TABLES]

    if len(valid_datasets) != len(datasets):
        invalid_tables = set(datasets) - set(valid_datasets)
        logger.warning(f"Invalid tables ignored: {invalid_tables}. Valid tables used: {valid_datasets}")
        datasets = valid_datasets
    else:
        logger.info(f"All {len(datasets)} tables in dataset are valid")

    if not datasets:
        logger.warning(F"No TVM tables found!")

    load_configs = []
    for dataset in datasets:
        dataset_upper = dataset.upper().strip()

        if dataset_upper == "TVM_CISCO_VULNS":
            load_configs.append({"dataset": TABLE_CRMA_HIVE_TVM_CISCO_VULNS, "load_strategy": "upsert", "pk_columns": ["finding_id"]})
        
        if dataset_upper == "TVM_ASSETS":
            load_configs.append({"dataset": TABLE_CRMA_HIVE_TVM_ASSETS, "load_strategy": "upsert", "pk_columns": ["id"] })

        # if dataset_upper == "TVM_SCANS":
        #     load_configs.append({"dataset": TABLE_CRMA_HIVE_TVM_SCANS, "load_strategy": "upsert", "pk_columns": ["id"] })

        if dataset_upper == "TVM_POLICIES":
            load_configs.append({"dataset": TABLE_CRMA_HIVE_TVM_POLICIES, "load_strategy": "upsert", "pk_columns": ["id"] })

    for load_config in load_configs:
        tvmUtil.load_tvm_data_to_db(
            engine= engine, 
            bucket_name = bucket_name, 
            load_config=load_config,
            schema=schema,
            backfill_date=backfill_date,
            )
        utility.log_memory_usage() 


def main():
    start_time = time.time()

    engine = connection.postgres_engine("edae-crma-hive")
    
    bucket_name = connection.get_secret("edae-crma-hive")["lz_bucket_name"]

    if not bucket_name:
       raise ValueError("S3 bucket not found!")
    
    logger.info("Extracting workflow run protperies.")
    workflow_params = utility.get_workflow_props()
    backfill_date = workflow_params.get("--backfill_date")
    hist_table_name = workflow_params.get("--hist_table_name")
    datasets = utility.parse_datasets_list( workflow_params.get("--datasets",""))
    
    backfill_mode = bool(backfill_date and hist_table_name)
    if backfill_mode:
        run_backfill_mode(
            engine=engine, 
            schema=BACKFILL_SCHEMA, 
            bucket_name=bucket_name, 
            hist_table_name=hist_table_name, 
            backfill_date=backfill_date, 
            datasets=datasets,        
        ) 

    else:
        run_normal_mode(
            engine=engine,
            bucket_name = bucket_name, 
            schema=MAIN_SCHEMA,
        )

    elapsed_minutes = (time.time() - start_time)/60
    logger.info(f'Complete. Total elapsed time : {elapsed_minutes}')


if __name__ == "__main__":
    main()
