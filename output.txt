    def get_or_create_tvm_export_job(
        self,
        base_url: str, 
        headers: dict, 
        payload: dict, 
        dataset: str
    ) -> str:
        """
        Start a new Tenable Vulnerability Management (TVM) export job or retrieve an existing one for the specified dataset.
        Args:
            base_url (str): The base URL of the Tenable VM API.
            headers (dict): The HTTP headers for authentication and content type.
            payload (dict): The request payload specifying export job parameters.
            dataset (str): The dataset type to export (e.g., 'vulns', 'assets').
        Returns:
            str: The job uuid.
        """      
        export_url = f"{base_url}/{dataset}/{'v2/export' if dataset == 'assets' else 'export'}"  #if assets use api v2 else default
        logger.info(f"Export payload: {payload}")   
        logger.info(f"Export URL: {export_url}") 
        resp = requests.post(export_url, headers=headers, json=payload)
        status_code = resp.status_code
        data = resp.json()
        if status_code in (200, 202) and 'export_uuid' in data:
            export_uuid = data['export_uuid']
            logger.info(f"Started new export job: {export_uuid}")
        elif status_code == 429:
            logger.info("Job already exists")
            message = data.get('message', '')
            match = re.search(r'job_uuid=([a-f0-9\-]+)', message)
            if match:
                export_uuid = match.group(1)
                logger.info(f"Existing export job: {export_uuid}")
            else:
                raise Exception(f"429 error: Duplicate export not allowed, and could not extract job_uuid. Response: {data}")
        else:
            raise Exception(f"Failed to start export: {resp.text}")
        
        return export_uuid

    def poll_export_status(
        self,
        base_url: str,
        headers: dict,
        dataset: str,
        export_uuid: str
    ) -> dict:
        """
        Retrieve the status of a specific Tenable VM export job.
        Args:
            base_url (str): The base URL of the Tenable VM API.
            headers (dict): The HTTP headers for authentication and content type.
            dataset (str): The dataset type for the export job (e.g., 'vulns', 'assets').
            export_uuid (str): The UUID of the export job.
        Returns:
            dict: The JSON response from the API containing the export job status.
        """       
        status_url = f"{base_url}/{dataset}/export/{export_uuid}/status"
        while True:
            try:
                resp = requests.get(status_url, headers=headers)
                resp.raise_for_status()
                status = resp.json()
            except RequestException as e:
                logger.error(f"Error checking export job {export_uuid} status: {e}")
                raise
            except ValueError as e:
                logger.error(f"Invalid JSON in status response for export job {export_uuid}: {e}")
                raise            
            if status['status'] == 'PROCESSING':
                logger.info("Export job still processing...")
                time.sleep(60)
            elif status['status'] == 'QUEUED':
                logger.info("Export job in queue...")
                time.sleep(60)                
            elif status['status'] == 'FINISHED':
                logger.info("Export job finished!")
                return status
            else:
                msg = f"Export job {export_uuid} failed with status: {status['status']}"
                logger.error(msg)
                raise RuntimeError(msg)

    def download_and_parse_export_chunks(
            self,
            base_url: str,
            headers: dict,
            dataset: str,
            export_uuid: str,
            status: dict,
            cols_to_parse: Optional[dict] = None, 
            cols_to_drop: Optional[list] = None
        ) -> pd.DataFrame:
        """
        Download all available export chunks for a given Tenable VM export job and assemble
        them into a single DataFrame, with optional extraction of specific subfields from
        nested dictionary fields.

        Args:
            base_url (str): The base URL of the Tenable VM API.
            headers (dict): The HTTP headers for authentication and content type.
            dataset (str): The dataset type for the export job (e.g., 'vulns', 'assets').
            export_uuid (str): The UUID of the export job.
            status (dict): The export job status dictionary containing 'chunks_available'.
            fields_to_parse (dict): Mapping of nested-field name to a list of subfields
            fields_to_drop (list): List of columns to drop               
        Returns:
            pd.DataFrame: A DataFrame containing the combined records from all downloaded
                        chunks with extracted subfields.
        """

        if cols_to_parse is None:
            cols_to_parse = {}

        data = []
        chunks = status.get('chunks_available', [])
        num_chunks = len(chunks)
        logger.info(f"Download and assemble chunks to df. Total chunks available = {num_chunks}")

        count = 0 
        for chunk_id in chunks:
            chunk_url = f"{base_url}/{dataset}/export/{export_uuid}/chunks/{chunk_id}"
            try:
                chunk_resp = requests.get(chunk_url, headers=headers, stream=True)
                chunk_resp.raise_for_status()
                chunk_data = chunk_resp.json()
            except Exception as e:
                logger.error(f"Failed to download chunk {chunk_id}: {e}")
                raise

            for record in chunk_data:
                cleaned_record = record.copy()

                # Extract requested subfields from nested dictionaries
                for field, subfields in cols_to_parse.items():
                    nested_value = cleaned_record.get(field)

                    # Only process if the field is present and is a dict
                    if isinstance(nested_value, dict):
                        for subfield in subfields:
                            new_key = f"{field}_{subfield}"
                            cleaned_record[new_key] = nested_value.get(subfield)
                        # Drop the original nested field once we've extracted values
                        cleaned_record.pop(field, None)

                for col in cols_to_drop:
                    cleaned_record.pop(col, None)

                data.append(cleaned_record)
            count += 1
            logger.info(f"Downloaded chunk {count} of {num_chunks}" )

        df = pd.DataFrame(data)
        logger.info(f"Total records: {len(df)}")

        return df

    def download_parse_and_stream_export_chunks(
            self,
            base_url: str,
            headers: dict,
            dataset: str,
            export_uuid: str,
            status: dict,
            bucket_name: str,
            cols_to_parse: Optional[dict] = None, 
            cols_to_drop: Optional[list] = None
        ) -> None:
        """
        Download all available export chunks for a given Tenable VM export job and stream them into
        S3 bucket, with optional extraction of specific subfields from
        nested dictionary fields.

        Args:
            base_url (str): The base URL of the Tenable VM API.
            headers (dict): The HTTP headers for authentication and content type.
            dataset (str): The dataset type for the export job (e.g., 'vulns', 'assets').
            export_uuid (str): The UUID of the export job.
            status (dict): The export job status dictionary containing 'chunks_available'.
            bucket_name (str): Landing zone bucket name
            fields_to_parse (dict): Mapping of nested-field name to a list of subfields
            fields_to_drop (list): List of columns to drop               
        Returns:
            None
        """
        def _flush_buffer_to_s3(current_part: int, records: list) -> None:
            """Convert records to DataFrame and write to S3, then clear the list."""
            if not records:
                return

            logger.info(f"Loading PART_{current_part} to S3 with {len(records)} records")
            df = pd.DataFrame(records)

            # Main destination - Parquet
            filename = f"{s3_folder}/PART_{current_part}"
            self.rw_util.load_to_s3_multipart_datasets(bucket_name, df, filename, "PARQUET")

            # # *Uncomment if storing in test destination - CSV* 
            # test_filename = f"{test_s3_folder}/PART_{current_part}"
            # self.rw_util.load_to_s3_multipart_datasets(bucket_name, df, test_filename, "CSV")

            # Explicitly free DataFrame and clear buffer
            del df
            records.clear()


        if cols_to_parse is None:
            cols_to_parse = {}

        logger.info("Setup destinaation folder:")
        s3_folder = f"raw_data/TVM/CRMA_HIVE_TVM_{dataset.upper()}_{datetime.now(timezone.utc).strftime('%m%d%y_%H%M%S')}" 
        logger.info(f"Destination folder: {s3_folder}")

        # # *uncomment this if storing in test destination*
        # test_s3_folder = f"test/raw/TVM/CRMA_HIVE_TVM_{dataset.upper()}_{datetime.now(timezone.utc).strftime('%m%d%y_%H%M%S')}" 
        # #**

        buffer = []
        total_records = 0
        part=0
        MAX_RECORDS_PER_PART = 100000
        chunks = status.get('chunks_available', [])
        num_chunks = len(chunks)
        logger.info(f"Download and stream chunks to S3 bucket. Total chunks available = {num_chunks}")


        for idx, chunk_id in enumerate(chunks, start=1):
            chunk_url = f"{base_url}/{dataset}/export/{export_uuid}/chunks/{chunk_id}"
            try:
                chunk_resp = requests.get(chunk_url, headers=headers, stream=True)
                chunk_resp.raise_for_status()
                chunk_data = chunk_resp.json()
            except Exception as e:
                logger.error(f"Failed to download chunk {chunk_id}: {e}")
                raise

            for record in chunk_data:

                # Extract requested subfields from nested dictionaries
                for field, subfields in cols_to_parse.items():
                    nested_value = record.get(field)

                    # Only process if the field is present and is a dict
                    if isinstance(nested_value, dict):
                        for subfield in subfields:
                            new_key = f"{field}_{subfield}"
                            record[new_key] = nested_value.get(subfield)
                        # Drop the original nested field once we've extracted values
                        record.pop(field, None) 
                if cols_to_drop:
                    for col in cols_to_drop:
                        record.pop(col, None)               

                buffer.append(record)

            if len(buffer) >= MAX_RECORDS_PER_PART:
                _flush_buffer_to_s3(part, buffer) 
                part += 1                

            logger.info(f"Finished processing chunk {idx} of {num_chunks}.Deliting processed chunk to free memory")
            total_records += len(chunk_data)        
            del chunk_data

        if buffer:
            _flush_buffer_to_s3(part, buffer)
            logger.info(f"Download complate. Stored a total of {total_records} records in {part+1} files to landing zone")


    def get_timestamp_n_days_ago(
            self,
            n: int
            ) -> tuple[int, datetime]:
        """
        Calculate the Unix timestamp (in UTC) for a specified number of days ago from the current time.

        Args:
            days (int): The number of days to go back from the current UTC time.

        Returns:
            int: The Unix timestamp (UTC) representing the time exactly 'days' days ago.
        """
        dt = datetime.now() - timedelta(days=n)
        ts =int(dt.timestamp())
        logger.info(f"timestamp: {ts}")
        logger.info(f"datetime: {dt.isoformat()}")

        return ts
    def fetch_exportable_datasets(
            self,
            base_url: str, 
            headers: dict, 
            target: dict, 
            bucket_name: Optional[str] = None,
            cols_to_parse: Optional[dict] = None, 
            cols_to_drop: Optional[list] = None
            ) ->  pd.DataFrame:
        """
            Fetches exportable datasets from a TenableVM export API  and returns a processed DataFrame for some datasets and streams to S3 for others.
            Args:
                base_url (str): The base URL of the API endpoint.
                headers (dict): HTTP headers to include in the API request.
                target (dict): Parameters or payload specifying the dataset to fetch.
                bucket_name Optional[str] : Landing zone bucket name if streaming 
                cols_to_parse (Optional[dict]): Dictionary specifying columns to parse and their parsing functions or formats.
                cols_to_drop (Optional[list]): List of column names to drop from the resulting DataFrame.

            Returns:
                Optional[pd.DataFrame]: returns df for select datasts and streams s3 for others 
        """
        payload = target["payload"]  
        dataset = target["dataset"]   

        logger.info("Start or get existing export job")  
        export_uuid = self.get_or_create_tvm_export_job(
            base_url=base_url, 
            headers=headers, 
            payload=payload, 
            dataset=dataset
            )            

        logger.info("Pooling export job status...")    
        status = self.poll_export_status(
            base_url=base_url,
            headers=headers, 
            dataset=dataset, 
            export_uuid=export_uuid
            )
        
        num_chunks = len(status.get('chunks_available', []))
        if num_chunks == 0:
            logger.warning(F"No chunks download. Total chunks available = {num_chunks}.")
            return

        if bucket_name:
            logger.info("Download exported chunks and stream to S3 landing zone")
            self.download_parse_and_stream_export_chunks(
                base_url=base_url, 
                headers=headers, 
                dataset=dataset, 
                export_uuid=export_uuid, 
                status=status,
                bucket_name=bucket_name,
                cols_to_parse = cols_to_parse,
                cols_to_drop = cols_to_drop       
            )

        else:
            logger.info("Download exported chunks and assemble to dataframe")
            return self.download_and_parse_export_chunks(
                base_url=base_url, 
                headers=headers, 
                dataset=dataset, 
                export_uuid=export_uuid, 
                status=status,
                cols_to_parse = cols_to_parse,
                cols_to_drop = cols_to_drop       
            )

            self,

            "scan": ["uuid"],
            "port": ["port", "protocol"]
        }

        self.fetch_exportable_datasets(
            base_url = base_url,
            headers = headers, 
            target = target,
            bucket_name =bucket_name, 
            cols_to_parse = cols_to_parse,
            cols_to_drop = cols_to_drop

    def get_all_export_jobs(
            self,
            base_url: str, 
            headers: dict, 
            dataset: str
        ) -> pd.DataFrame:
        """
        Retrieve all export jobs for a specified dataset from the Tenable VM API.

        Args:
            base_url (str): The base URL of the Tenable VM API.
            headers (dict): The HTTP headers for authentication and content type.
            dataset (str): The dataset type to retrieve export jobs for (e.g., 'vulns', 'assets').

        Returns:
            pd.DataFrame: A DataFrame containing the list of export jobs for the specified dataset.
        """
        url = f"{base_url}/{dataset}/export/status"
        try:
            response = requests.get(url, headers=headers, timeout=720)
            if response.status_code == 200:
                jobs = response.json().get("exports", []) 

                # Convert to DataFrame
    
                logger.info(f"Total jobs found: {len(jobs)}")
                df = pd.DataFrame(jobs)[["uuid", "status"]]
                return df            
            else:
                logger.error(f"Error: {response.status_code} - {response.text}")

        except Exception as e:
            logger.error(f"Error fetching data: {e}")
            raise
  
  
  
    def cancel_export_jobs(
            self,
            base_url: str, 
            headers: dict, 
            dataset: str, 
            uuids: list[str]
        ) -> None:
        """
        Cancel export jobs for a specified dataset and list of UUIDs using the Tenable VM API.
        Args:
            base_url (str): The base URL of the Tenable VM API.
            headers (dict): The HTTP headers for authentication and content type.
            dataset (str): The dataset type for which to cancel export jobs (e.g., 'vulns', 'assets').
            uuids (list[str]): A list of job UUIDs to cancel.

        Returns:
            None
        """
        for uuid in uuids:
            url = f"{base_url}/{dataset}/export/{uuid}/cancel"
            try:
                response = requests.post(url, headers=headers)
                if response.status_code == 200:
                    print(f"Deleted job {uuid}")
                else:
                    print(f"Failed to delete job {uuid}: {response.status_code} - {response.text}")
            except Exception as e:
                logger.error(f"Error deleting job {uuid}: {e}")


    def load_tvm_data_to_db(
            self,
            engine: Engine,
            bucket_name: str,
            load_config: dict,
            schema: str,
            backfill_date: Optional[str] = None
    ) -> None:
        """
        Loads data to desttination database
        
        Parameters:
            engine (create_engine): Database connection engine
            bucket (str): Name of the S3 bucket to fetch the data from
            load_config (dict): The parameters of the dataset to store to database
            schema(str): The database schema in which data is loaded.
            backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525)
        """
        dataset = load_config["dataset"]
        logger.info(f"Preparing to load {dataset} tables to database...")

        classes = ["CLEAN", "QUARANTINE", "WARNING"]

        for cls in classes:
            prefix = f"processed/TVM/{cls}/CRMA_HIVE"
            base_name = f"{prefix}_{dataset}_{cls}" 

            if backfill_date:
                df = self.rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")                
                if df is not None:
                    df["etl_date"] = datetime.strptime(backfill_date, "%m%d%y").strftime("%Y-%m-%d")

            else:         
                df = self.rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")
                if df is not None:
                    df["etl_date"]=datetime.now().date() 
            
            if cls == "CLEAN":
                if df is None:
                    raise ValueError("Dataset not found")
                        
                table_name = f"crma_hive_{dataset.lower()}"
                logger.info(f"Loading {table_name} ...")

                if load_config["load_strategy"]=="upsert":  
                    pk_columns = load_config["pk_columns"]    
                    logger.info(f"Dataset pk_columns: {pk_columns}")          
                    self.rw_util.bulk_upsert_with_copy(
                        engine=engine,
                        df=df,
                        schema=schema,
                        target_table=table_name,
                        pk_columns=pk_columns
                        )
                    logger.info(f"Perfomming post load validation on table '{table_name}' in '{schema}' schema")

    def load_tvm_data_to_db_hist_tables(
            self,
            engine: Engine,
            bucket_name: str,
            dataset: str,
            target_table:str,
            load_control_table:str,
            schema: str,
            backfill_date: Optional[str] = None
    ) -> None:
        """
        Loads data to desttination database
        
        Parameters:
            engine (create_engine): Database connection engine
            bucket (str): Name of the S3 bucket to fetch the data from
            dataset (str): The dataset to store to database
            backfill_date (str): Date string used to find which integraiton source file to re-procecss (e.g., 121525)
        """

        prefix = f"processed/TVM/CLEAN/CRMA_HIVE"
        base_name = f"{prefix}_{dataset}_CLEAN"   
        etl_date = datetime.now().date() 

        if backfill_date:
            df = self.rw_util.fetch_file_by_date(bucket_name, prefix, base_name, backfill_date, "PARQUET")
            if df is not None:
                df["etl_date"] = datetime.strptime(backfill_date, "%m%d%y").strftime("%Y-%m-%d") 

        else:    
            df = self.rw_util.fetch_file_from_s3(bucket_name, prefix, base_name, "PARQUET")
            if df is not None:
                df["etl_date"]=etl_date
        
        if df is None:
            raise ValueError("Dataset not found")
        
        db_load_flag = self.rw_util.bulk_load_with_copy_daily_idempotent(
            etl_date=etl_date,
            engine=engine,
            df=df,
            schema=schema,
            target_table=target_table,
            load_control_table=load_control_table,
        )
        if db_load_flag:
            logger.info(f"Perfomming post load validation on table '{target_table}' in '{schema}' schema")
            if self.rw_util.post_load_validation_hist_tables(
                engine=engine, 
                df=df, 
                schema_name=schema, 
                table_name=target_table,
                backfill_date=backfill_date
                ):
                logger.info(f"Post load validation check passed for table '{target_table}' in '{schema}' schema")             
            else:
                raise RuntimeError(f"Post load validation check failed for table '{target_table}' in '{schema}' schema")

    def classify_operating_system(self, df: pd.DataFrame) -> pd.Series:
        """
        Classifies a device's operating sytem based on tvm_vulns.asset_operating_system and tvm_vulns.asset_hostname fields

        Parameters:
            df(pd.Dataframe): Must containe columns 'asset_operating_system' and 'asset_hostname'

        Returns:
            pd.Series: Seris containing classifed OS values(windows, linux, juniper, esxi, unkonwn)
        """
        
        # define hostname prefixes that indicate a windows device
        windows_prefixes = ("LI", "bx", "si", "bf", "di", "ei", "lo", "lf", "xi", "ti", "ei", "bi", "ci")

        def _classify_row(row):
            os_array = row.get("asset_operating_system")
            hostname = row.get("asset_hostname")

            # handle array os field
            if os_array is None or (hasattr(os_array, "__len__") and ( len(os_array) == 0 or pd.isna(os_array).all())):
                # use hostname logic
                if pd.isna(hostname) or hostname is None:
                    return "unknown"
                hostname = str(hostname).strip().lower()
                if hostname.startswith(tuple(p.lower() for p in windows_prefixes)):
                    return "windows"
                return "unknown"
            
            # convert array to string
            os_strings = [str(item).strip() for item in os_array if item is not None and str(item).strip()]
            os_str = " ".join(os_strings).lower()
            if "vmware" in os_str:
                return "esxi"
            elif "microsoft" in os_str:
                return "windows"
            elif "linux" in os_str:
                return "linux"
            elif "cisco" in os_str:
                return "cisco"
            elif "juniper" in os_str:
                return "juniper"         
            else:
                return "unkonwn"               
        return df.apply(_classify_row, axis = 1)
            
    def classify_device_type(self, df: pd.DataFrame) -> pd.Series:
        """
        Classifies a device's type based on "axonius.devices_type", "tvm_vulns.asset_operating_system", and "tvm_vulns.asset_hostname"

        Parameters:
            df(pd.Dataframe): Must containe columns "devices_type" and "asset_operating_system" and "asset_operating_system"

        Returns:
            pd.Series: Seris containing classifed device types values (workstation, server, network, hypervisor, unknown)
        """
        workstation_prefixes =  ("LI", "bx", "si", "bf", "di", "ei", "lo", "lf", "xi", "ti", "ei", "bi", "ci")

        def _classify_row(row):
            device_type_array = row.get("devices_type")
            os_array = row.get("asset_operating_system")
            hostname = row.get("asset_hostname")

            def _array_to_str(arr):
                if arr is None:
                    return ""
                if hasattr(arr, "__len__") and len(arr) == 0:
                    return ""
                items = [str(item).strip() for item in arr if item is not None and str(item).strip()]
                return " ".join(items).lower()
            
            os_str = _array_to_str(os_array)
            
            # case1: os contains vmware
            if "vmware" in os_str:
                return "hypervisor"
            
            # case2: devices_type is null
            device_type_str = _array_to_str(device_type_array)
            if not device_type_str:
                # use os if available
                if "microsft" in os_str and ("enterprise" in os_str or "pro" in os_str):
                    return "workstation"
                elif "linux" in os_str or "server" in os_str:
                    return "server"
                elif "cisco" in os_str or "juniper" in os_str:
                    return "network"
                elif "vmware" in os_str:
                    return "hypervisor"     

                # use hostname
                if hostname and not pd.isna(hostname):
                    hostname_str = str(hostname).lower()
                    if hostname_str.startswith(tuple(p.lower() for p in workstation_prefixes)):
                        return "workstation"
                return "unknown"

                
            # case3: devices_type is not null
            if any(x in device_type_str for x in ["desktop", "workstation", "laptop"]):
                return "workstation"
            elif "server" in device_type_str:
                return "server"
            elif "network" in device_type_str:
                return "network"
            else:
                return "unknown"
        return df.apply(_classify_row, axis=1)
